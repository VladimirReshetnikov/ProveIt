\begin{abstract}
In Gowers's proof of Szemer\'edi's theorem, the structural step of Section~16
(the single-piece lift) must produce one multilinear map that agrees with the
\emph{remainder} $\varphi''$ on a dense piece of a box. The remainder is a
signed sum over the $2^k-1$ non-top vertices of the cube $\{0,1\}^k$, and each
vertex contributes a function that lives on a lower-dimensional face. If the
vertices are handled one after another, each at the density left by the
previous one, the degrees of the density losses multiply, and the resulting
envelope grows like $3^k$ in the doubly-logarithmic scale. That outgrows the
printed budgets $2^{d+10}$ (Theorem~18.1) and $2^{L+9}$ (Theorem~18.2) at
degree $10$ and length $14$. We show that the vertex \emph{covers} can be
composed at a precision fixed in advance, with a single pigeonhole at the end.
The losses then add, the envelope grows like $2^k$, and Theorems~18.1, 18.2
and Corollary~18.7 hold in every degree and length, at the printed constants
and with some room to spare. Every statement below is formalized in Lean
(module names are given in \S\ref{sec:lean}).
\end{abstract}

\section{Covers}

Throughout, $N$ is prime, a \emph{box} is a product of arithmetic progressions
in $\mathbb{Z}_N$ with a common difference, and $w(P)$ is the least side
length of a box $P$. A map is \emph{multilinear} in the sense of the paper.

\begin{definition}[Cover]
Let $\Gamma\subseteq\mathbb{Z}_N^d\times\mathbb{Z}_N$ be a relation and
$\Qb,\Eb\colon(0,1]\to\mathbb{R}$. We say $\Gamma$ is \emph{covered with controls
$(\Qb,\Eb)$} if for every $\theta\in(0,1]$ and every proper box $P$ there are
\begin{itemize}
\item a partition of $P$ into proper boxes $Q_1,\dots,Q_M$ with
  $w(Q_j)\ge w(P)^{\Eb(\theta)}$,
\item a set $H\subseteq P$ with $|H|\ge(1-\theta)|P|$,
\item for each $j$ at most $\Qb(\theta)$ multilinear maps
  $\mu_{j,1},\dots,\mu_{j,q}$,
\end{itemize}
such that every $(x,y)\in\Gamma$ with $x\in Q_j\cap H$ satisfies
$y=\mu_{j,i}(x)$ for some $i$.
\end{definition}

The precision $\theta$ is chosen by the user of the cover \emph{before} it is
applied, and the loss $\theta|P|$ is additive. A \emph{provider}, by contrast,
takes a set of density $t$ and returns a piece of density $c(t)$; composing
providers composes the functions $c$.

\section{Joint covers}

\begin{lemma}[Joint cover]\label{lem:joint}
Let $E$ be a finite index set, and for $e\in E$ let $f_e$ be a function on a
domain $D_e\subseteq\mathbb{Z}_N^d$ whose graph is covered with controls
$(\Qb_e,\Eb_e)$, with $\Qb_e,\Eb_e\ge0$. Fix precisions $\theta_e\in(0,1]$.
Then every proper box $P$ has a partition into proper boxes $R_1,\dots,R_M$, a
set $G\subseteq P$ and, for each $j$, at most $q\le\prod_e\Qb_e(\theta_e)$
tuples $(\nu_{j,i,e})_{e\in E}$ of multilinear maps, such that
\[
  |G|\ge\Bigl(1-\sum_{e}\theta_e\Bigr)|P|,\qquad
  w(R_j)\ge w(P)^{\prod_e\Eb_e(\theta_e)},
\]
and every $x\in R_j\cap G$ lying in all domains $D_e$ has an index $i$ with
$f_e(x)=\nu_{j,i,e}(x)$ for all $e$.
\end{lemma}

\begin{proof}
Induction on $E$. For $E=\emptyset$ take the trivial partition, $G=P$ and one
empty tuple. For $E\cup\{a\}$, apply the induction hypothesis to $P$, obtaining
cells $Q_j$, a good set $G$ and tuples $\nu_{j,i}$. Apply the cover of $f_a$ at
precision $\theta_a$ to every cell $Q_j$; this gives sub-cells $R_{j,l}$, good
sets whose union $G'$ satisfies $|G'|\ge(1-\theta_a)|P|$ (the cells partition
$P$), and maps $\mu_{j,l,i'}$. Take the sub-cells $R_{j,l}$ as the new
partition, the good set $G\cap G'$, of size at least
$|G|+|G'|-|P|\ge(1-\sum_E\theta_e-\theta_a)|P|$, and the product tuples
$(\nu_{j,i},\mu_{j,l,i'})$. The width bound follows from
$w(R_{j,l})\ge w(Q_j)^{\Eb_a}\ge w(P)^{\Eb_a\prod_E\Eb_e}$.
\end{proof}

\begin{corollary}[One piece]\label{cor:piece}
In the setting of Lemma~\ref{lem:joint}, put $s=\sum_e\theta_e$ and
$Q=\prod_e\Qb_e(\theta_e)$. If $H\subseteq P$ lies in every domain $D_e$ and
$|H|\ge\theta|P|$ with $\theta>s$, then some cell $R$ and some maps $\nu_e$
satisfy
\[
  \bigl|\{x\in H\cap R: f_e(x)=\nu_e(x)\text{ for all }e\}\bigr|
  \ge\frac{\theta-s}{Q}\,|R| .
\]
\end{corollary}

\begin{proof}
$|H\cap G|\ge(\theta-s)|P|$, so some cell $R$ has
$|H\cap G\cap R|\ge(\theta-s)|R|$. Each of its points is covered by one of at
most $Q$ tuples; the most popular tuple covers a $1/Q$ fraction.
\end{proof}

The only density loss is the single factor $1/Q$, and $Q$ is a product of
counts evaluated at \emph{fixed} precisions. If $\Qb_e(\theta)\le(C/\theta)^{D_e}$,
the density has degree $1+\sum_e D_e$ in $1/\theta$: the degrees add.

\section{The remainder of the single-piece step}

On the good domain of the step, the remainder is
$\varphi''(x)=-\sum_{e\ne\mathbf 1}(-1)^{k+|e|}\varphi_e(x)$, where $\varphi_e$ is
the translated cube vertex $e$, a pullback of $\varphi$ to the face spanned by
the directions $S_e$, of dimension $|e|+1$. By induction on the dimension, the
face carries a cover of degree $D_{|S_e|}$. It lifts to the ambient box without
any loss in the precision: the count becomes $\max(\Qb,3^{k+1})$ and the
exponent $\Eb/16^{k+1-|S_e|}$. Applying Corollary~\ref{cor:piece} with
$\theta_e=\theta_2/2^{k+1}$ to the dense good set (density $\theta_2$) gives one
map $M''=-\sum_e(-1)^{k+|e|}\nu_e$ on a piece of density
\[
  \rho_1=\frac{\theta_2/2}{\prod_{e\ne\mathbf 1}\max\bigl(\Qb_{|S_e|}(\theta_2/2^{k+1}),3^{k+1}\bigr)}
\]
and width at least $N^{\prod_e\Eb_e/16^{k+1-|S_e|}}$.

\begin{remark}[Nesting versus joint covers]
With the class degrees $D_l=2^{68\cdot2^l}$, nesting one provider per vertex
gives $\log_2\deg\rho_1^{-1}\approx\sum_{e}68\cdot2^{|e|+1}=136\cdot3^k$. The
joint cover gives $\log_2\bigl(1+\sum_eD_{|S_e|}\bigr)\le68\cdot2^k+k$.
\end{remark}

\section{The envelope}

Write $X=2/\alpha$. The inverse induction carries bounds
$\beta^{-1},\sigma^{-1}\le X^{2^{\mu}}$ from degree $d-1$ to degree $d$. With the
joint-cover remainder, one step in dimension $k$ costs
\[
  \mu_{\mathrm{new}}=\mu+\underbrace{(68\cdot2^k+\mu_0+k+3)}_{\rho_1}
  +2\cdot68\cdot2^k+O(k^2),
\]
where $\mu_0$ bounds the base budget of the cover controls. With the standard
$\mu_0\approx64\cdot2^k$ one step costs $268\cdot2^k+O(k^2)$. Hence

\begin{theorem}
For every $k\ge1$ the Theorem~18.1 envelope satisfies $\mu_k\le1000\cdot2^k$,
and the Theorem~18.2 envelope (which also carries thresholds) satisfies
$\mu_k\le1000\cdot2^k$ as well. Both are well inside the printed budgets
$2^{d+10}=4096\cdot2^{d-2}$ and $2^{L+9}=8192\cdot2^{L-4}$. Consequently
Theorems~18.1 and 18.2 and Corollary~18.7 hold in every degree and length.
\end{theorem}

\section{Sharper constants}

The same proofs give smaller tower constants. Moreover, the class base
$V=1+\sum_{j<k}(K_j+p_j)$ satisfies $V\le2^{2^{2^{k+4}+1}}$, so the base budget
$\mu_0=2^{k+5}+9$ suffices; one step then costs $236\cdot2^k+O(k^2)$ and the
envelopes grow like $472\cdot2^k<512\cdot2^k$.

The $W(k,r)$ row is Theorem~18.2 at density $\delta=1/r$: in an $r$-colouring of
$[1,N]$ one colour class has at least $N/r$ elements. This gives Gowers's
van der Waerden bound, sharpened from $k+9$ to $k+5$. The headline
Theorem~1.3 follows from Theorem~18.2 by a calculation in which the constant
enters only through $2^{k+9}$; repeated with $k+5$, it gives the density
exponent $2^{-2^{k+5}}$.

With the printed $\theta_1$ the method stops at the tight-base constants. The
envelope recursion grows like $472\cdot2^k$, and $c=6$ for Theorem~18.1 (or
$c=4$ for Theorem~18.2) needs a step cost below $128\cdot2^k$, while the
vertex covers and the two slice controls alone cost $204\cdot2^k$. The last
column of the table comes from changing $\theta_1$ itself
(\S\ref{sec:fejer}).

For Theorem~18.2 at $c=5$ one more point matters. The modulus threshold of
the inverse step contains two pure constants, the Section~5 refinement
constant $\le2^{2^{40d^3+2}}$ and a boundary constant $\le2^{2^{1083}}$. A
cruder bound charges them at the level of powers of $X$, which puts a cubic
term $40(k+2)^3$ into the envelope and blocks $L+5$ for $L\le10$. The
threshold is only needed below $\exp(\exp(X^{2^\mu}))$, and
$2^{2^m}\le\exp(X^{2^j})$ as soon as $m\le2^j$. At that level the
constants cost only $k+16$, and $L+5$ holds in every length.

\section{A Fej\'er selection and the constants $d+6$, $L+4$}\label{sec:fejer}

The class degree $68\cdot2^l$ and the base budget both come from the
Section~16 density $\theta_1=(\theta\gamma/4)^{2^{2^{k+5}}}$. Its double
exponential is created in Lemma~15.5. A point is kept with probability
$(1+\cos)/2$, which gives a respected arrangement of $M=2^{k+4}$ vertices an
advantage of only $1+2^{1-M}$ per stage, so $2^{M-1}$ stages are needed.

\emph{One Fej\'er stage.} Keep a point with the Fej\'er weight
$F_m(u)=|\sum_{j\le m}e(ju/N)|^2/(m+1)^2$ of order
$m+1=M\lceil6/(\alpha\eta)\rceil$. Averaged over the coefficients, the
survival probability of an arrangement is a sum over the integer vectors
$\eta\in[-m,m]^M$ annihilating all its monomials. For arrangements that are
not $m$-degenerate, these are the multiples $j\varepsilon$ of the parity
vector. They contribute $\sum_{|j|\le m}(1-|j|/(m+1))^M\ge T$ for respected
arrangements and only $j=0$ otherwise. Lemma~15.4 extends to coefficients in
$[-m,m]$, since Corollary~15.2 holds for every coefficient vector, and it
bounds the $m$-degenerate arrangements. The resulting Lemma~15.5 has exponent
$2^{k+4}(k+6)$, and Lemma~15.6 has
\[
  \theta_1'=(\theta\gamma/4)^{F},\qquad F\le2^{5k+14},
\]
in place of $2^{2^{k+5}}$.

\emph{The tight radius.} The Bohr radius $\zeta=2^{-s}$ only has to satisfy
Lemma~16.5's condition, which needs $s\ge2^{83}/(\theta_1/4)^{11}$. With
$\theta_1'$ this is $s'\le(2/\theta\gamma)^{22F+105}$, instead of the printed
$(2/\theta\gamma)^{2^{2^{k+6}}}$.

\emph{Propagation.} With these inputs the dimension-two class has degree
$2^{46}$ (printed $2^{132}$). The split-class recursion keeps
$\log_2D_l\le22\cdot2^l$, so the class covers have degree $2^{23\cdot2^l}$
in place of $2^{68\cdot2^l}$. Since $\theta_1'\ge\theta_1$, the budget, $\theta_2$, $\delta$
and the radius of Section~18 are at least the printed ones, and every printed
fixed-parameter bound transfers unchanged. One step of the inverse induction
then costs
\[
  \mu_{\mathrm{new}}=\mu+\underbrace{23\cdot2^k}_{\text{vertices}}
  +\underbrace{32\cdot2^k}_{\mu_0}+\underbrace{2\cdot23\cdot2^k}_{\text{slices}}
  +O(k^2)=\mu+101\cdot2^k+O(k^2),
\]
and both envelopes are at most $230\cdot2^k$ for $k\ge4$, below the
$256\cdot2^k$ of $c=6$ and $c=4$.

\begin{theorem}\label{thm:fejer}
Theorem~18.1 holds in every degree $d$ with exponent $\alpha^{2^{2^{d+6}}}$, and
Theorem~18.2 and Corollary~18.7 hold in every length $L$ with $L+4$ in the
threshold. Hence $W(k,r)\le2^{2^{r^{2^{2^{k+4}}}}}$, and Theorem~1.3 holds with
density exponent $2^{-2^{k+4}}$.
\end{theorem}

\emph{Large degrees.} The proved class bound $23\cdot2^l$ is loose. The class
recursion only needs slack that grows additively, and indeed
\[
  2^{l+6}a_l\le2^{14\cdot2^l},\qquad D_l\le2^{14\cdot2^l+3(l+4)^2}
\]
in every dimension $l\ge2$, since $3(l+5)^2-3(l+4)^2=6l+27$ covers the
$5l+22$ that one step adds. Moreover the Fej\'er $\theta_1$ no longer
forces the base budget $32\cdot2^k$: the class-base term $16\cdot2^k$
suffices. One step then costs $58\cdot2^k+O(k^2)$, and the envelopes stay
below $128\cdot2^k$ from $k=10$ on.

\begin{theorem}\label{thm:fejer5}
Theorem~18.1 holds with exponent $\alpha^{2^{2^{d+5}}}$ at every degree
$d\ge12$, and Theorem~18.2 and Corollary~18.7 hold with $L+3$ at every
length $L\ge14$; in particular $W(k,r)\le2^{2^{r^{2^{2^{k+3}}}}}$ for $k\ge14$,
and Theorem~1.3 holds with density exponent $2^{-2^{k+3}}$ for $k\ge14$.
\end{theorem}

The small cases stay at $d+6$ and $L+4$ because of the class recursion's
transient near $l=4$, where the family exponent squares,
$a_{l+1}\approx2^{l+5}a_l^2$.

\emph{The dimension-two family exponent.} Since the family exponent squares,
$\log_2a_l\approx(\log_2a_2/4)\,2^l$, so the constant $14$ above is set by
$a_2=2^{46}$. That value only reflects the passage from a power class, which
sets the family exponent equal to the parameter degree. The dimension-two
controls are polynomials of fixed degree in the number $M=n\,m$ of pieces:
the base controls are linear in $M$. In the two bases $\nu=n+1$ and
$W=2V/(\gamma\theta\rho)$:
\begin{itemize}
\item the count is at most $\nu^{11}W^{2^{41}}$;
\item the inverse width exponent is at most $\nu^{29}W^{2^{46}}$.
\end{itemize}
So dimension two is a split class with $a_2=32$ and $D_2=2^{48}$, and the
invariants become
\[
  2^{l+6}a_l\le2^{4\cdot2^l},\qquad D_l\le2^{4\cdot2^l+3(l+4)^2+30}.
\]
The class degree is then $4\cdot2^l+3(l+4)^2+32$.

At this class degree the density degree of the step is about $28\cdot2^k$.
That is below the $32\cdot2^k$ which the $\eta$ step of the printed
envelope asks of it, but the $\eta$ step only absorbs the constant
$2^{2(k+2)^3}\,2^{k+1}(2/\alpha)^2$, and $3(k+4)^2$ suffices for that. One
step then costs $28\cdot2^k+O(k^2)$, so the Theorem~18.1 envelope tends to
$56\cdot2^k<64\cdot2^k$.

\begin{theorem}\label{thm:fejer4}
Theorem~18.1 holds with exponent $\alpha^{2^{2^{d+4}}}$ at every degree
$d\ge13$, and with $\alpha^{2^{2^{d+5}}}$ at every degree $d\ge9$.
Theorem~18.2 and Corollary~18.7 hold with $L+3$ at every length $L\ge11$; in
particular $W(k,r)\le2^{2^{r^{2^{2^{k+3}}}}}$ and Theorem~1.3 holds with
density exponent $2^{-2^{k+3}}$ for $k\ge11$.
\end{theorem}

This Theorem~18.2 envelope cannot reach $L+2$: its threshold branch alone is
$64\cdot2^k+8>64\cdot2^k$ (but see Theorem~\ref{thm:fejer2}). Nor can $d+3$ be
reached, since the base budget contributes $16\cdot2^k$ to every step.

\emph{Exact small dimensions.} In small dimensions the slack $3(l+4)^2+32$
dominates the class degree, but there the split degrees are explicit numbers.

\begin{theorem}\label{thm:fejer5all}
Theorem~18.1 holds with exponent $\alpha^{2^{2^{d+5}}}$ in every degree $d$,
and with $\alpha^{2^{2^{d+4}}}$ at every degree $d\ge11$. Theorem~18.2 and
Corollary~18.7 hold with $L+3$ in every length $L$. Hence
$W(k,r)\le2^{2^{r^{2^{2^{k+3}}}}}$ for every $k$, and Theorem~1.3 holds with
density exponent $2^{-2^{k+3}}$.
\end{theorem}

\emph{The Bohr radius.} The threshold branch $64\cdot2^k$ of the
Theorem~18.2 envelope comes from the inverse tight Bohr radius, which was
bounded by comparison with the printed $\theta_1$ as
$\exp(X^{2^{2^{k+6}+1}})$. Under the Fej\'er $\theta_1$ the radius is
polynomial in $\theta_1'^{-1}$, so $\zeta^{-1}\le\exp(X^{2^{5k+21}})$. The
threshold branch then drops to $16\cdot2^k+5k+25$, which is the threshold of
Lemma~15.6. Every step costs at most $\max(\mu+28\cdot2^k+O(k^2),\,
32\cdot2^k+9)$.

\begin{theorem}\label{thm:fejer2}
Theorem~18.2 and Corollary~18.7 hold with $L+2$ at every length $L\ge13$; in
particular $W(k,r)\le2^{2^{r^{2^{2^{k+2}}}}}$ and Theorem~1.3 holds with
density exponent $2^{-2^{k+2}}$ for $k\ge13$.
\end{theorem}

\emph{The base budget.} Every step also pays the base budget
$\mu_0\approx16\cdot2^k$, and both of its sources are comparison bounds.
Under the Fej\'er $\theta_1$ the fixed parameters (budget, $\theta_2/2$,
$\delta$) are above $X^{-2^{5k+21}}$. The explicit family constants satisfy
$K_k+p_k\le2^{2^{4k+16}}$, a double logarithm linear in $k$, so
$V\le2^{2^{4k+13}}$. With $\mu_0=5k+23$, one step costs only
$12\cdot2^k+O(k^2)$ beyond the table, and both envelopes tend to
$24\cdot2^k$.

\begin{theorem}\label{thm:fejer3}
Theorem~18.1 holds with exponent $\alpha^{2^{2^{d+3}}}$ at every degree
$d\ge11$, and with $\alpha^{2^{2^{d+4}}}$ at every degree $d\ge8$.
Theorem~18.2 and Corollary~18.7 hold with $L+1$ at every length $L\ge13$, and
with $L+2$ at every length $L\ge10$. In particular
$W(k,r)\le2^{2^{r^{2^{2^{k+1}}}}}$ and Theorem~1.3 holds with density exponent
$2^{-2^{k+1}}$ for $k\ge13$.
\end{theorem}

\section{Theorem 16.2 and Corollary 16.11 with polynomial controls}

The printed Theorem~16.2 restricts a relation with the product property to a
set of density $1-\theta$ on which it is multiply linear, with graph count
$q(r^{-1}\rho,\gamma,k)^r$ at precision $\rho$, where
$r=\gamma^{-2}s(\theta,\gamma,k)$ and $s=(2/\theta\gamma)^{2^{2^{k+6}}}$. This
count is exponential in a polynomial of $1/\theta$. The covers built for the
dimension induction give a count that is \emph{polynomial} in $1/(\gamma\theta\rho)$:
\[
  Q(\gamma,\theta,\rho)=\max\Bigl(1,\Bigl(\frac{4V}{\gamma\theta\rho}\Bigr)^{2^{68\cdot2^k}}\Bigr),
  \qquad V=2^{2^{2^{k+4}+1}},
\]
with width exponent $Q^{-1}$, in every dimension $k\ge1$
(\lean{theorem\_16\_2\_polynomial}). This form implies the printed one
(\lean{theorem\_16\_2\_polynomial\_implies\_printed}).

The same holds for Corollary~16.11. For a function that is not uniform of
degree $k+2$, the single-piece step with the joint-cover remainder gives a
box of dimension $k+1$, of width $N^e$, on which a multilinear frequency is
large on an $e$ fraction, with
\[
  e^{-1}\le(2/\alpha)^{2^{236\cdot2^k+k+24}}
\]
for every $k\ge2$ (\lean{corollary\_16\_11\_polynomial}). The printed exponent
is $(\alpha/8)\,c(r^{-1}\alpha/8,\alpha/2)^{r}$ with $r$ polynomial in
$1/\alpha$, so its reciprocal is exponential in a polynomial of $1/\alpha$.

With the Fej\'er classes of \S\ref{sec:fejer} both exponents shrink:
\begin{itemize}
\item the graph count becomes
  $\max(1,(4V/\gamma\theta\rho)^{2^{23\cdot2^k}})$\\
  (\lean{theorem\_16\_2\_polynomial\_fejer});
\item the corollary has $e^{-1}\le(2/\alpha)^{2^{101\cdot2^k+k+24}}$\\
  (\lean{corollary\_16\_11\_polynomial\_fejer}).
\end{itemize}
With the dimension-two family exponent $32$ (Theorem~\ref{thm:fejer4}) they
shrink further:
\begin{itemize}
\item the graph count becomes
  $\max(1,(4V/\gamma\theta\rho)^{2^{4\cdot2^k+3(k+4)^2+32}})$\\
  (\lean{theorem\_16\_2\_polynomial\_fejerS});
\item the corollary has $e^{-1}\le(2/\alpha)^{2^{32\cdot2^k+9(k+4)^2+k+120}}$\\
  (\lean{corollary\_16\_11\_polynomial\_fejerS}).
\end{itemize}

\section{Further questions}

\begin{enumerate}
\item \emph{Joint slices.} The single-piece step applies its slice control
  twice in succession, $\theta\,c(c(\theta))$, and so pays $2\cdot68\cdot2^k$
  in the step cost. Composing the two slice covers jointly, as in
  Lemma~\ref{lem:joint}, should cost about $68\cdot2^k$. The envelope would
  drop from about $472\cdot2^k$ to about $336\cdot2^k$. This changes none of
  the constants in the table.
\item \emph{Smaller offsets.} With the polynomial base budget
  (Theorem~\ref{thm:fejer3}) the envelopes tend to $24\cdot2^k$. That is
  below $32\cdot2^k$ (so $d+3$, $L+1$) but above $16\cdot2^k$. The
  remaining cost is the class degree beyond dimension $12$: there
  $4\cdot2^l$ is used, while the true constant is about $2.4$. A sharper
  split invariant ($\approx2.4\cdot2^l$) would give $7.2\cdot2^k$ per step and
  an envelope $14.4\cdot2^k<16\cdot2^k$, hence $d+2$ and $L+0$ for large $d$
  and $L$. Below the thresholds $11$ and $13$ the small dimensions dominate.
\item \emph{Fixed-precision slices with the Fej\'er classes.} Composing the
  two slice controls jointly would lower the step cost from $101\cdot2^k$ to
  about $78\cdot2^k$. That is still above $64\cdot2^k$, so it changes no
  constant in the table.
\end{enumerate}
