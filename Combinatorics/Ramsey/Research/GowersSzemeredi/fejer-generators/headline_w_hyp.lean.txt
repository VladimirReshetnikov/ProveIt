/-- The polynomial bound on the family constants: `K, p ≤ 2^(2^(4k+16))`. -/
def genMuKW (k : Nat) : Nat := 4 * k + 16

/-- The standing hypotheses at the polynomial base budget. -/
theorem tightHyp_of_profiles_fejerT (Kc pc : Nat → Nat) (hK : ∀ k, 2 ≤ Kc k) (hp : ∀ k, 0 < pc k)
    (hconst : ∀ k, (Kc k : Real) + pc k ≤ 2 ^ ((2 : Nat) ^ ((2 : Nat) ^ (k + 5)))) {k : Nat}
    (hk : 2 ≤ k) :
    (⟨classV Kc pc k, k, Kc k, pc k, tightMu0T k⟩ : GenParamsT).CHyp (tightMuV k) (genMuK k) := by
  have hKr : (Kc k : Real) ≤ 2 ^ ((2 : Nat) ^ genMuK k) := by
    have := hconst k; unfold genMuK; linarith [Nat.cast_nonneg (α := Real) (pc k)]
  have hpr : (pc k : Real) ≤ 2 ^ ((2 : Nat) ^ genMuK k) := by
    have := hconst k; unfold genMuK; linarith [Nat.cast_nonneg (α := Real) (Kc k)]
  refine ⟨one_le_classV Kc pc k, classV_le_poly Kc pc hconst k, le_rfl, ?_, hK k, ?_, hp k, ?_⟩
  · show k ≤ 2 ^ tightMu0T k
    calc k ≤ 2 ^ k := Nat.lt_two_pow_self.le
      _ ≤ 2 ^ tightMu0T k := Nat.pow_le_pow_right (by norm_num) (by
          unfold tightMu0T; have := five_k_le_fixDegQ k
          have := le_max_right (tightMuV k) (fixDegQ k); omega)
  · have : ((Kc k : Nat) : Real) ≤ ((2 ^ 2 ^ genMuK k : Nat) : Real) := by push_cast; exact hKr
    exact_mod_cast this
  · have : ((pc k : Nat) : Real) ≤ ((2 ^ 2 ^ genMuK k : Nat) : Real) := by push_cast; exact hpr
    exact_mod_cast this
