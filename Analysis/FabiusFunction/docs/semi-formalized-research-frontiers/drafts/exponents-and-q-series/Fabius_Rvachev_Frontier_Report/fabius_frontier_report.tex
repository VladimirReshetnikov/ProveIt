\begin{abstract}
The current \texttt{ProveIt/Analysis/FabiusFunction/docs} corpus develops the Fabius and
Rvachev functions from probability, functional equations, exact dyadic arithmetic,
$q$-calculus, infinite sinc products, Thue--Morse cancellation, Legendre expansions,
Lambert--$W$ endpoint asymptotics, and formalized analysis.  Its integrated frontier
volume explicitly leaves open, for the geometric-uniform family, the density
interpretation at negative parameters, centered affine transport, log-concavity and
central plateaux, explicit Bernoulli cumulants and Bell moments in the generic
parameter, and Gaussian-limit statements.

This report gives paper-level solutions or substantial advances for those gaps.  For
\(0<r<1\), if
\[
 X_q=(1-q)\sum_{n\ge0}q^nU_n,
 \qquad U_n\stackrel{\mathrm{iid}}\sim\operatorname{Unif}(0,1),
\]
then the central identity is
\[
 X_{-r}-\frac12\ \doteqL\
 \frac{1+r}{1-r}\left(X_r-\frac12\right).
\]
It gives the exact negative-parameter CDF, quantile, support, smooth density, entropy
scaling, sinc product, and all derivative transports.  It also diagnoses a sign issue
that was deliberately left uninterpreted in the repository: the previously selected
total function at \(q<0\) is the \emph{negative} of the Radon--Nikodym density.  At
\(q=-1/2\) one obtains
\[
 F_{-1/2}(x)=F\!\left(\frac{x+1}{3}\right),\qquad
 g_{-1/2}(x)=\frac23\RvachevUp\!\left(\frac{2x-1}{3}\right),\qquad
 Q_{-1/2}(u)=3F^{-1}(u)-1,
\]
which transports the complete Lambert--\(W_{-1}\), Legendre, and Thue--Morse derivative
structures to the support \([-1,2]\).

A second exact duality concerns normalized finite geometric sums: reversal of the
digits proves invariance under \(q\mapsto q^{-1}\).  Consequently the bases
\(2,-2,4,-4\), and all real \(\lvert q\rvert>1\), have canonical weak limits given by
the reciprocal-base laws.  A residue-class decomposition yields a multiplication
formula for the laws and their infinite sinc products.  The same framework proves
log-concavity for every real \(\lvert q\rvert<1\), exact central plateaux when
\(\lvert q\rvert<1/2\), and a Gaussian boundary as \(\lvert q\rvert\uparrow1\).  The
latter comes with an explicit Berry--Esseen bound, closed standardized Bernoulli
cumulants, Bell-polynomial moment corrections, and a quartic characteristic-function
expansion.  Reproducible numerical experiments accompany the proofs.

All claims of novelty in this report mean \emph{novel relative to the inspected
repository corpus}; no claim of global publication priority is made.
\end{abstract}

\section{Geometric-uniform preliminaries}

\subsection{The law, its centered form, and its transform}

Let \((U_n)_{n\ge0}\) be independent \(\operatorname{Unif}(0,1)\) random variables.  For
real \(q\) with \(\lvert q\rvert<1\), define
\begin{equation}
 X_q=(1-q)\sum_{n=0}^{\infty}q^nU_n.                 \label{eq:Xq-def}
\end{equation}
The series converges absolutely almost surely.  Put
\begin{equation}
 V_n=U_n-\frac12,
 \qquad
 Y_q=X_q-\frac12=(1-q)\sum_{n=0}^{\infty}q^nV_n.     \label{eq:Yq-def}
\end{equation}
The centering follows from \((1-q)\sum_{n\ge0}q^n=1\).  Every \(V_n\) is symmetric,
which is the source of the negative-parameter duality.

Write
\[
 F_q(x)=\Probability(X_q\le x),
 \qquad
 \psi_q(t)=\Expectation e^{itY_q},
 \qquad
 \SincRad z=\begin{cases}\sin z/z,&z\ne0,\\1,&z=0.\end{cases}
\]
Independence gives the centered infinite sinc product
\begin{equation}
 \psi_q(t)=\prod_{n=0}^{\infty}
 \SincRad\!\left(\frac{(1-q)q^nt}{2}\right).             \label{eq:psi-product}
\end{equation}
The signs of the individual arguments disappear because \(\SincRad\) is even, but the
factor \(1-q\) remains and will encode the affine stretch between \(q=r\) and \(q=-r\).

\subsection{The affine fixed point}

If \(X_q'\) is an independent copy of \(X_q\), then splitting off the first digit yields
\begin{equation}
 X_q\ \doteqL\ (1-q)U_0+qX_q'.                       \label{eq:affine-fixed-point}
\end{equation}
The repository proves that this equation uniquely characterizes the law among all
probability measures for every \(\lvert q\rvert<1\), without a support or density
assumption.  In the positive range, \(X_r\) is supported on \([0,1]\).  At \(r=1/2\),
\begin{equation}
 F_{1/2}=F,
 \qquad
 g_{1/2}(x)=2\RvachevUp(2x-1),                                \label{eq:half-bridge}
\end{equation}
where \(F\) is the bounded Fabius function and \(\RvachevUp\) is the Rvachev up-function.

\subsection{$q$-brackets, Bernoulli cumulants, and Bell polynomials}

For an integer \(m\ge1\), set
\begin{equation}
 [m]_q=1+q+\cdots+q^{m-1}=\frac{1-q^m}{1-q}             \label{eq:q-bracket}
\end{equation}
when \(q\ne1\), with the continuous value \([m]_1=m\).  We use the Bernoulli convention
\(B_2=1/6\), \(B_4=-1/30\), \(B_6=1/42\).  For a centered uniform
\(V\sim\operatorname{Unif}(-1/2,1/2)\),
\begin{equation}
 \cump_{2m}(V)=\frac{B_{2m}}{2m},
 \qquad
 \cump_{2m+1}(V)=0.                                    \label{eq:uniform-cumulants}
\end{equation}
The complete exponential Bell polynomial \(\mathcal B_n\) converts cumulants to moments:
\begin{equation}
 \Expectation Z^n=\mathcal B_n\bigl(\cump_1(Z),\ldots,\cump_n(Z)\bigr). \label{eq:bell-moment}
\end{equation}

\section{Negative parameters: an exact affine duality}

\subsection{The coupling theorem}

\begin{theorem}[Negative/positive affine duality]\label{thm:negative-duality}
Let \(0<r<1\) and
\begin{equation}
 c_r=\frac{1+r}{1-r}.                                    \label{eq:cr-def}
\end{equation}
Then
\begin{equation}
 \boxed{
 X_{-r}-\frac12\ \doteqL\ c_r\left(X_r-\frac12\right).}
                                                               \label{eq:negative-duality}
\end{equation}
More strongly, there is a digitwise coupling under which the equality holds almost
surely.
\end{theorem}

\begin{proof}
Let \(V_n=U_n-1/2\).  Define reflected digits
\[
 V_n^-=(-1)^nV_n,
 \qquad
 U_n^-=\frac12+V_n^-=
 \begin{cases}
 U_n,&n\ \text{even},\\
 1-U_n,&n\ \text{odd}.
 \end{cases}
\]
The sequence \((U_n^-)\) is again i.i.d. uniform.  Using it to construct the negative-base
law gives, almost surely,
\begin{align*}
 X_{-r}-\frac12
  &=(1+r)\sum_{n\ge0}(-r)^nV_n^-\\
  &=(1+r)\sum_{n\ge0}(-r)^n(-1)^nV_n\\
  &=(1+r)\sum_{n\ge0}r^nV_n\\
  &=\frac{1+r}{1-r}\left((1-r)\sum_{n\ge0}r^nV_n\right)\\
  &=c_r\left(X_r-\frac12\right).
\end{align*}
Absolute convergence justifies the termwise coupling.  Forgetting the coupling proves
the distributional identity.
\end{proof}

\begin{corollary}[Support, CDF, and quantile]\label{cor:negative-cdf}
Let
\begin{equation}
 \ell_r=-\frac{r}{1-r},
 \qquad
 u_r=\frac{1}{1-r}.                                      \label{eq:negative-support}
\end{equation}
Then \(\SupportOperator(X_{-r})=[\ell_r,u_r]\), and for every real \(x\),
\begin{equation}
 \boxed{
 F_{-r}(x)=F_r\!\left(\frac12+\frac{x-1/2}{c_r}\right).}
                                                               \label{eq:negative-cdf}
\end{equation}
For generalized inverse quantiles,
\begin{equation}
 \boxed{
 Q_{-r}(u)=\frac12+c_r\left(Q_r(u)-\frac12\right),
 \qquad 0<u<1.}                                           \label{eq:negative-quantile}
\end{equation}
\end{corollary}

\begin{proof}
The affine map \(T_r(y)=1/2+c_r(y-1/2)\) is increasing.  It maps \([0,1]\) onto
\([\ell_r,u_r]\), and pushforward under an increasing affine map gives
\cref{eq:negative-cdf,eq:negative-quantile}.
\end{proof}

\subsection{The correct negative-parameter density and the sign diagnosis}

The integrated source defines, as a total symbolic function,
\begin{equation}
 \widetilde f_q(x)=
 \frac{F_q(x/q)-F_q((x-(1-q))/q)}{1-q}.                   \label{eq:source-selected-f}
\end{equation}
For \(q>0\) it proves that this is the density.  At \(q<0\), division by \(q\) reverses
the CDF interval.  The next result resolves the open interpretation exactly.

\begin{theorem}[Radon--Nikodym density for every nonzero parameter]\label{thm:all-sign-density}
For \(0<\lvert q\rvert<1\), the law of \(X_q\) has density
\begin{equation}
 g_q(x)=\frac{1}{1-q}\,
 \Probability\bigl(x-(1-q)\le qX_q'\le x\bigr),                  \label{eq:density-probability}
\end{equation}
where \(X_q'\) is an independent copy.  Equivalently,
\begin{equation}
 \boxed{
 g_q(x)=\Sign(q)\,\widetilde f_q(x).}                     \label{eq:density-sign}
\end{equation}
Thus for \(-1<q<0\), the repository's selected total function is not a probability
density but its negative.  In particular it integrates to \(-1\), while \(g_q\ge0\)
and integrates to \(1\).
\end{theorem}

\begin{proof}
Condition on \(X_q'=y\) in \cref{eq:affine-fixed-point}.  The conditional law is uniform
on the interval \([qy,qy+1-q]\), of length \(1-q>0\).  Averaging the conditional density
therefore gives \cref{eq:density-probability}.  If \(q>0\), the event in that formula is
\[
 \frac{x-(1-q)}q\le X_q'\le\frac xq,
\]
which gives \(g_q=\widetilde f_q\).  If \(q<0\), the inequalities reverse and
\[
 g_q(x)=\frac{F_q((x-(1-q))/q)-F_q(x/q)}{1-q}
       =-\widetilde f_q(x).
\]
The CDF is continuous, so endpoint conventions do not matter.
\end{proof}

\begin{corollary}[Smooth negative density]\label{cor:negative-density}
For \(0<r<1\),
\begin{equation}
 \boxed{
 g_{-r}(x)=\frac1{c_r}\,
 g_r\!\left(\frac12+\frac{x-1/2}{c_r}\right).}           \label{eq:negative-density}
\end{equation}
Consequently \(g_{-r}\in C_c^\infty(\RealNumbers)\), with support \([\ell_r,u_r]\), and
\begin{equation}
 g_{-r}^{(k)}(x)=c_r^{-(k+1)}
 g_r^{(k)}\!\left(\frac12+\frac{x-1/2}{c_r}\right).      \label{eq:negative-derivative}
\end{equation}
\end{corollary}

\begin{proof}
Differentiate the affine CDF relation or use the change-of-variables formula for an
affine pushforward.  The derivative formula follows by the chain rule.
\end{proof}

\begin{corollary}[Norm and entropy transport]\label{cor:norm-entropy}
For \(1\le p\le\infty\) and \(k\ge0\),
\begin{equation}
 \Norm{g_{-r}^{(k)}}_p
 =c_r^{1/p-k-1}\Norm{g_r^{(k)}}_p,                        \label{eq:sobolev-scaling}
\end{equation}
with \(1/\infty=0\).  The differential entropy satisfies
\begin{equation}
 h(X_{-r})=h(X_r)+\log c_r.                               \label{eq:entropy-scaling}
\end{equation}
The same affine law gives the standard Rényi-entropy scaling.
\end{corollary}

\begin{proof}
Use \cref{eq:negative-derivative} and the substitution
\(y=1/2+(x-1/2)/c_r\).  The entropy identity is the usual Jacobian formula for
\(aX+b\) with \(a=c_r>0\).
\end{proof}

\subsection{Characteristic functions and sinc products}

\begin{corollary}[Centered transform scaling]\label{cor:negative-char}
For all real \(t\),
\begin{equation}
 \boxed{\psi_{-r}(t)=\psi_r(c_rt).}                       \label{eq:negative-char}
\end{equation}
Equivalently,
\begin{equation}
 \prod_{n\ge0}\SincRad\!\left(\frac{(1+r)r^nt}{2}\right)
 =\prod_{n\ge0}\SincRad\!\left(\frac{(1-r)r^nc_rt}{2}\right).
\end{equation}
\end{corollary}

\begin{proof}
Apply characteristic functions to \cref{eq:negative-duality}.  The product identity is
also immediate from \((1-r)c_r=1+r\).
\end{proof}

\section{The exact negative dyadic law}

\subsection{Fabius, inverse Fabius, and Rvachev formulas}

Set \(r=1/2\).  Then \(c_r=3\), \(\ell_r=-1\), and \(u_r=2\).

\begin{theorem}[Negative dyadic bridge]\label{thm:negative-dyadic}
For every real \(x\) and every \(0<u<1\),
\begin{align}
 F_{-1/2}(x)&=F\!\left(\frac{x+1}{3}\right),              \label{eq:negative-dyadic-cdf}\\
 g_{-1/2}(x)&=\frac23\RvachevUp\!\left(\frac{2x-1}{3}\right),  \label{eq:negative-dyadic-density}\\
 Q_{-1/2}(u)&=3F^{-1}(u)-1.                               \label{eq:negative-dyadic-quantile}
\end{align}
The support is \([-1,2]\).  The source-selected total function satisfies
\begin{equation}
 \widetilde f_{-1/2}(x)=-\frac23\RvachevUp\!\left(\frac{2x-1}{3}\right). \label{eq:negative-dyadic-selected}
\end{equation}
\end{theorem}

\begin{proof}
Insert \(r=1/2\) into \cref{eq:negative-cdf,eq:negative-density,eq:negative-quantile},
then use the half-base bridge \cref{eq:half-bridge}.  The sign statement follows from
\cref{eq:density-sign}.
\end{proof}

This is a direct and unexpectedly economical link between a negative geometric base and
the inverse Fabius function.  It also means that every exact formula for \(F^{-1}\) or
its asymptotics is automatically an exact formula for the negative dyadic quantile.

\subsection{Thue--Morse derivative peaks on a stretched dyadic grid}

The frontier corpus proves the sharp Rvachev derivative identities
\begin{equation}
 \Norm{\RvachevUp^{(m)}}_\infty=2^{\binom{m+1}{2}},
 \qquad
 \RvachevUp^{(m)}(c_{m,j})=\ThueMorseSign{j}2^{\binom{m+1}{2}},       \label{eq:up-peaks-baseline}
\end{equation}
where
\begin{equation}
 c_{m,j}=-1+2^{-m}+j2^{1-m},
 \qquad 0\le j<2^m.                                      \label{eq:cmj}
\end{equation}
It also proves higher-order flatness at this grid: \(\RvachevUp^{(k)}(c_{m,j})=0\) for
\(k>m\).  Transport gives a new negative-base atlas.

\begin{corollary}[Negative dyadic Thue--Morse atlas]\label{cor:negative-thue-morse}
Define
\begin{equation}
 x_{m,j}=\frac{1+3c_{m,j}}2
 =-1+\frac{3}{2^{m+1}}+\frac{3j}{2^m}.                    \label{eq:xmj}
\end{equation}
Then
\begin{align}
 g_{-1/2}^{(m)}(x_{m,j})
 &=\ThueMorseSign{j}
   \left(\frac23\right)^{m+1}2^{\binom{m+1}{2}},          \label{eq:negative-peak-values}\\
 \Norm{g_{-1/2}^{(m)}}_\infty
 &=\frac{2^{\binom{m+1}{2}+m+1}}{3^{m+1}},               \label{eq:negative-peak-norm}\\
 g_{-1/2}^{(k)}(x_{m,j})&=0\qquad(k>m).                   \label{eq:negative-flatness}
\end{align}
Thus the same Thue--Morse signs occur on a grid whose spacing is stretched from
\(2^{1-m}\) to \(3\cdot2^{-m}\).
\end{corollary}

\begin{proof}
From \cref{eq:negative-dyadic-density},
\[
 g_{-1/2}^{(m)}(x)=\left(\frac23\right)^{m+1}
 \RvachevUp^{(m)}\!\left(\frac{2x-1}{3}\right).
\]
Set \((2x-1)/3=c_{m,j}\) and use \cref{eq:up-peaks-baseline} and the higher-flatness
result.
\end{proof}

\subsection{Legendre-series transport}

Suppose the even Legendre expansion on \([-1,1]\) is
\begin{equation}
 \RvachevUp(t)=\sum_{n=0}^{\infty}b_nP_{2n}(t)                   \label{eq:up-legendre}
\end{equation}
in \(L^2\), or in any stronger mode established for the coefficient sequence.  Then the
negative dyadic density has the immediate expansion
\begin{equation}
 \boxed{
 g_{-1/2}(x)=\frac23\sum_{n=0}^{\infty}
 b_nP_{2n}\!\left(\frac{2x-1}{3}\right),
 \qquad -1\le x\le2.}                                    \label{eq:negative-legendre}
\end{equation}
No coefficients need to be recomputed: only the Jacobian and affine Legendre coordinate
change.  More generally, if
\[
 g_r(y)=\sum_{n\ge0}a_n^{(r)}P_{2n}(2y-1)
\]
on \([0,1]\), then
\begin{equation}
 g_{-r}(x)=\frac1{c_r}\sum_{n\ge0}a_n^{(r)}
 P_{2n}\!\left(\frac{2x-1}{c_r}\right).                  \label{eq:general-negative-legendre}
\end{equation}
This supplies a direct bridge from the positive-$q$ Legendre theory to every negative
parameter.

\subsection{Exact transport of the Lambert--\texorpdfstring{$W_{-1}$}{W-1} endpoint phase}

Let \(L=\log2\), and let \(\lambda(z)\) be the large positive solution of
\begin{equation}
 z=\lambda2^{-\lambda}.
\end{equation}
For sufficiently small positive \(z\),
\begin{equation}
 \lambda(z)=-\frac1L W_{-1}(-Lz).                         \label{eq:lambert-phase}
\end{equation}
The corpus emphasizes that periodic coefficient functions must be evaluated at this
\emph{exact} phase rather than at a phase with merely bounded error.

At the left endpoint of the negative dyadic support, write \(x=-1+\xi\).  Then
\begin{equation}
 F_{-1/2}(-1+\xi)=F(\xi/3),                               \label{eq:endpoint-transport}
\end{equation}
and the exact transported phase is
\begin{equation}
 \boxed{
 \lambda_-(\xi)=\lambda(\xi/3)
 =-\frac1L W_{-1}\!\left(-\frac{L\xi}{3}\right).}        \label{eq:negative-lambert-phase}
\end{equation}
Therefore every all-orders Fabius endpoint expansion, including its smooth one-periodic
logarithmic corrections, transfers verbatim after the replacement
\(\lambda\mapsto\lambda_-\) and the elementary Jacobian factors for derivatives.  The
right endpoint follows by reflection.  For the inverse law, the exact relation is even
simpler:
\begin{equation}
 Q_{-1/2}(u)+1=3F^{-1}(u).                                \label{eq:inverse-endpoint-transport}
\end{equation}

\section{Shape theory: log-concavity and exact plateaux}

\subsection{Log-concavity for all real parameters}

\begin{definition}
A probability measure \(\mu\) on \(\RealNumbers\) is log-concave if for nonempty compact sets
\(A,B\) and \(0<\theta<1\),
\[
 \mu(\theta A+(1-\theta)B)
 \ge \mu(A)^\theta\mu(B)^{1-\theta}.
\]
An integrable function is log-concave when its logarithm is concave on its positivity
set, with the usual extended-value convention.
\end{definition}

\begin{theorem}[Log-concavity of the geometric-uniform family]\label{thm:logconcavity}
For every real \(q\) with \(\lvert q\rvert<1\), the law of \(X_q\) is log-concave.  For
\(q\ne0\), its smooth density \(g_q\) is log-concave on \(\RealNumbers\); at \(q=0\), the density
is \(\mathbf 1_{[0,1]}\), which is log-concave in the extended sense.
\end{theorem}

\begin{proof}
For finite \(N\), the partial sum
\[
 X_q^{(N)}=(1-q)\sum_{n=0}^{N}q^nU_n
\]
is a convolution of uniform laws on compact intervals.  A negative coefficient merely
reflects an interval, so each factor remains log-concave.  Affine images and convolutions
preserve log-concavity by the Pr\'ekopa theorem \cite{prekopa1973}.  Hence every finite
law is log-concave.  The partial sums converge almost surely, therefore weakly, to \(X_q\).
The class of log-concave probability measures is closed under weak convergence, so the
limit law is log-concave.  A log-concave measure that has a continuous density has that
density as its log-concave representative.  Smoothness for \(q\ne0\) follows from the
positive theory and \cref{cor:negative-density}; \(q=0\) is uniform.
\end{proof}

\begin{remark}
This result supplies a global shape theorem that the repository lists as open.  It implies
unimodality and, together with reflection symmetry, places the mode at \(1/2\).  It does
\emph{not} by itself imply strict log-concavity; the next theorem exhibits a whole
parameter region where strictness is impossible.
\end{remark}

\subsection{A dominant-digit plateau lemma}

\begin{lemma}[Uniform convolution plateau]\label{lem:uniform-plateau}
Let \(U\sim\operatorname{Unif}(0,1)\), independent of a random variable \(Y\) supported
on \([\alpha,\beta]\).  Let \(a>0\) and \(X=aU+Y\).  If \(a\ge\beta-\alpha\), then the
density of \(X\) is exactly \(1/a\) on
\begin{equation}
 [\beta,\alpha+a].                                        \label{eq:general-plateau}
\end{equation}
When \(a>\beta-\alpha\), this is a nondegenerate interval.
\end{lemma}

\begin{proof}
Conditioning on \(Y\),
\[
 f_X(x)=\frac1a\Probability(x-a\le Y\le x).
\]
For \(x\in[\beta,\alpha+a]\), one has \(x\ge\beta\) and \(x-a\le\alpha\), so the event
contains the whole support of \(Y\) and has probability one.
\end{proof}

\begin{theorem}[Exact positive and negative central plateaux]\label{thm:geometric-plateau}
Let \(0\le r<1/2\).

For the positive law,
\begin{equation}
 \boxed{
 g_r(x)=\frac1{1-r}\quad\text{for }r\le x\le1-r.}         \label{eq:positive-plateau}
\end{equation}
For the negative law,
\begin{equation}
 \boxed{
 g_{-r}(x)=\frac1{1+r}\quad\text{for }
 \frac{r^2}{1-r}\le x\le1-\frac{r^2}{1-r}.}              \label{eq:negative-plateau}
\end{equation}
At \(r=1/2\) each interval collapses to the point \(x=1/2\).  The plateau probability
mass in either law is
\begin{equation}
 \frac{1-2r}{1-r}.                                        \label{eq:plateau-mass}
\end{equation}
\end{theorem}

\begin{proof}
From the fixed-point equation,
\[
 X_r=(1-r)U+rX_r',
\]
and \(rX_r'\) is supported on \([0,r]\).  Apply
\cref{lem:uniform-plateau} with \(a=1-r\), \(\alpha=0\), \(\beta=r\).  The condition
\(1-r\ge r\) is exactly \(r\le1/2\), and the height is \(1/(1-r)\).

For the negative law, apply the affine density transport \cref{eq:negative-density} to
the interval \([r,1-r]\).  Its lower endpoint maps to
\[
 \frac12+c_r\left(r-\frac12\right)=\frac{r^2}{1-r},
\]
and symmetry gives the upper endpoint.  The height is
\(c_r^{-1}(1-r)^{-1}=1/(1+r)\).  Multiplying height by interval length gives
\cref{eq:plateau-mass}.
\end{proof}

\begin{conjecture}[Strict-log-concavity phase transition]\label{conj:strict-logconcavity}
For every \(1/2\le r<1\), the density \(g_r\) is strictly log-concave on \((0,1)\) in
the multiplicative-inequality sense.  For \(0<r<1/2\), the interval
\([r,1-r]\) is the maximal constant interval, and each shoulder is strictly
log-concave.  By affine duality, the corresponding statements hold for \(g_{-r}\).
\end{conjecture}

The critical value \(r=1/2\) is structurally natural: it is exactly where the first
uniform digit and the entire tail have equal support length.  Proving strictness above
the transition likely requires an equality-case analysis for repeated Pr\'ekopa
convolution or a direct argument from the refinement equation.

\section{Residue-class decomposition and multiplication formulas}

\subsection{Decomposition into independent \texorpdfstring{$q^m$}{q^m} laws}

\begin{theorem}[Residue-class multiplication law]\label{thm:residue-decomposition}
Let \(m\ge1\) and \(\lvert q\rvert<1\).  Let
\(X_{q^m}^{(0)},\ldots,X_{q^m}^{(m-1)}\) be independent copies of \(X_{q^m}\).  Then
\begin{equation}
 \boxed{
 X_q\ \doteqL\ \frac1{[m]_q}\sum_{j=0}^{m-1}q^jX_{q^m}^{(j)}.} \label{eq:residue-decomposition}
\end{equation}
The centered variables satisfy the same weighted identity:
\begin{equation}
 Y_q\ \doteqL\ \frac1{[m]_q}\sum_{j=0}^{m-1}q^jY_{q^m}^{(j)}. \label{eq:centered-residue}
\end{equation}
\end{theorem}

\begin{proof}
Split the defining series into residue classes \(n=mk+j\):
\begin{align*}
 X_q
 &=(1-q)\sum_{j=0}^{m-1}q^j\sum_{k\ge0}q^{mk}U_{mk+j}\\
 &=\frac{1-q}{1-q^m}\sum_{j=0}^{m-1}q^j
 \left((1-q^m)\sum_{k\ge0}(q^m)^kU_{mk+j}\right).
\end{align*}
The parenthesized variables are independent copies of \(X_{q^m}\), and
\((1-q)/(1-q^m)=1/[m]_q\).  Since the weights sum to one, subtracting \(1/2\) yields the
centered identity.
\end{proof}

\begin{corollary}[Transform and cumulant multiplication]\label{cor:multiplication}
For all real \(t\),
\begin{equation}
 \boxed{
 \psi_q(t)=\prod_{j=0}^{m-1}
 \psi_{q^m}\!\left(\frac{q^jt}{[m]_q}\right).}            \label{eq:psi-multiplication}
\end{equation}
Whenever the cumulant of order \(s\) is considered,
\begin{equation}
 \boxed{
 \cump_s(X_q)=\frac{[m]_{q^s}}{[m]_q^s}\,
 \cump_s(X_{q^m}).}                                      \label{eq:cumulant-multiplication}
\end{equation}
\end{corollary}

\begin{proof}
Characteristic functions multiply over independent sums.  Cumulants of order \(s\)
scale by the \(s\)-th power of a coefficient and add, giving
\[
 \frac{1}{[m]_q^s}\sum_{j=0}^{m-1}q^{js}
 =\frac{[m]_{q^s}}{[m]_q^s}.
\]
\end{proof}

\subsection{Even--odd splitting}

For \(m=2\), \cref{eq:residue-decomposition} becomes
\begin{align}
 X_r&\doteqL\frac{X_{r^2}^{(0)}+rX_{r^2}^{(1)}}{1+r},     \label{eq:positive-even-odd}\\
 X_{-r}&\doteqL\frac{X_{r^2}^{(0)}-rX_{r^2}^{(1)}}{1-r}. \label{eq:negative-even-odd}
\end{align}
Thus a negative-base law is a normalized difference of two independent positive-base
laws.  Combining this with \cref{thm:negative-duality} gives the distribution identity
\begin{equation}
 \frac{X_{r^2}^{(0)}-rX_{r^2}^{(1)}}{1-r}-\frac12
 \ \doteqL\
 c_r\left(
 \frac{X_{r^2}^{(0)}+rX_{r^2}^{(1)}}{1+r}-\frac12\right). \label{eq:sum-difference-identity}
\end{equation}

For centered transforms, \(\psi_{q^2}\) is even, so
\begin{equation}
 \psi_{-r}(t)=
 \psi_{r^2}\!\left(\frac{t}{1-r}\right)
 \psi_{r^2}\!\left(\frac{rt}{1-r}\right).                \label{eq:negative-even-odd-char}
\end{equation}
Together with \(\psi_{-r}(t)=\psi_r(c_rt)\), this yields the sinc-product identity
\begin{equation}
 \boxed{
 \psi_r(c_rt)=
 \psi_{r^2}\!\left(\frac{t}{1-r}\right)
 \psi_{r^2}\!\left(\frac{rt}{1-r}\right).}              \label{eq:sinc-splitting}
\end{equation}
At \(r=1/2\),
\begin{equation}
 \boxed{\psi_{1/2}(3t)=\psi_{1/4}(2t)\psi_{1/4}(t).}      \label{eq:dyadic-quarter-product}
\end{equation}
This is precisely the interlacing of the even and odd factors of the dyadic infinite sinc
product into two quarter-base products.

\subsection{Computational significance}

The residue formula provides a multiscale density algorithm.  Choose \(m\) so that
\(\lvert q\rvert^m\) is small.  The \(q^m\)-law then has a broad exact plateau and short
refinement depth; convolving its independently scaled copies reconstructs \(g_q\).  This
can outperform direct evaluation of a long sinc product when \(\lvert q\rvert\) is close
to one.  For negative \(q\), taking even \(m\) reduces every building block to a positive
parameter.

\section{Reciprocal bases and finite geometric normalization}

\subsection{Exact \texorpdfstring{$q\leftrightarrow q^{-1}$}{q <-> q^-1} reversal}

For an integer \(N\ge1\), define the normalized finite law
\begin{equation}
 X_{q,N}=\frac1{[N]_q}\sum_{n=0}^{N-1}q^nU_n,             \label{eq:finite-Xq}
\end{equation}
whenever \([N]_q\ne0\).  The coefficients sum to one, so \(\Expectation X_{q,N}=1/2\).  At
\(q=1\), the continuous extension is the arithmetic mean \(N^{-1}\sum U_n\).

\begin{theorem}[Finite reciprocal-base duality]\label{thm:reciprocal-duality}
Let \(q\ne0\) and suppose both normalizations are defined.  Then
\begin{equation}
 \boxed{X_{q,N}\ \doteqL\ X_{1/q,N}.}                    \label{eq:reciprocal-law}
\end{equation}
More strongly, if the digits on the reciprocal side are reversed, the two random
variables are equal almost surely.
\end{theorem}

\begin{proof}
The $q$-bracket reciprocity identity is
\begin{equation}
 [N]_{1/q}=q^{1-N}[N]_q.                                  \label{eq:qbracket-reciprocity}
\end{equation}
Therefore
\[
 \frac{(1/q)^{N-1-n}}{[N]_{1/q}}
 =\frac{q^{-N+1+n}}{q^{1-N}[N]_q}
 =\frac{q^n}{[N]_q}.
\]
The coefficient of \(U_n\) at base \(q\) is exactly the coefficient of the reversed digit
\(U_{N-1-n}\) at base \(1/q\).  Since the digit vector is exchangeable, the laws agree;
using the reversed vector gives pathwise equality.
\end{proof}

\subsection{Canonical limits for \texorpdfstring{$\lvert q\rvert>1$}{|q|>1}}

\begin{theorem}[Reciprocal-base compactification]\label{thm:outside-limit}
Let \(\lvert q\rvert>1\) and put \(p=1/q\).  Then
\begin{equation}
 X_{q,N}\ \doteqL\ X_{p,N}\ \Longrightarrow\ X_p
 \qquad(N\to\infty).                                     \label{eq:outside-limit}
\end{equation}
Thus the normalized finite laws at every real base outside the unit interval have a
canonical weak limit.  Explicitly,
\begin{align*}
 q=2&:\quad X_{q,N}\Longrightarrow X_{1/2},\\
 q=-2&:\quad X_{q,N}\Longrightarrow X_{-1/2}\doteqL3X_{1/2}-1,\\
 q=4&:\quad X_{q,N}\Longrightarrow X_{1/4},\\
 q=-4&:\quad X_{q,N}\Longrightarrow X_{-1/4}
 \doteqL\frac12+\frac53\left(X_{1/4}-\frac12\right).
\end{align*}
\end{theorem}

\begin{proof}
The reciprocal identity is \cref{thm:reciprocal-duality}.  Since \(\lvert p\rvert<1\),
\([N]_p\to(1-p)^{-1}\) and the normalized finite series converges almost surely to
\((1-p)\sum_{n\ge0}p^nU_n=X_p\).  The special cases use
\cref{thm:negative-duality}.
\end{proof}

\begin{proposition}[Exponential coupling rate]\label{prop:finite-rate}
For \(0\le p<1\), there is a common-digit coupling such that
\begin{equation}
 \AbsoluteValue{X_{p,N}-X_p}\le p^N\qquad\text{almost surely}.      \label{eq:positive-finite-rate}
\end{equation}
For \(p=-r\), \(0<r<1\), and even \(N\), there is a coupling such that
\begin{equation}
 \AbsoluteValue{X_{-r,N}-X_{-r}}\le c_rr^N.                         \label{eq:negative-finite-rate}
\end{equation}
Hence the same bounds hold for the \(W_\infty\) Wasserstein distance.
\end{proposition}

\begin{proof}
For \(p\ge0\), write
\[
 A_N=(1-p)\sum_{n<N}p^nU_n,\qquad
 T_N=(1-p)\sum_{n\ge N}p^nU_n.
\]
Then \(0\le A_N\le1-p^N\), \(0\le T_N\le p^N\),
\(X_{p,N}=A_N/(1-p^N)\), and \(X_p=A_N+T_N\).  Hence
\[
 X_{p,N}-X_p=\frac{p^N}{1-p^N}A_N-T_N
\]
is the difference of two numbers in \([0,p^N]\).

For negative \(p=-r\), finite digit reflection gives
\[
 X_{-r,N}-\frac12\ \doteqL\
 c_{r,N}\left(X_{r,N}-\frac12\right),
 \qquad
 c_{r,N}=c_r\frac{1-r^N}{1-(-r)^N}.
\]
When \(N\) is even, \(c_{r,N}=c_r\).  Couple both finite and infinite negative laws by
the same affine images of the positive coupling, and apply
\cref{eq:positive-finite-rate}.
\end{proof}

\subsection{Bell/$q$-bracket reciprocity}

Let \(u_1=1/2\), and for \(s\ge2\) let \(u_s\) be the $s$-th cumulant of a uniform
\(U\), so \(u_s=B_s/s\) for even \(s\) and \(u_s=0\) for odd \(s>1\).

\begin{theorem}[Finite cumulants and moments]\label{thm:finite-cumulants}
For every admissible \(q,N\),
\begin{equation}
 \boxed{
 \cump_s(X_{q,N})=u_s\frac{[N]_{q^s}}{[N]_q^s}.}          \label{eq:finite-cumulants}
\end{equation}
Consequently
\begin{equation}
 \Expectation X_{q,N}^k=
 \mathcal B_k\!\left(
 u_1,
 u_2\frac{[N]_{q^2}}{[N]_q^2},\ldots,
 u_k\frac{[N]_{q^k}}{[N]_q^k}
 \right).                                                  \label{eq:finite-bell-moment}
\end{equation}
Every ratio in \cref{eq:finite-cumulants} is invariant under
\(q\mapsto q^{-1}\), giving an algebraic Bell-polynomial proof of the moment part of
\cref{thm:reciprocal-duality}.
\end{theorem}

\begin{proof}
Cumulants add over independent variables and scale homogeneously, so
\[
 \cump_s(X_{q,N})=u_s\sum_{n=0}^{N-1}
 \left(\frac{q^n}{[N]_q}\right)^s
 =u_s\frac{[N]_{q^s}}{[N]_q^s}.
\]
Apply the moment--cumulant Bell formula.  Finally,
\[
 [N]_{q^{-s}}=q^{s(1-N)}[N]_{q^s}
\]
and \([N]_{q^{-1}}^s=q^{s(1-N)}[N]_q^s\), so the ratio is reciprocal-invariant.
\end{proof}

This reciprocity is the probability-law analogue of the involution underlying Gaussian
binomial reciprocity.  It suggests a systematic dictionary between finite geometric
uniform sums, symmetric functions of the coefficient multiset, and denominator-free
$q$-identities.

\section{The Gaussian boundary \texorpdfstring{$\lvert q\rvert\uparrow1$}{|q| -> 1}}

\subsection{Sign-invariant standardization}

For \(-1<q<1\),
\begin{equation}
 \sigma_q^2=\Variance(X_q)
 =\frac{(1-q)^2}{12(1-q^2)}
 =\frac{1-q}{12(1+q)}.                                    \label{eq:variance-q}
\end{equation}
Define
\begin{equation}
 Z_q=\frac{X_q-1/2}{\sigma_q}.                             \label{eq:Zq}
\end{equation}

\begin{proposition}[Standardized sign collapse]\label{prop:standard-sign}
For \(0<r<1\),
\begin{equation}
 \boxed{Z_{-r}\ \doteqL\ Z_r.}                           \label{eq:standard-sign}
\end{equation}
Thus the standardized family depends only on \(r=\lvert q\rvert\).
\end{proposition}

\begin{proof}
From \cref{eq:variance-q}, \(\sigma_{-r}=c_r\sigma_r\).  Divide
\cref{eq:negative-duality} by this identity.
\end{proof}

The centered characteristic function of \(Z_r\) has the useful normalized product
\begin{equation}
 \Psi_r(t)=\Expectation e^{itZ_r}
 =\prod_{n=0}^{\infty}
 \SincRad\!\left(\sqrt{3(1-r^2)}\,r^nt\right).              \label{eq:standardized-sinc}
\end{equation}

\subsection{A quantitative central limit theorem}

\begin{theorem}[Berry--Esseen Gaussian boundary]\label{thm:berry-esseen}
Let \(C_{\mathrm{BE}}\) be any universal constant valid in the classical
Berry--Esseen inequality for independent, non-identically distributed summands.  Then
for \(0<r<1\),
\begin{equation}
 \boxed{
 \sup_{x\in\RealNumbers}\AbsoluteValue{\Probability(Z_r\le x)-\PhiN(x)}
 \le
 C_{\mathrm{BE}}\frac{3\sqrt3}{4}
 \frac{(1-r^2)^{3/2}}{1-r^3}.}                            \label{eq:BE-bound}
\end{equation}
The same bound holds for \(Z_{-r}\).  In particular,
\begin{equation}
 Z_{\pm r}\Longrightarrow N(0,1)
 \qquad(r\uparrow1),                                     \label{eq:Gaussian-limit}
\end{equation}
and the right side of \cref{eq:BE-bound} is
\begin{equation}
 C_{\mathrm{BE}}\sqrt{\frac32}\sqrt{1-r}
 \bigl(1+\BigO(1-r)\bigr).                               \label{eq:BE-asymptotic}
\end{equation}
\end{theorem}

\begin{proof}
Write \(Y_q=\sum_{n\ge0}a_{n,q}V_n\) with
\(a_{n,q}=(1-q)q^n\) and \(r=\lvert q\rvert\).  Since
\[
 \Expectation\AbsoluteValue{V_n}^3=\frac1{32},
 \qquad
 \Variance(V_n)=\frac1{12},
\]
we have
\begin{align*}
 \sum_{n\ge0}\AbsoluteValue{a_{n,q}}^3&=\frac{\AbsoluteValue{1-q}^3}{1-r^3},\\
 \sum_{n\ge0}a_{n,q}^2&=\frac{(1-q)^2}{1-r^2}.
\end{align*}
The Berry--Esseen Lyapunov ratio is therefore
\begin{align*}
 \frac{\sum_n\Expectation\AbsoluteValue{a_{n,q}V_n}^3}
 {\left(\sum_n\Variance(a_{n,q}V_n)\right)^{3/2}}
 &=\frac{(1/32)}{(1/12)^{3/2}}
   \frac{(1-r^2)^{3/2}}{1-r^3}\\
 &=\frac{3\sqrt3}{4}\frac{(1-r^2)^{3/2}}{1-r^3}.
\end{align*}
Apply the finite-sum inequality and pass to the almost-sure limit.  The sign collapse
\cref{eq:standard-sign} handles \(-r\).  Expanding
\(1-r^2\sim2(1-r)\) and \(1-r^3\sim3(1-r)\) gives
\cref{eq:BE-asymptotic}.
\end{proof}

\subsection{Closed standardized Bernoulli cumulants}

\begin{theorem}[All standardized cumulants]\label{thm:standard-cumulants}
For \(m\ge1\), the even standardized cumulant of either \(Z_r\) or \(Z_{-r}\) is
\begin{equation}
 \boxed{
 \lambda_{2m}(r):=\cump_{2m}(Z_r)
 =\frac{B_{2m}}{2m}\,12^m
 \frac{(1-r^2)^m}{1-r^{2m}}.}                             \label{eq:standard-cumulants}
\end{equation}
All odd cumulants vanish.  In particular,
\begin{align}
 \lambda_4(r)&=-\frac65\frac{1-r^2}{1+r^2},              \label{eq:lambda4}\\
 \lambda_6(r)&=\frac{48}{7}
 \frac{(1-r^2)^2}{1+r^2+r^4}.                            \label{eq:lambda6}
\end{align}
For fixed \(m\ge2\), \(\lambda_{2m}(r)=\BigO((1-r)^{m-1})\).
\end{theorem}

\begin{proof}
The cumulant of order \(2m\) of \(X_r-1/2\) is
\[
 \frac{B_{2m}}{2m}(1-r)^{2m}\sum_{n\ge0}r^{2mn}
 =\frac{B_{2m}}{2m}\frac{(1-r)^{2m}}{1-r^{2m}}.
\]
Divide by
\[
 \sigma_r^{2m}=\left(\frac{1-r}{12(1+r)}\right)^m
\]
to obtain \cref{eq:standard-cumulants}.  Odd cumulants vanish by symmetry.  The
negative law has the same standardized distribution.  Factoring
\(1-r^{2m}=(1-r^2)(1+r^2+\cdots+r^{2m-2})\) gives the order estimate.
\end{proof}

\subsection{Bell-polynomial moment correction}

\begin{corollary}[First correction to Gaussian moments]\label{cor:moment-correction}
For every fixed integer \(m\ge2\),
\begin{align}
 \Expectation Z_r^{2m}
 &=(2m-1)!!
 +\binom{2m}{4}(2m-5)!!\,\lambda_4(r)
 +\BigO((1-r)^2)                                          \label{eq:moment-correction-1}\\
 &=(2m-1)!!
 -\frac65\binom{2m}{4}(2m-5)!!
  \frac{1-r^2}{1+r^2}
 +\BigO((1-r)^2).                                         \label{eq:moment-correction-2}
\end{align}
All odd moments vanish.
\end{corollary}

\begin{proof}
Use the moment--cumulant Bell expansion.  The Gaussian term consists entirely of pair
blocks and equals \((2m-1)!!\).  At first order in \(1-r\), the only possible
non-Gaussian partition has one block of size four and \(m-2\) pairs.  Its number is
\[
 \binom{2m}{4}(2m-5)!!.
\]
Every block of size at least six contributes \(\BigO((1-r)^2)\), as does a product of
two fourth cumulants.  Insert \cref{eq:lambda4}.
\end{proof}

\subsection{Quartic characteristic correction}

\begin{theorem}[First mod-Gaussian-type correction]\label{thm:quartic-correction}
For every fixed \(T>0\), uniformly for \(\AbsoluteValue t\le T\),
\begin{equation}
 \boxed{
 \log\Psi_r(t)
 =-\frac{t^2}{2}
 -\frac{1-r^2}{20(1+r^2)}t^4
 +\BigO_T((1-r)^2).}                                      \label{eq:quartic-logpsi}
\end{equation}
Consequently, for every fixed real \(t\),
\begin{equation}
 \lim_{r\uparrow1}
 \frac{\log\Psi_r(t)+t^2/2}{1-r}
 =-\frac{t^4}{20}.                                        \label{eq:quartic-limit}
\end{equation}
\end{theorem}

\begin{proof}
Use \cref{eq:standardized-sinc} and the local expansion
\[
 \log\SincRad z=-\frac{z^2}{6}-\frac{z^4}{180}+\BigO(z^6).
\]
For \(z_n=\sqrt{3(1-r^2)}r^nt\),
\begin{align*}
 \sum_{n\ge0}z_n^2&=3t^2,\\
 \sum_{n\ge0}z_n^4&=9t^4\frac{(1-r^2)^2}{1-r^4}
 =9t^4\frac{1-r^2}{1+r^2},\\
 \sum_{n\ge0}\AbsoluteValue{z_n}^6
 &=27\AbsoluteValue t^6\frac{(1-r^2)^3}{1-r^6}
 =\BigO_T((1-r)^2).
\end{align*}
For \(r\) close enough to one, all \(z_n\) lie in a fixed neighborhood where the
sixth-order remainder is uniform.  Substitution gives \cref{eq:quartic-logpsi}.
Finally, \((1-r^2)/(1+r^2)=(1-r)+\BigO((1-r)^2)\).
\end{proof}

\subsection{Expected Edgeworth and entropy refinements}

Let \(H_k\) be the probabilists' Hermite polynomial.  Fourier inversion of the formal
cumulant expansion predicts
\begin{equation}
 f_{Z_r}(z)=\phiN(z)
 \left[1+\frac{\lambda_4(r)}{24}H_4(z)\right]
 +\BigO((1-r)^2),                                         \label{eq:formal-edgeworth-density}
\end{equation}
and
\begin{equation}
 \Probability(Z_r\le z)=\PhiN(z)
 -\frac{\lambda_4(r)}{24}H_3(z)\phiN(z)
 +\BigO((1-r)^2).                                         \label{eq:formal-edgeworth-cdf}
\end{equation}
The fixed-frequency theorem above is rigorous; making the remainders uniform in \(z\)
requires a high-frequency estimate uniform in \(r\), which is not supplied here.

\begin{conjecture}[Uniform first Edgeworth expansion]\label{conj:edgeworth}
The remainders in \cref{eq:formal-edgeworth-density,eq:formal-edgeworth-cdf} are
\(\BigO((1-r)^2)\) uniformly in \(z\), with the density statement remaining valid after
any fixed number of derivatives when multiplied by a suitable polynomial weight.
The second-order term is the standard Bell--Hermite combination of
\(\lambda_6H_6/720\) and \(\lambda_4^2H_8/(2\cdot24^2)\).
\end{conjecture}

\begin{conjecture}[Entropy deficit]\label{conj:entropy-deficit}
Let \(D(\LawOperator(Z_r)\Vert N(0,1))\) be relative entropy.  Then
\begin{equation}
 D(\LawOperator(Z_r)\Vert N(0,1))
 =\frac{\lambda_4(r)^2}{48}+o((1-r)^2)
 =\frac{3}{100}(1-r)^2+o((1-r)^2).                        \label{eq:entropy-deficit}
\end{equation}
Equivalently,
\[
 h(Z_r)=\frac12\log(2\pi e)-\frac{3}{100}(1-r)^2+o((1-r)^2).
\]
\end{conjecture}

The coefficient \(1/48\) follows formally by inserting
\(f/\phiN=1+(\lambda_4/24)H_4+\cdots\) into relative entropy and using
\(\Expectation_{\phiN}H_4^2=4!\).  A proof should combine the uniform Edgeworth expansion with
tail control supplied by log-concavity.

\section{Further conjectures and frontier directions}

\subsection{All-orders Gaussian asymptotics}

\begin{conjecture}[Complete Bernoulli--Bell--Hermite expansion]\label{conj:all-orders}
For every integer \(M\ge1\), the standardized density has an expansion through order
\((1-r)^M\) whose coefficients are finite Bell polynomials in the exact cumulants
\(\lambda_4(r),\lambda_6(r),\ldots,\lambda_{2M+2}(r)\), expressed in the Hermite basis.
The expansion is uniform with polynomial weights and differentiable termwise a fixed
number of times.
\end{conjecture}

A proof would turn the corpus's Bernoulli and Bell machinery into a full parameter-edge
asymptotic calculus.  The crucial analytic estimate is uniform control of
\cref{eq:standardized-sinc} outside the low-frequency window.

\subsection{A generic-$q$ Lambert--\texorpdfstring{$W$}{W} endpoint theory}

For fixed \(0<r<1\), the negative Laplace transform of \(X_r\) is
\begin{equation}
 \Expectation e^{-sX_r}
 =\prod_{n\ge0}
 \frac{1-e^{-(1-r)r^ns}}{(1-r)r^ns}.                      \label{eq:general-laplace-product}
\end{equation}
Its logarithm is a geometric harmonic sum.  Mellin analysis should produce a quadratic
polynomial in \(\log s\), a smooth periodic fluctuation in
\(\log_{1/r}s\), and descending corrections.  Saddle inversion then produces a
Lambert--\(W_{-1}\) phase, just as in the dyadic theory.

\begin{conjecture}[Generic geometric endpoint expansion]\label{conj:generic-endpoint}
For every fixed \(0<r<1\), \(F_r(x)\) admits an all-orders small-\(x\) saddle expansion
whose coefficient functions are smooth and one-periodic in the exact logarithmic saddle
phase.  The leading inversion is expressible using \(W_{-1}\), while the periodic phase
must be kept exact.  Under \cref{thm:negative-duality}, the entire expansion transfers to
\(F_{-r}\) at its left endpoint by \(x\mapsto(x-\ell_r)/c_r\).
\end{conjecture}

This project would unify the corpus's dyadic Lambert analysis with the generic
geometric-uniform family and explain how the log-periodic oscillation changes with the
base.

\subsection{Legendre coefficients and endpoint saddles}

Because each \(g_q\) is compactly supported and smooth, its affine Legendre coefficients
decay faster than every power.  The precise stretched-exponential envelope should be
controlled by the same complex saddle geometry that governs endpoint flatness.

\begin{conjecture}[Legendre--Lambert asymptotic bridge]\label{conj:legendre-lambert}
For the even Legendre coefficients of \(g_r\), the logarithm of the coefficient magnitude
has a complete asymptotic expansion whose principal scale is obtained by balancing the
large-degree Legendre boundary layer with the Lambert saddle of \(F_r\).  A smooth
log-periodic factor survives in the coefficient asymptotics.  Negative-base coefficients
obey exactly the same law after the affine normalization in
\cref{eq:general-negative-legendre}.
\end{conjecture}

This would connect three parts of the existing program that are currently largely
parallel: endpoint Lambert asymptotics, Legendre expansions, and infinite sinc products.

\subsection{Reciprocal-base algebra and $q$-identities}

The coefficient multiset proof of \cref{thm:reciprocal-duality} suggests a broader
principle: every symmetric polynomial or entire symmetric function of
\(\{q^n/[N]_q:0\le n<N\}\) is reciprocal-invariant.  Power sums give cumulants; elementary
symmetric functions give Gaussian-binomial expressions.

\begin{problem}[A probability proof of a full Gaussian-binomial reciprocity package]
Construct natural random variables or random polynomials whose elementary-symmetric
statistics of the finite geometric weights recover
\(\GaussianBinomial{N}{k}{q}\), and derive
\(\GaussianBinomial{N}{k}{1/q}=q^{-k(N-k)}\GaussianBinomial{N}{k}{q}\) directly from digit reversal.  Then seek
mixed Bell/Gaussian identities obtained by converting between power sums and elementary
symmetric functions.
\end{problem}

\subsection{A formalization roadmap}

The strongest immediate formal targets are elementary enough to be realistic additions
to the existing Lean development:

\begin{enumerate}[leftmargin=2.2em]
\item define the centered geometric-uniform law and prove the reflected-digit coupling;
\item derive the affine pushforward equality
      \(\mu_{-r}=T_{r*}\mu_r\);
\item prove the CDF, quantile, support, and smooth-density transports;
\item replace the negative-parameter density ambiguity by the sign-corrected theorem
      \(g_q=\Sign(q)\widetilde f_q\);
\item formalize the residue-class multiplication law and its characteristic-function
      corollary;
\item prove finite reciprocal-base duality by coefficient extensionality and list
      \(q=\pm2,\pm4\) wrappers;
\item formalize the exact plateau theorem from the CDF tails;
\item package Bernoulli cumulants and standardized sign collapse.
\end{enumerate}

Log-concavity may require additional convex-measure infrastructure.  Berry--Esseen and
uniform Edgeworth theory are longer-term targets, but the exact cumulant and quartic
formal-power-series identities can be separated from those analytic theorems.

\section{Conclusions}

The geometric-uniform family has more symmetry than is visible from the positive
parameter alone.  Centering turns every alternating geometric digit into a reflected
uniform digit, and this collapses the entire negative half of the parameter interval to
an affine copy of the positive half.  That one observation resolves the negative density
frontier, identifies the sign of the previously selected total function, and transports
Fabius, inverse Fabius, Rvachev, Legendre, Lambert--\(W\), sinc-product, and Thue--Morse
structures in one step.

Two further reorganizations complete the parameter picture.  Residue classes produce a
$q$-multiplication law, while finite digit reversal identifies \(q\) with \(q^{-1}\).
Thus negative parameters, bases outside the unit interval, and smaller positive bases
are not separate theories but affine or reciprocal views of the same laws.

Finally, the parameter edge \(\lvert q\rvert\uparrow1\) is a genuine Gaussian boundary.
The rate is controlled by a closed Berry--Esseen ratio, and the departure from normality
is encoded by Bernoulli cumulants and Bell partitions.  The quartic coefficient
\(-t^4/20\) is the first term of an all-orders asymptotic program that should connect the
probability family to the repository's existing Bell, Bernoulli, Lambert, Legendre, and
formal saddle machinery.

\section{Auxiliary algebra}

\subsection{Support endpoints under negative affine transport}

Since \(T_r(y)=1/2+c_r(y-1/2)\),
\[
 T_r(0)=\frac{1-c_r}{2}=-\frac r{1-r},
 \qquad
 T_r(1)=\frac{1+c_r}{2}=\frac1{1-r}.
\]
For the positive plateau endpoint \(y=r\),
\[
 T_r(r)=\frac12+\frac{1+r}{1-r}\left(r-\frac12\right)
 =\frac{r^2}{1-r}.
\]
These identities explain the apparently asymmetric support and the symmetric central
plateau of the negative law.

\subsection{Finite negative/positive scale}

Digit reflection in \cref{eq:finite-Xq} gives
\[
 X_{-r,N}-\frac12\ \doteqL\
 \frac{[N]_r}{[N]_{-r}}\left(X_{r,N}-\frac12\right),
\]
where
\[
 \frac{[N]_r}{[N]_{-r}}
 =\frac{1+r}{1-r}\frac{1-r^N}{1-(-r)^N}.
\]
For even \(N\), this is exactly \(c_r\); for odd \(N\), it converges exponentially to
\(c_r\).

\subsection{Low standardized cumulants}

The first values from \cref{eq:standard-cumulants} are
\begin{align*}
 \lambda_2(r)&=1,\\
 \lambda_4(r)&=-\frac65\frac{1-r^2}{1+r^2},\\
 \lambda_6(r)&=\frac{48}{7}\frac{(1-r^2)^2}{1+r^2+r^4},\\
 \lambda_8(r)&=-\frac{432}{5}
 \frac{(1-r^2)^3}{1+r^2+r^4+r^6}.
\end{align*}
The last coefficient uses \(B_8=-1/30\).  These feed directly into complete Bell
polynomials and higher Edgeworth terms.
