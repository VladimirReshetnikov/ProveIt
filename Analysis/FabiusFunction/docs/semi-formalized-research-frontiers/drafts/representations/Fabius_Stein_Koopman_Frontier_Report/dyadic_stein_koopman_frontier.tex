\begin{abstract}
The centered Rvachev law is the stationary distribution of the affine smoothing chain
\[
 X_{k+1}=\frac{X_k+U_{k+1}}2,
 \qquad U_k\sim\operatorname{Unif}[-1,1].
\]
The repository already develops its reciprocal-moment Appell polynomials, Bernoulli cumulants, exact dyadic arithmetic, sinc products, and all-orders inverse-Fabius endpoint asymptotics. This report shows that these ingredients fit a previously unexploited dynamical framework. The inverse-moment Appell polynomial of degree $n$ is an exact Koopman eigenfunction with eigenvalue $2^{-n}$. From this one obtains explicit polynomial spectral projectors, dual eigendistributions $\upf^{(n)}$, closed Poisson and resolvent formulae, a $q$-Weyl commutation relation, and the transfer determinant
\[
 \det(I-zP\vert_{\Pi_d})=(z;1/2)_{d+1}.
\]
An Appell-Fock completion gives a trace-class realization with Fredholm determinant $(z;1/2)_\infty$.

The stationary fixed-point equation also supplies a Stein operator $I-P$ whose discrepancy is quantitatively equivalent to Wasserstein distance, while the differentiated operator
\[
 \Scal f=f'(x)-f\!\left(\frac{x+1}{2}\right)+f\!\left(\frac{x-1}{2}\right)
\]
is the weak adjoint of Rvachev's differential-dilation equation. Polynomial and exponential eigenmodes generate exact martingales. Their lag correlations reveal genuine time asymmetry: $\E[\Acal_2(Y)\Acal_4(Y)]=-128/33075$, giving distinct forward and reverse one-step correlations and hence a rigorous nonreversibility certificate. A general factorization proves that every nontrivial affine-uniform chain with $0<|r|<1$ is nonreversible.

A second strand derives the classical scalar Stein kernel directly from the Fabius function:
\[
 \tau(x)=
 \frac{|x|F((1-|x|)/2)+2F((1-|x|)/4)}{F(1-|x|)}.
\]
It has exact rational values at dyadic points, an inverse-Fabius integral representation, and Bell--Bernoulli Appell moments $\E[\tau(Y)\Acal_n(Y)]=\kappa_{n+2}/(n+1)$. Combining this exact formula with the repository's phase-aware inverse asymptotics yields
\[
 \tau(1-G(y))=G(y)\left[
 \frac1\rho+\frac{(\Psi'(\theta)-1)/\log2-1}{\rho^2}
 +\BigO\!\left(\frac{1+\log\rho}{\rho^3}\right)
 \right],
\]
and therefore $\tau(1-q)\sim q/\lambda(q)$, where $\lambda$ is the exact lower-Lambert coordinate defined in \cref{eq:lambda-q}. Exact arithmetic through dyadic depth $120$ and symbolic checks through degree $10$ accompany the proofs. The report closes with concrete conjectures on natural-space spectra, full Stein-kernel transseries, Thue--Morse spectral projectors, Legendre conditioning, and multivariate atomic masks.
\end{abstract}

\section{Normalization and imported interfaces}
\label{sec:normalization}

\subsection{Fabius, Rvachev, and the centered probability law}

Let $F:[0,1]\to[0,1]$ be the standard bounded Fabius distribution, characterized by
\begin{equation}
 F'(x)=2F(2x)\quad(0\le x\le1/2),
 \qquad F(1-x)=1-F(x),
 \label{eq:Fabius-basic}
\end{equation}
with $F(0)=0$ and $F(1)=1$. Rvachev's compactly supported density is
\begin{equation}
 u(x):=\upf(x)=F(1-|x|)\mathbf 1_{[-1,1]}(x).
 \label{eq:up-F}
\end{equation}
It is even, nonnegative, $C^\infty$, flat at $\pm1$, and has total mass one. Its differential-dilation equation is
\begin{equation}
 u'(x)=2\bigl(u(2x+1)-u(2x-1)\bigr).
 \label{eq:up-dde}
\end{equation}

Let $U_1,U_2,\ldots$ be independent $\operatorname{Unif}[-1,1]$ variables. The centered Rvachev random variable is
\begin{equation}
 Y=\sum_{j=1}^{\infty}2^{-j}U_j,
 \qquad \Pp(Y\in\dd x)=u(x)\dd x.
 \label{eq:Y-series}
\end{equation}
Its moment generating function and cumulant generating function are
\begin{align}
 M(t)&=\E\e^{tY}
 =\prod_{j=1}^{\infty}\frac{\sinh(t/2^j)}{t/2^j},
 \label{eq:M-product}\\
 K(t)&=\log M(t)
 =\sum_{m\ge1}\kappa_{2m}\frac{t^{2m}}{(2m)!},
 \qquad
 \kappa_{2m}=\frac{2^{2m-1}B_{2m}}{m(4^m-1)}.
 \label{eq:cumulants}
\end{align}
On the imaginary axis, $M(2\pi\ii\xi)$ is the repository's infinite sinc product. Thus the moment, Fourier, and dyadic scaling pictures are restrictions of one entire function.

The series splits after its first digit:
\begin{equation}
 Y\ \stackrel d=\ \frac{Y+U}{2},
 \qquad
 M(t)=M(t/2)\frac{\sinh(t/2)}{t/2},
 \label{eq:fixedpoint}
\end{equation}
where the $Y$ and $U$ on the right are independent.

\subsection{The imported inverse-moment Appell family}

The repository defines the reciprocal-moment Appell polynomials by
\begin{equation}
 \frac{\e^{xt}}{M(t)}
 =\sum_{n=0}^{\infty}\Acal_n(x)\frac{t^n}{n!}.
 \label{eq:Appell-gen}
\end{equation}
They satisfy
\begin{equation}
 \Acal_n'(x)=n\Acal_{n-1}(x),
 \qquad
 \Acal_n(-x)=(-1)^n\Acal_n(x),
 \qquad
 \E\Acal_n(Y)=\delta_{n0},
 \label{eq:Appell-basic}
\end{equation}
and the exact deconvolution law
\begin{equation}
 \int_{-1}^{1}\Acal_n(x-y)u(y)\dd y=x^n.
 \label{eq:Appell-deconv}
\end{equation}
The first nontrivial members are
\begin{align}
 \Acal_2(x)&=x^2-\frac19,
 &\Acal_3(x)&=x^3-\frac{x}{3},
 \notag\\
 \Acal_4(x)&=x^4-\frac23x^2+\frac{31}{675},
 &\Acal_6(x)&=x^6-\frac53x^4+\frac{31}{45}x^2-\frac{2347}{59535}.
 \label{eq:Appell-low}
\end{align}
The new point is not the existence of these polynomials. It is that they diagonalize the exact affine dynamics behind $u$.

\section{The affine smoothing chain}
\label{sec:chain}

\subsection{A geometric-uniform family}

For $-1<r<1$, let $U_k\sim\operatorname{Unif}[-1,1]$ be independent and define
\begin{equation}
 X_{k+1}=rX_k+(1-r)U_{k+1}.
 \label{eq:r-chain}
\end{equation}
The stationary series is
\begin{equation}
 Y_r=(1-r)\sum_{j=0}^{\infty}r^jU_{j+1}.
 \label{eq:Yr}
\end{equation}
For $0<r<1$ it is supported on $[-1,1]$; at $r=1/2$ it is exactly $Y$. Write $\pi_r$ for its law and
\begin{equation}
 (P_rf)(x)=\frac12\int_{-1}^{1}f(rx+(1-r)v)\dd v
 \label{eq:Pr}
\end{equation}
for the Markov/Koopman operator acting on observables.

\begin{proposition}[Stationarity and Wasserstein contraction]
\label{prop:contraction}
For every $p\ge1$ and probability measures $\mu,\nu$ with finite $p$th moment,
\begin{equation}
 W_p(\mu P_r^k,\nu P_r^k)\le |r|^kW_p(\mu,\nu).
 \label{eq:Wp-contract}
\end{equation}
Consequently $\pi_r$ is the unique stationary law with finite $p$th moment, and every initial law converges to it geometrically in $W_p$.
\end{proposition}

\begin{proof}
Couple $X\sim\mu$ and $Z\sim\nu$ optimally and use the same innovations in both chains. Their difference after one step is exactly $r(X-Z)$; iteration gives \eqref{eq:Wp-contract}. Uniqueness follows by applying the bound to two stationary laws.
\end{proof}

Let $M_r(t)=\E\e^{tY_r}$ and $m_r(t)=\E\e^{t(1-r)U}$. Stationarity is equivalent to the exact cocycle
\begin{equation}
 M_r(t)=M_r(rt)m_r(t),
 \qquad
 m_r(t)=\frac{\sinh((1-r)t)}{(1-r)t}.
 \label{eq:Mr-cocycle}
\end{equation}
The even cumulants are
\begin{equation}
 \kappa_{2m}(Y_r)
 =\frac{2^{2m-1}B_{2m}}{m}\,
 \frac{(1-r)^{2m}}{1-r^{2m}},
 \qquad \kappa_{2m+1}(Y_r)=0.
 \label{eq:r-cumulants}
\end{equation}
At $r=1/2$, this reduces to \eqref{eq:cumulants}.

\section{Appell--Koopman diagonalization}
\label{sec:koopman}

\subsection{The master affine theorem}

Let a general affine recursion be
\begin{equation}
 X_{k+1}=rX_k+\xi_{k+1},
 \qquad |r|<1,
 \label{eq:general-AR}
\end{equation}
where the innovations have an MGF near zero and the stationary law has MGF $M$. Define its reciprocal-moment Appell polynomials by \eqref{eq:Appell-gen} with this $M$.

\begin{theorem}[Appell--Koopman eigenbasis]
\label{thm:koopman-eigen}
For every $n\ge0$,
\begin{equation}
 P_r\Acal_n=r^n\Acal_n.
 \label{eq:koopman-eigen}
\end{equation}
Equivalently, for the stationary chain,
\begin{equation}
 \E[\Acal_n(X_k)\mid X_0=x]=r^{nk}\Acal_n(x).
 \label{eq:conditional-eigen}
\end{equation}
In the Rvachev case $r=1/2$, the complete polynomial spectrum is $1,2^{-1},2^{-2},\ldots$.
\end{theorem}

\begin{proof}
Let $m_\xi(t)=\E\e^{t\xi}$. Stationarity gives $M(t)=M(rt)m_\xi(t)$. Apply $P_r$ to the generating function:
\begin{align*}
 P_r\left(\frac{\e^{tx}}{M(t)}\right)
 &=\frac{\e^{rtx}m_\xi(t)}{M(t)}
 =\frac{\e^{rtx}}{M(rt)}\\
 &=\sum_{n\ge0}\Acal_n(x)\frac{(rt)^n}{n!}.
\end{align*}
Coefficient comparison proves \eqref{eq:koopman-eigen}; iteration proves \eqref{eq:conditional-eigen}.
\end{proof}

\begin{resultbox}[New organizing principle]
The repository's inverse-moment Appell polynomials are not only a deconvolution basis. They are the exact eigenmodes of the affine refinement dynamics that generates the Rvachev law. The same MGF quotient simultaneously inverts convolution and diagonalizes conditional expectation.
\end{resultbox}

\subsection{Exact polynomial spectral coordinates}

Let $\Pi_d$ denote the space of polynomials of degree at most $d$.

\begin{theorem}[Spectral coordinate functional]
\label{thm:spectral-coordinate}
Every $p\in\Pi_d$ has the unique expansion
\begin{equation}
 p(x)=\sum_{n=0}^{d}\Lcal_n(p)\Acal_n(x),
 \qquad
 \boxed{\displaystyle
 \Lcal_n(p)=\frac1{n!}\E\bigl[p^{(n)}(Y_r)\bigr].}
 \label{eq:spectral-coordinate}
\end{equation}
Hence
\begin{equation}
 P_r^kp=\sum_{n=0}^{d}r^{nk}\Lcal_n(p)\Acal_n,
 \qquad
 \PiD_np:=\Lcal_n(p)\Acal_n
 \label{eq:iterate-projectors}
\end{equation}
are exact iterates and rank-one spectral projectors.
\end{theorem}

\begin{proof}
Write $p=\sum c_m\Acal_m$. By the Appell derivative law,
\[
 \Acal_m^{(n)}=\frac{m!}{(m-n)!}\Acal_{m-n}
 \quad(m\ge n).
\]
Taking expectation under $\pi_r$ kills every positive-degree Appell polynomial, so $\E p^{(n)}(Y_r)=n!c_n$. The remaining statements follow from \cref{thm:koopman-eigen}.
\end{proof}

\begin{corollary}[A Legendre--Appell spectral bridge]
\label{cor:legendre-appell}
For the Legendre polynomial $P_m^{\mathrm{Leg}}$,
\begin{equation}
 P_m^{\mathrm{Leg}}(x)
 =\sum_{n=0}^{m}c_{m,n}^{(r)}\Acal_n^{(r)}(x),
 \qquad
 c_{m,n}^{(r)}=\frac{(2n-1)!!}{n!}
 \E\!\left[C_{m-n}^{n+1/2}(Y_r)\right],
 \label{eq:legendre-appell}
\end{equation}
where $C_j^\lambda$ is the Gegenbauer polynomial and $(-1)!!=1$. Only terms with $m-n$ even survive. Therefore
\begin{equation}
 P_r^kP_m^{\mathrm{Leg}}
 =\sum_{n=0}^{m}r^{nk}c_{m,n}^{(r)}\Acal_n^{(r)}.
 \label{eq:legendre-iterate}
\end{equation}
All coefficients are rational in the dyadic case.
\end{corollary}

\begin{proof}
Use $(P_m^{\mathrm{Leg}})^{(n)}=(2n-1)!!C_{m-n}^{n+1/2}$ in \eqref{eq:spectral-coordinate}. Symmetry gives the parity statement, and the Rvachev moments are rational.
\end{proof}

This is complementary to the repository's spatial expansions of $u$ in Legendre polynomials and its finite shifted-$u$ representations of arbitrary polynomials. Here Legendre modes are propagated exactly by the affine dynamics after a finite change of basis to the Appell eigenmodes.

\subsection{Dual eigendistributions}

Regard $u_r$ as the stationary density when it exists and $u_r^{(n)}$ as its distributional derivatives.

\begin{theorem}[Dual derivative spectrum]
\label{thm:dual-spectrum}
For every $n\ge0$,
\begin{equation}
 P_r^*u_r^{(n)}=r^nu_r^{(n)}.
 \label{eq:dual-eigen}
\end{equation}
Moreover the primal and dual families are biorthogonal:
\begin{equation}
 \left\langle u_r^{(m)},\Acal_n^{(r)}\right\rangle
 =(-1)^m m!\,\delta_{mn}.
 \label{eq:biorthogonal}
\end{equation}
Thus the projector can be written
\begin{equation}
 \PiD_np=\frac{(-1)^n}{n!}
 \left\langle u_r^{(n)},p\right\rangle\Acal_n^{(r)}.
 \label{eq:projector-dual}
\end{equation}
\end{theorem}

\begin{proof}
Differentiation under the affine expectation gives
\begin{equation}
 DP_r=rP_rD,
 \qquad D^nP_r=r^nP_rD^n.
 \label{eq:DPr}
\end{equation}
Hence, for a test function $f$,
\[
 \langle P_r^*u_r^{(n)},f\rangle
 =\langle u_r^{(n)},P_rf\rangle
 =(-1)^nr^n\int u_rP_rf^{(n)}
 =r^n\langle u_r^{(n)},f\rangle,
\]
using stationarity in the last step. Equation \eqref{eq:biorthogonal} follows from the Appell derivative formula and mean-zero cancellation.
\end{proof}

For $r=1/2$, the dual eigen-distributions are the derivative tower of Rvachev's bump. Finite spline approximants express these derivatives through signed dyadic jump measures whose signs are controlled by Thue--Morse words. Equation \eqref{eq:projector-dual} therefore gives a spectral interpretation of that automatic sign structure: the limiting Thue--Morse derivative measures extract individual Koopman coordinates.

\section{\texorpdfstring{A $q$-Weyl and Bell--Bernoulli operator calculus}{A q-Weyl and Bell--Bernoulli operator calculus}}
\label{sec:qweyl}

\subsection{The derivative--Koopman commutation relation}

Equation \eqref{eq:DPr} is a genuine $q$-commutation law. With $q=r$, the derivative and the affine Koopman operator satisfy
\begin{equation}
 DP_r=qP_rD.
 \label{eq:qcommute}
\end{equation}
Consequently the Gaussian binomial theorem applies at the operator level.

\begin{theorem}[Gaussian-binomial operator power]
\label{thm:qbinomial-operator}
For every integer $N\ge0$,
\begin{equation}
 \boxed{\displaystyle
 (P_r+D)^N
 =\sum_{j=0}^{N}\qbinom{N}{j}{r}
 P_r^{N-j}D^j.}
 \label{eq:qbinomial-operator}
\end{equation}
The identity holds on polynomials and on every smooth class on which the displayed compositions are defined.
\end{theorem}

\begin{proof}
If operators $A,B$ satisfy $BA=qAB$, the standard induction behind the finite $q$-binomial theorem gives
$(A+B)^N=\sum_j\qbinom{N}{j}{q}A^{N-j}B^j$.
Take $A=P_r$ and $B=D$.
\end{proof}

At $r=1/2$, the Gaussian coefficient is exactly the half-base object that governs the repository's geometric interpolation and finite-difference formulas. Here it appears for a different reason: it is the noncommutative coefficient forced by affine differentiation.

\subsection{The Appell raising operator}

Let $K_r(t)=\log M_r(t)$ and define, on polynomials,
\begin{equation}
 R_r:=x-K_r'(D)
 =x-\sum_{m\ge1}\kappa_{2m}(Y_r)
 \frac{D^{2m-1}}{(2m-1)!}.
 \label{eq:raising}
\end{equation}
The series terminates on every polynomial.

\begin{theorem}[Bell--Bernoulli ladder]
\label{thm:ladder}
The operators $D$ and $R_r$ act on the Appell basis by
\begin{equation}
 D\Acal_n^{(r)}=n\Acal_{n-1}^{(r)},
 \qquad
 R_r\Acal_n^{(r)}=\Acal_{n+1}^{(r)}.
 \label{eq:ladder-actions}
\end{equation}
They satisfy
\begin{equation}
 [D,R_r]=I,
 \qquad
 P_rR_r=rR_rP_r.
 \label{eq:ladder-commutations}
\end{equation}
If $N_r:=R_rD$, then
\begin{equation}
 N_r\Acal_n^{(r)}=n\Acal_n^{(r)},
 \qquad
 \boxed{P_r=r^{N_r}}
 \label{eq:number-operator}
\end{equation}
in the spectral/formal functional calculus on polynomials.
\end{theorem}

\begin{proof}
Differentiate the generating function with respect to $t$:
\[
 \partial_t\frac{\e^{xt}}{M_r(t)}
 =\bigl(x-K_r'(t)\bigr)\frac{\e^{xt}}{M_r(t)}.
\]
Since $D$ acts on the generating function as multiplication by $t$, coefficient comparison gives the raising law. The commutator $[D,R_r]=[D,x]=I$. The relation $P_rR_r=rR_rP_r$ follows on each basis vector from \cref{thm:koopman-eigen}. The number-operator identities are then immediate.
\end{proof}

\begin{resultbox}[A dyadic $q$-oscillator]
The quadruple $(D,R_{1/2},N_{1/2},P_{1/2})$ is a concrete Bernoulli-deformed Weyl representation in which $D$ lowers Appell degree, $R$ raises it, $N$ counts it, and the affine refinement operator is the dyadic exponential $2^{-N}$. The Bernoulli cumulants enter only through the raising operator; the eigenvalue geometry is pure $q$-scaling.
\end{resultbox}

\section{Transfer determinants, resolvents, and Poisson equations}
\label{sec:determinants}

\subsection{Finite polynomial transfer determinants}

\begin{theorem}[$q$-Pochhammer characteristic polynomial]
\label{thm:finite-determinant}
On $\Pi_d$,
\begin{align}
 \det\bigl(I-zP_r\vert_{\Pi_d}\bigr)
 &=\prod_{n=0}^{d}(1-zr^n)
 =\qpoch{z}{r}{d+1},
 \label{eq:finite-det}\\
 \tr\bigl(P_r^k\vert_{\Pi_d}\bigr)
 &=\sum_{n=0}^{d}r^{nk}
 =\frac{1-r^{(d+1)k}}{1-r^k}.
 \label{eq:finite-trace}
\end{align}
The determinant has the Gaussian-binomial expansion
\begin{equation}
 \qpoch{z}{r}{d+1}
 =\sum_{j=0}^{d+1}(-1)^jr^{\binom j2}
 \qbinom{d+1}{j}{r}z^j.
 \label{eq:finite-det-qbinom}
\end{equation}
For $|z|$ sufficiently small,
\begin{equation}
 \log\det\bigl(I-zP_r\vert_{\Pi_d}\bigr)
 =-\sum_{k\ge1}\frac{z^k}{k}
 \frac{1-r^{(d+1)k}}{1-r^k}.
 \label{eq:finite-lambert-log}
\end{equation}
\end{theorem}

\begin{proof}
The Appell basis diagonalizes $P_r$ with eigenvalues $r^0,\ldots,r^d$. The product, trace, finite $q$-binomial theorem, and logarithmic expansion follow in succession.
\end{proof}

The repository already contains sophisticated spectral determinants built from the integer zero divisor of the infinite sinc product. The determinant \eqref{eq:finite-det} is a different object: it is the characteristic polynomial of a specified transfer operator on polynomial observables. Its spectrum is the geometric contraction ladder, not the Fourier zero multiset.

\subsection{Closed resolvents and the polynomial Poisson equation}

\begin{theorem}[Polynomial resolvent]
\label{thm:polynomial-resolvent}
For $p\in\Pi_d$ and $z$ away from the poles $r^{-n}$,
\begin{equation}
 (I-zP_r)^{-1}p
 =\sum_{n=0}^{d}\frac{\Lcal_n(p)}{1-zr^n}\Acal_n^{(r)}.
 \label{eq:poly-resolvent}
\end{equation}
If $h\in\Pi_d$ is centered under $\pi_r$, then the unique centered polynomial solution of
\begin{equation}
 (I-P_r)g=h
 \label{eq:Poisson}
\end{equation}
is
\begin{equation}
 \boxed{\displaystyle
 g=\sum_{n=1}^{d}\frac{\Lcal_n(h)}{1-r^n}\Acal_n^{(r)}.}
 \label{eq:poly-Poisson}
\end{equation}
\end{theorem}

\begin{proof}
Apply the diagonal decomposition \eqref{eq:spectral-coordinate}; centeredness is exactly the absence of the $n=0$ term.
\end{proof}

The denominators $1-r^n$ are $q$-integers up to the common factor $1-r$:
\[
 \frac1{1-r^n}=\frac1{(1-r)[n]_r}.
\]
Thus the polynomial Poisson solver is a diagonal $q$-integration operator in Appell degree.

\subsection{An Appell-Fock completion}

Define a Hilbert space $\Hcal_r^{\mathrm A}$ by completing the polynomial span under
\begin{equation}
 \left\|\sum_{n\ge0}c_n\Acal_n^{(r)}\right\|_{\Hcal_r^{\mathrm A}}^2
 :=\sum_{n\ge0}|c_n|^2.
 \label{eq:Fock-norm}
\end{equation}
This is an engineered coefficient space, not the natural $L^2(\pi_r)$ space.

\begin{theorem}[Trace-class Appell realization]
\label{thm:Fock-determinant}
For $|r|<1$, $P_r$ extends to a normal trace-class operator on $\Hcal_r^{\mathrm A}$ with
\begin{align}
 \operatorname{spec}(P_r)&=\{r^n:n\ge0\}\cup\{0\},
 \label{eq:Fock-spectrum}\\
 \tr(P_r^k)&=\frac1{1-r^k},
 \label{eq:Fock-trace}\\
 \det(I-zP_r)&=\prod_{n=0}^{\infty}(1-zr^n)
 =\qpoch{z}{r}{\infty}.
 \label{eq:Fock-det}
\end{align}
On the centered subspace $c_0=0$, the determinant is $\qpoch{zr}{r}{\infty}$.
\end{theorem}

\begin{proof}
Under the coefficient isometry to $\ell^2(\N_0)$, $P_r$ is the diagonal operator $\diag(1,r,r^2,\ldots)$. Its singular values are $1,|r|,|r|^2,\ldots$, whose sum is finite. The trace and Fredholm determinant are therefore the displayed convergent sums and products.
\end{proof}

\begin{cautionbox}[Natural versus engineered spectrum]
\Cref{thm:Fock-determinant} is fully rigorous but does not settle the spectrum of $P_r$ on $L^2(\pi_r)$, Sobolev spaces, analytic function spaces, or the repository's representation spaces. In the natural $L^2$ geometry the Appell basis is nonorthogonal, and nonreversibility later shows that $P_r$ is not self-adjoint. Natural-space spectral completeness is therefore a genuine frontier problem, not a corollary of the Fock construction.
\end{cautionbox}

\section{A fixed-point Stein method}
\label{sec:fixedpoint-stein}

\subsection{The invariant fixed-point operator}

For the target law $\pi_r$, define
\begin{equation}
 \Kcal_r f:=(I-P_r)f.
 \label{eq:fixed-Stein}
\end{equation}
Stationarity gives $\E_{\pi_r}\Kcal_rf=0$.

\begin{theorem}[Fixed-point Stein characterization]
\label{thm:Stein-characterization}
Let $\mu$ be a probability law with finite first moment. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\item $\mu=\pi_r$;
\item $\int(I-P_r)f\dd\mu=0$ for every bounded Lipschitz $f$;
\item $\mu P_r=\mu$.
\end{enumerate}
\end{theorem}

\begin{proof}
The equivalence of (ii) and (iii) is the definition of push-forward by the Markov kernel. By \cref{prop:contraction}, the invariant law is unique.
\end{proof}

\subsection{Stein equation and sharp elementary factors}

For a Lipschitz test function $h$, put $\bar h=h-\pi_rh$.

\begin{theorem}[Contractive Stein solution]
\label{thm:Stein-solution}
The series
\begin{equation}
 g_h:=\sum_{k=0}^{\infty}P_r^k\bar h
 \label{eq:Stein-series}
\end{equation}
converges modulo constants, solves
\begin{equation}
 (I-P_r)g_h=\bar h,
 \label{eq:Stein-equation}
\end{equation}
and satisfies
\begin{equation}
 \Lip(g_h)\le\frac{\Lip(h)}{1-|r|}.
 \label{eq:Stein-factor}
\end{equation}
For $r=1/2$, the factor is $2$.
\end{theorem}

\begin{proof}
The common-innovation coupling gives $\Lip(P_r^kh)\le|r|^k\Lip(h)$. The geometric series yields \eqref{eq:Stein-factor}, and telescoping the partial sums yields \eqref{eq:Stein-equation}. A fixed normalization, such as $g_h(0)=0$, removes the irrelevant additive constant.
\end{proof}

Define the fixed-point Stein discrepancy
\begin{equation}
 \Delta_r(\mu):=
 \sup_{\Lip(f)\le1}\left|\int(I-P_r)f\dd\mu\right|.
 \label{eq:Stein-discrepancy}
\end{equation}

\begin{theorem}[Equivalence with Wasserstein distance]
\label{thm:discrepancy-W1}
For every $\mu$ with finite first moment,
\begin{equation}
 \boxed{\displaystyle
 (1-|r|)W_1(\mu,\pi_r)
 \le\Delta_r(\mu)
 \le(1+|r|)W_1(\mu,\pi_r).}
 \label{eq:W1-equivalence}
\end{equation}
In the dyadic case,
\begin{equation}
 W_1(\mu,\pi_{1/2})\le2\Delta_{1/2}(\mu),
 \qquad
 \Delta_{1/2}(\mu)\le\frac32W_1(\mu,\pi_{1/2}).
 \label{eq:dyadic-W1-equivalence}
\end{equation}
\end{theorem}

\begin{proof}
For $\Lip(h)\le1$, use \cref{thm:Stein-solution}:
\[
 |\mu h-\pi_rh|
 =|\mu(I-P_r)g_h|
 \le\frac{\Delta_r(\mu)}{1-|r|}.
\]
Taking the Kantorovich--Rubinstein supremum proves the first inequality. Conversely, if $\Lip(f)\le1$, then
\[
 \Lip((I-P_r)f)\le1+|r|,
\]
and $\pi_r(I-P_r)f=0$. Therefore
\[
 |\mu(I-P_r)f|
 =|(\mu-\pi_r)(I-P_r)f|
 \le(1+|r|)W_1(\mu,\pi_r).
\]
\end{proof}

\begin{resultbox}[A computable approximation metric]
Any approximation to the Rvachev law can now be assessed by a residual of its own fixed-point equation. The discrepancy is not merely one-sided: it is quantitatively equivalent to $W_1$ with explicit constants. This supplies a direct route from spline, finite-product, quadrature, or sampling approximants to transport error estimates.
\end{resultbox}

For polynomial tests, \cref{thm:polynomial-resolvent} turns \eqref{eq:Stein-series} into the finite closed form \eqref{eq:poly-Poisson}. Thus the abstract contractive Stein equation and the exact $q$-Appell Poisson equation are the same object at two levels of regularity.

\section{The differential-dilation Stein operator}
\label{sec:differential-stein}

\subsection{A local/nonlocal hybrid}

Differentiate before applying the fixed-point operator:
\begin{equation}
 \Scal_r:=(I-P_r)D.
 \label{eq:S-general-def}
\end{equation}
Since
\begin{equation}
 P_rf'(x)=
 \frac{f(rx+1-r)-f(rx-(1-r))}{2(1-r)},
 \label{eq:Pr-derivative}
\end{equation}
we obtain
\begin{equation}
 \boxed{\displaystyle
 \Scal_rf(x)=f'(x)-
 \frac{f(rx+1-r)-f(rx-(1-r))}{2(1-r)}.}
 \label{eq:S-general}
\end{equation}
At $r=1/2$ this becomes
\begin{equation}
 \boxed{\displaystyle
 \Scal f(x)=f'(x)-f\!\left(\frac{x+1}{2}\right)
 +f\!\left(\frac{x-1}{2}\right).}
 \label{eq:S-dyadic}
\end{equation}

\begin{theorem}[Differential-dilation Stein identity]
\label{thm:differential-Stein}
For every sufficiently regular $f$,
\begin{equation}
 \E\Scal_rf(Y_r)=0.
 \label{eq:S-identity}
\end{equation}
On Appell eigenmodes,
\begin{equation}
 \boxed{\displaystyle
 \Scal_r\Acal_n^{(r)}
 =n(1-r^{n-1})\Acal_{n-1}^{(r)}.}
 \label{eq:S-lowering}
\end{equation}
In particular, $\Scal_r$ annihilates constants and affine functions.
\end{theorem}

\begin{proof}
Stationarity gives $\E(I-P_r)g(Y_r)=0$ with $g=f'$. For the lowering law, use $D\Acal_n=n\Acal_{n-1}$ and $P_r\Acal_{n-1}=r^{n-1}\Acal_{n-1}$.
\end{proof}

The affine kernel means that \eqref{eq:S-identity} alone is not a full characterization without additional moment or normalization conditions. The fixed-point operator $I-P_r$ is the cleaner characterizing Stein operator; $\Scal_r$ is the better bridge to the differential equation and Appell ladder.

\subsection{Rvachev's equation as the adjoint Stein equation}

\begin{theorem}[Weak adjoint equivalence]
\label{thm:adjoint-DDE}
For a compactly supported smooth density $v$, the identity
\begin{equation}
 \int_{\R}\Scal f(x)v(x)\dd x=0
 \quad\text{for all }f\in C_c^\infty(\R)
 \label{eq:weak-Stein}
\end{equation}
is equivalent to
\begin{equation}
 v'(x)=2\bigl(v(2x+1)-v(2x-1)\bigr)
 \label{eq:adjoint-dde}
\end{equation}
in the distributional sense.
\end{theorem}

\begin{proof}
Integrate the derivative term in \eqref{eq:S-dyadic} by parts. In the two dilation terms use $x=2y-1$ and $x=2y+1$, respectively. The resulting coefficient of the arbitrary test function is
\[
 -v'(y)-2v(2y-1)+2v(2y+1).
\]
Its vanishing is exactly \eqref{eq:adjoint-dde}.
\end{proof}

Thus Rvachev's defining differential-dilation equation is literally the adjoint stationarity equation of the differentiated affine Stein operator. This makes the probabilistic fixed point and the analytic refinement equation two faces of one operator identity.

\section{Martingales, lag correlations, and irreversibility}
\label{sec:martingales}

\subsection{Polynomial and exponential martingales}

\begin{theorem}[Appell eigenmartingales]
\label{thm:Appell-martingales}
Assume $r\ne0$ and let $(X_k)$ be the stationary or nonstationary affine chain \eqref{eq:general-AR}. For every $n\ge0$,
\begin{equation}
 M_k^{(n)}:=r^{-nk}\Acal_n(X_k)
 \label{eq:poly-martingale}
\end{equation}
is a martingale with respect to the natural filtration. In the dyadic case this is $2^{nk}\Acal_n(X_k)$.
\end{theorem}

\begin{proof}
By \cref{thm:koopman-eigen},
\[
 \E[M_{k+1}^{(n)}\mid X_k]
 =r^{-n(k+1)}P_r\Acal_n(X_k)
 =r^{-nk}\Acal_n(X_k).
\]
\end{proof}

The whole ladder is packaged by a Wald-type exponential martingale.

\begin{theorem}[Exponential Appell martingale]
\label{thm:exponential-martingale}
For real $t$ (and for complex $t$ away from zeros of the relevant MGFs),
\begin{equation}
 Z_k(t):=\frac{\exp(r^{-k}tX_k)}{M(r^{-k}t)}
 \label{eq:exp-martingale}
\end{equation}
is a martingale. Its Taylor coefficients in $t$ are the martingales \eqref{eq:poly-martingale}.
\end{theorem}

\begin{proof}
Put $s=r^{-(k+1)}t$. The MGF cocycle $M(s)=M(rs)m_\xi(s)$ gives
\[
 \E[Z_{k+1}(t)\mid X_k]
 =\frac{\e^{rsX_k}m_\xi(s)}{M(s)}
 =\frac{\e^{r^{-k}tX_k}}{M(r^{-k}t)}.
\]
Expansion of \eqref{eq:Appell-gen} gives the coefficient statement.
\end{proof}

For imaginary $t$, the denominator is the Rvachev sinc product. Hence the repository's Fourier product normalizes a complex exponential into an exact affine-chain martingale, except at its known zero divisor.

\subsection{The Appell Gram kernel}

Define
\begin{equation}
 G_{mn}^{(r)}:=\E\bigl[\Acal_m^{(r)}(Y_r)\Acal_n^{(r)}(Y_r)\bigr].
 \label{eq:Gram}
\end{equation}
The bivariate generating function is immediate but highly useful.

\begin{proposition}[Exact Gram generating function]
\label{prop:Gram-EGF}
\begin{equation}
 \boxed{\displaystyle
 \sum_{m,n\ge0}G_{mn}^{(r)}\frac{s^m}{m!}\frac{t^n}{n!}
 =\frac{M_r(s+t)}{M_r(s)M_r(t)}
 =\exp\bigl(K_r(s+t)-K_r(s)-K_r(t)\bigr).}
 \label{eq:Gram-EGF}
\end{equation}
In particular, $G_{mn}^{(r)}=0$ when $m+n$ is odd.
\end{proposition}

\begin{proof}
Multiply the two Appell generating functions and take expectation. Parity follows from the evenness of $K_r$.
\end{proof}

\begin{theorem}[Exact forward and reverse lag correlations]
\label{thm:lag-correlations}
For a stationary chain and $k\ge0$,
\begin{align}
 \E\bigl[\Acal_m(X_0)\Acal_n(X_k)\bigr]
 &=r^{nk}G_{mn}^{(r)},
 \label{eq:forward-lag}\\
 \E\bigl[\Acal_n(X_0)\Acal_m(X_k)\bigr]
 &=r^{mk}G_{mn}^{(r)}.
 \label{eq:reverse-lag}
\end{align}
\end{theorem}

\begin{proof}
Condition on $X_0$ and apply \eqref{eq:conditional-eigen}; then use symmetry of the Gram matrix.
\end{proof}

The decay exponent is determined by the degree at the \emph{future} time. This orientation dependence is invisible in ordinary stationary moments and supplies an exact test for reversibility.

\subsection{An exact dyadic nonreversibility certificate}

Extracting the coefficient $s^2t^4$ from \eqref{eq:Gram-EGF} gives the universal identity
\begin{equation}
 G_{24}^{(r)}=\kappa_6(Y_r)+8\kappa_2(Y_r)\kappa_4(Y_r).
 \label{eq:G24-cumulant}
\end{equation}
For $r=1/2$,
\begin{equation}
 \boxed{\displaystyle
 G_{24}=-\frac{128}{33075}.}
 \label{eq:G24-dyadic}
\end{equation}
Therefore at lag one,
\begin{align}
 \E[\Acal_2(X_0)\Acal_4(X_1)]
 &=-\frac8{33075},
 \label{eq:forward-witness}\\
 \E[\Acal_4(X_0)\Acal_2(X_1)]
 &=-\frac{32}{33075},
 \label{eq:reverse-witness}\\
 \text{forward}-\text{reverse}
 &=\frac8{11025}.
 \label{eq:witness-difference}
\end{align}

\begin{theorem}[Nonreversibility of the Rvachev affine chain]
\label{thm:dyadic-nonreversible}
The stationary chain
\(
 X_{k+1}=(X_k+U_{k+1})/2
\)
is not reversible. Equivalently, its Koopman operator is not self-adjoint on $L^2(\pi_{1/2})$.
\end{theorem}

\begin{proof}
For a reversible stationary chain, $\E[f(X_0)g(X_1)]=\E[g(X_0)f(X_1)]$ for all square-integrable $f,g$. Equations \eqref{eq:forward-witness} and \eqref{eq:reverse-witness} contradict this with $f=\Acal_2$ and $g=\Acal_4$.
\end{proof}

This also explains why the Appell eigenfunctions are not mutually orthogonal in the natural stationary inner product. A self-adjoint operator with distinct real eigenvalues would have orthogonal eigenspaces, whereas $G_{24}\ne0$ couples the eigenvalues $r^2$ and $r^4$.

\subsection{A general affine-uniform theorem}

Substituting \eqref{eq:r-cumulants} into \eqref{eq:G24-cumulant} and factoring gives
\begin{equation}
 \boxed{\displaystyle
 G_{24}^{(r)}
 =-\frac{16(r-1)^4(12r^4+7r^2+2)}
 {315(r+1)^2(r^2+1)(r^2-r+1)(r^2+r+1)}.}
 \label{eq:G24-factor}
\end{equation}
Every factor in the denominator is positive for $-1<r<1$, and the quartic in the numerator is positive.

\begin{theorem}[Uniform affine chains are generically irreversible]
\label{thm:general-nonreversible}
For every $r$ with $0<|r|<1$, the stationary affine-uniform chain \eqref{eq:r-chain} is nonreversible. The endpoint $r=0$ is the independent chain and is reversible.
\end{theorem}

\begin{proof}
Equation \eqref{eq:G24-factor} gives $G_{24}^{(r)}<0$. By \cref{thm:lag-correlations}, the two lag-one mixed moments are $r^4G_{24}^{(r)}$ and $r^2G_{24}^{(r)}$, which differ because $r^4\ne r^2$ for $0<|r|<1$. At $r=0$, every new state is an independent uniform innovation.
\end{proof}

\section{The scalar Stein kernel in Fabius coordinates}
\label{sec:scalar-kernel}

\subsection{Definition and exact Fabius formula}

For a centered density $u$ on $[-1,1]$, the canonical one-dimensional Stein kernel is
\begin{equation}
 \tau(x)=\frac1{u(x)}\int_x^1 y u(y)\dd y,
 \qquad -1<x<1.
 \label{eq:Stein-kernel-def}
\end{equation}
It satisfies
\begin{equation}
 \E[Yf(Y)]=\E[\tau(Y)f'(Y)]
 \label{eq:classical-Stein}
\end{equation}
for smooth $f$. In the Rvachev case, evenness gives $\tau(-x)=\tau(x)$.

\begin{lemma}[Two Fabius primitives]
\label{lem:F-primitives}
For $0\le q\le1$,
\begin{align}
 \int_0^qF(s)\dd s&=F(q/2),
 \label{eq:F-primitive0}\\
 \int_0^q sF(s)\dd s&=qF(q/2)-2F(q/4).
 \label{eq:F-primitive1}
\end{align}
\end{lemma}

\begin{proof}
Both right-hand sides vanish at $q=0$. Differentiate and use $F'(x)=2F(2x)$ at $x=q/2$ and $x=q/4$.
\end{proof}

\begin{theorem}[Exact Fabius Stein kernel]
\label{thm:exact-tau}
For $-1<x<1$,
\begin{equation}
 \boxed{\displaystyle
 \tau(x)=
 \frac{|x|F((1-|x|)/2)+2F((1-|x|)/4)}{F(1-|x|)}.}
 \label{eq:tau-global}
\end{equation}
Equivalently, for $0<q\le1$,
\begin{equation}
 \boxed{\displaystyle
 \tau(1-q)=
 \frac{(1-q)F(q/2)+2F(q/4)}{F(q)}.}
 \label{eq:tau-q}
\end{equation}
In particular,
\begin{equation}
 \tau(0)=\frac5{36},
 \qquad
 \tau(1/2)=\frac1{12}.
 \label{eq:tau-special}
\end{equation}
Every dyadic value $\tau(1-a/2^N)$ is rational whenever the corresponding Fabius values are rational.
\end{theorem}

\begin{proof}
For $x=1-q\ge0$, substitute $s=1-y$ in \eqref{eq:Stein-kernel-def} and use $u(y)=F(1-y)$:
\[
 \int_{1-q}^{1}yu(y)\dd y
 =\int_0^q(1-s)F(s)\dd s.
\]
Apply \cref{lem:F-primitives} and divide by $u(1-q)=F(q)$. Evenness gives the global formula. The special values use $F(1/2)=1/2$, $F(1/4)=5/72$, and $F(1/8)=1/288$.
\end{proof}

The exact kernel links a classical local Stein identity to the repository's dyadic arithmetic. Unlike the fixed-point operator $I-P$, it is a coefficient in a first-order differential identity; the two Stein structures are complementary.

\subsection{An inverse-Fabius identity}

Let $G=F^{-1}:(0,1)\to(0,1)$.

\begin{theorem}[Quantile bridge]
\label{thm:tau-quantile}
For $0<y<1$ and $q=G(y)$,
\begin{equation}
 \boxed{\displaystyle
 \tau(1-G(y))
 =G(y)-\frac{G(y)^2}{2}
 -\frac1y\int_0^y
 \left(G(v)-\frac{G(v)^2}{2}\right)\dd v.}
 \label{eq:tau-quantile}
\end{equation}
\end{theorem}

\begin{proof}
Let $H(s)=s-s^2/2$. The numerator in \eqref{eq:tau-q} is
\[
 \int_0^q(1-s)F(s)\dd s
 =H(q)F(q)-\int_0^qH(s)F'(s)\dd s.
\]
Use $v=F(s)$ in the last integral and $F(q)=y$; then divide by $y$.
\end{proof}

Equation \eqref{eq:tau-quantile} turns endpoint information about $G$ into endpoint information about $\tau$ without evaluating the density or differentiating a rapidly varying function.

\subsection{Bell--Bernoulli moments of the kernel}

\begin{theorem}[Stein-kernel/Appell transform]
\label{thm:tau-Appell}
For $t$ near zero,
\begin{equation}
 \boxed{\displaystyle
 \E\left[\tau(Y)\frac{\e^{tY}}{M(t)}\right]
 =\frac{K'(t)}t.}
 \label{eq:tau-EGF}
\end{equation}
Consequently, for every $n\ge0$,
\begin{equation}
 \boxed{\displaystyle
 \E[\tau(Y)\Acal_n(Y)]
 =\frac{\kappa_{n+2}}{n+1}.}
 \label{eq:tau-Appell-moment}
\end{equation}
For $n=2m$ this is
\begin{equation}
 \E[\tau(Y)\Acal_{2m}(Y)]
 =\frac{2^{2m+1}B_{2m+2}}
 {(m+1)(2m+1)(4^{m+1}-1)},
 \label{eq:tau-Bernoulli}
\end{equation}
and the odd moments vanish.
\end{theorem}

\begin{proof}
Apply the classical Stein identity \eqref{eq:classical-Stein} to $f(y)=\e^{ty}/M(t)$. The left side is $M'(t)/M(t)=K'(t)$; the right side is $t$ times the expectation in \eqref{eq:tau-EGF}. Expand both sides in powers of $t$ and compare coefficients. Insert \eqref{eq:cumulants} for \eqref{eq:tau-Bernoulli}.
\end{proof}

The case $n=0$ gives
\begin{equation}
 \E\tau(Y)=\kappa_2=\Var(Y)=\frac19,
 \label{eq:mean-tau}
\end{equation}
as expected. Higher coefficients show that the scalar Stein kernel is an exact Bernoulli probe of the nonorthogonal Appell basis.

\section{Lambert--periodic endpoint asymptotics of the Stein kernel}
\label{sec:endpoint}

\subsection{Imported inverse elasticity}

Put
\begin{equation}
 L=\log2,
 \qquad
 T=\log(1/y),
 \qquad
 \rho=\sqrt{\frac{2T}{L}},
 \qquad
 \theta=\rho+\frac1L+\frac12.
 \label{eq:endpoint-vars}
\end{equation}
The repository's all-orders inverse analysis gives, as $y\downarrow0$,
\begin{equation}
 \mathcal E(y):=\frac{yG'(y)}{G(y)}
 =\frac1\rho+
 \frac{\Psi'(\theta)-1}{L\rho^2}
 +\BigO\!\left(\frac{1+\log\rho}{\rho^3}\right),
 \label{eq:elasticity-input}
\end{equation}
where $\Psi$ is the smooth mean-zero one-periodic Gamma--zeta fluctuation in the forward endpoint expansion. This is an imported analytic input, not reproved here.

\subsection{A Watson ratio for the Fabius primitive}

Define
\begin{equation}
 R(q):=\frac{F(q/2)}{F(q)}.
 \label{eq:Rq}
\end{equation}
If $q=G(y)$, integration by parts in \eqref{eq:F-primitive0} gives
\begin{equation}
 R(G(y))=G(y)-\frac1y\int_0^yG(v)\dd v.
 \label{eq:R-inverse}
\end{equation}
Write $q(T)=G(\e^{-T})$ and $E(T)=\mathcal E(\e^{-T})$. Since $q'(T)=-E(T)q(T)$,
\begin{equation}
 \frac{R(q(T))}{q(T)}
 =1-\int_0^\infty \e^{-s}
 \exp\!\left(-\int_0^sE(T+a)\dd a\right)\dd s.
 \label{eq:Watson-exact}
\end{equation}

\begin{lemma}[Two-term primitive ratio]
\label{lem:R-asymptotic}
As $T\to\infty$,
\begin{equation}
 \frac{R(q(T))}{q(T)}
 =E(T)-E(T)^2
 +\BigO\!\left(\frac{1+\log\rho}{\rho^3}\right).
 \label{eq:R-E}
\end{equation}
Therefore
\begin{equation}
 \frac{R(G(y))}{G(y)}
 =\frac1\rho+
 \frac{(\Psi'(\theta)-1)/L-1}{\rho^2}
 +\BigO\!\left(\frac{1+\log\rho}{\rho^3}\right).
 \label{eq:R-rho}
\end{equation}
\end{lemma}

\begin{proof}
In \eqref{eq:Watson-exact}, the weight $\e^{-s}$ localizes $s$ to a bounded range. On such ranges, \eqref{eq:elasticity-input} implies
$E(T+s)=E(T)+\BigO((1+\log\rho)/\rho^3)$ uniformly. Replacing the inner integral by $sE(T)$ changes the integral by the stated error. The remaining elementary integral is
\[
 1-\int_0^\infty\e^{-(1+E(T))s}\dd s
 =\frac{E(T)}{1+E(T)}
 =E(T)-E(T)^2+\BigO(E(T)^3).
\]
Insert \eqref{eq:elasticity-input}.
\end{proof}

The ratio also satisfies an exact first-order equation useful for an all-orders expansion. If
\begin{equation}
 r_*(T):=\frac{R(q(T))}{q(T)},
 \label{eq:rstar}
\end{equation}
then differentiation of \eqref{eq:R-inverse} gives
\begin{equation}
 r_*'(T)=(1+E(T))r_*(T)-E(T),
 \qquad
 r_*(T)=\frac{E(T)+r_*'(T)}{1+E(T)}.
 \label{eq:rstar-ode}
\end{equation}
This identity is the recursion seed for the conjectural full transseries in \cref{sec:conjectures}.

\subsection{The endpoint theorem}

Equation \eqref{eq:tau-q} factors as
\begin{equation}
 \tau(1-q)=R(q)\bigl(1-q+2R(q/2)\bigr).
 \label{eq:tau-factor-R}
\end{equation}
Since $q=G(y)$ is exponentially small in $\rho$ while $R(q/2)=\BigO(q/\rho)$, the factor in parentheses differs from one by less than any fixed algebraic power of $1/\rho$.

\begin{theorem}[Two-term Lambert-periodic Stein-kernel law]
\label{thm:tau-endpoint}
As $y\downarrow0$,
\begin{equation}
 \boxed{\displaystyle
 \tau(1-G(y))=G(y)\left[
 \frac1\rho+
 \frac{(\Psi'(\theta)-1)/L-1}{\rho^2}
 +\BigO\!\left(\frac{1+\log\rho}{\rho^3}\right)
 \right].}
 \label{eq:tau-endpoint}
\end{equation}
By evenness, the same profile governs the left endpoint $x\downarrow-1$.
\end{theorem}

\begin{proof}
Combine \cref{lem:R-asymptotic} with \eqref{eq:tau-factor-R}. The correction $-q+2R(q/2)$ is exponentially smaller than the displayed algebraic terms.
\end{proof}

The first coefficient is universal, while the second contains the derivative of the same Gamma--zeta fluctuation that controls the inverse Fabius function. The extra $-1$ is the Watson correction $-E^2$ from averaging the quantile over an exponential window.

\section{Signed bases and the broader geometric family}
\label{sec:signed-bases}

\subsection{Negative contraction parameters}

The operator framework remains meaningful for $-1<r<0$. The stationary law is still symmetric and compactly supported, but the contraction reverses orientation at every step. Put $r=-a$ with $0<a<1$ and
\begin{equation}
 c_a:=\frac{1+a}{1-a}.
 \label{eq:ca}
\end{equation}
Because the independent uniforms are symmetric, the alternating signs in the stationary series can be absorbed digit by digit.

\begin{proposition}[Positive/negative-base conjugacy]
\label{prop:negative-conjugacy}
For $0<a<1$,
\begin{align}
 Y_{-a}&\stackrel d=c_aY_a,
 \label{eq:Y-negative-scale}\\
 M_{-a}(t)&=M_a(c_at),
 \label{eq:M-negative-scale}\\
 \Acal_{n}^{(-a)}(x)&=c_a^n\Acal_n^{(a)}(x/c_a).
 \label{eq:A-negative-scale}
\end{align}
Nevertheless the Koopman eigenvalues retain the sign:
\begin{equation}
 P_{-a}\Acal_n^{(-a)}=(-a)^n\Acal_n^{(-a)}.
 \label{eq:negative-eigen}
\end{equation}
\end{proposition}

\begin{proof}
From \eqref{eq:Yr},
\[
 Y_{-a}=(1+a)\sum_{j\ge0}(-a)^jU_{j+1}
 \stackrel d=(1+a)\sum_{j\ge0}a^jU_{j+1}=c_aY_a.
\]
The MGF and Appell scaling follow. The eigenvalue law is \cref{thm:koopman-eigen}.
\end{proof}

Thus the negative base has a positive-law geometry but an alternating transfer spectrum. Its finite determinant is
\begin{equation}
 \det(I-zP_{-a}\vert_{\Pi_d})=(z;-a)_{d+1}.
 \label{eq:negative-det}
\end{equation}
The even/odd splitting of this product provides a natural dynamical companion to the repository's negative-$q$ and position-parity Thue--Morse formulas.

\section{Conjectures and frontier research directions}
\label{sec:conjectures}

\subsection{Natural-space transfer spectra}

The Appell polynomials span all polynomials, hence form a total set in $L^2(\pi_r)$ because the target has compact support. They are exact eigenvectors, but they are not orthogonal and, by \cref{thm:general-nonreversible}, the operator is nonnormal for $0<|r|<1$. The engineered space in \cref{thm:Fock-determinant} avoids this geometry by definition.

\begin{conjecture}[Natural analytic nuclearity]
\label{conj:natural-nuclearity}
For every $0<|r|<1$, there exists a canonical scale of analytic or Gevrey spaces adapted to the endpoint flatness of $u_r$ such that:
\begin{enumerate}[label=(\roman*)]
\item $P_r$ is nuclear;
\item the Appell polynomials form a Schauder basis;
\item the spectrum is exactly $\{r^n:n\ge0\}\cup\{0\}$;
\item the Fredholm determinant is $(z;r)_\infty$ without an engineered coefficient norm.
\end{enumerate}
\end{conjecture}

A plausible norm should balance derivative growth against the endpoint saddle and should make the coefficient functional $p\mapsto\E p^{(n)}(Y_r)/n!$ continuous with explicit $n$-dependence.

\begin{problem}[Natural $L^2$ spectrum and pseudospectrum]
Determine the full spectrum of $P_{1/2}$ on $L^2(u\dd x)$ and quantify the norms of the finite-degree spectral projectors \eqref{eq:projector-dual}. The nonzero off-diagonal Gram entries imply potentially large eigenbasis condition numbers. Relate their growth to the repository's Legendre, Lagrange, and shifted-$u$ synthesis matrices.
\end{problem}

\subsection{A full Stein-kernel transseries}

Equation \eqref{eq:rstar-ode} converts the all-orders inverse elasticity directly into an all-orders kernel expansion.

\begin{conjecture}[Phase-aware Stein-kernel expansion]
\label{conj:tau-transseries}
For each fixed $N$,
\begin{equation}
 \tau(1-G(y))
 =G(y)\left[
 \sum_{j=1}^{N}
 \frac{T_j(\theta,\log\rho)}{\rho^j}
 +\BigO_N\!\left(\frac{(1+\log\rho)^{N}}{\rho^{N+1}}\right)
 \right],
 \label{eq:tau-transseries}
\end{equation}
where every $T_j$ is a polynomial in $\log\rho$ with smooth one-periodic coefficients generated recursively from the inverse-elasticity coefficients. The first two are
\begin{equation}
 T_1=1,
 \qquad
 T_2=\frac{\Psi'(\theta)-1}{\log2}-1.
 \label{eq:T12}
\end{equation}
The highest periodic derivative and highest logarithmic degree should obey triangular laws analogous to the repository's inverse top-jet and top-log laws.
\end{conjecture}

The formal recursion is immediate from \eqref{eq:rstar-ode}; the hard part is a derivative-stable remainder theorem uniform over the exponentially weighted Watson window.

\begin{conjecture}[Kernel shape]
\label{conj:kernel-shape}
The Rvachev Stein kernel is strictly decreasing on $[0,1)$:
\begin{equation}
 \tau'(x)<0\qquad(0<x<1),
 \label{eq:tau-monotone}
\end{equation}
so its unique maximum is $\tau(0)=5/36$ and it decreases to zero with the Lambert law \eqref{eq:tau-lambert}.
\end{conjecture}

Exact dyadic values, the even log-concave convolution structure, and numerical evaluations support this statement, but a direct proof must control the logarithmic derivative of $F$ across all scales, not only asymptotically.

\subsection{Stein discrepancies for the finite sinc/spline hierarchy}

The repository obtains exact transport distances for several finite convolution and spline approximants. The fixed-point discrepancy suggests a more structural target.

\begin{problem}[Exact residuals for finite approximants]
For each finite sinc-product or box-spline law $\mu_N$, compute
\begin{equation}
 \Delta_{1/2}(\mu_N)
 =\sup_{\Lip f\le1}|\mu_N(I-P)f|
 \label{eq:finite-discrepancy}
\end{equation}
exactly or asymptotically. Determine whether the extremizer is piecewise linear on the Thue--Morse knot set and whether the exact $W_1$ collapse found in the repository is visible directly from the fixed-point residual.
\end{problem}

For Richardson-extrapolated signed measures, replace the supremum by the dual bounded-Lipschitz norm. The $q$-Poisson solver may expose cancellations not visible in absolute transport metrics.

\subsection{Thue--Morse projectors}

The dual projector \eqref{eq:projector-dual} pairs a test polynomial with $u^{(n)}$. Repository formulas express finite derivative approximants as signed atoms or truncated powers with Thue--Morse signs.

\begin{conjecture}[Automatic spectral extraction]
There are finite Thue--Morse signed measures $\sigma_{n,N}$, supported on the dyadic knot set, such that
\begin{equation}
 \sigma_{n,N}\longrightarrow u^{(n)}
 \label{eq:sigma-converge}
\end{equation}
in an explicit negative Sobolev norm at a supergeometric rate, and the finite functionals
\begin{equation}
 p\longmapsto\frac{(-1)^n}{n!}\langle\sigma_{n,N},p\rangle
 \label{eq:finite-projector}
\end{equation}
are exact on a degree range increasing with $N$. Gaussian-binomial weights should give accelerated versions whose error determinant is controlled by $(z;1/2)_N$.
\end{conjecture}

This would turn Prouhet cancellation into a convergent spectral-projector algorithm.

\subsection{Legendre conditioning and nonnormality}

The finite Legendre--Appell matrix in \eqref{eq:legendre-appell} is triangular with rational entries. It conjugates the Markov operator on $\Pi_d$ to a diagonal matrix.

\begin{problem}[Asymptotic condition numbers]
Determine the asymptotic growth of the condition number of the Legendre--Appell change of basis and of the projector norms $\|\PiD_n\|$ in Legendre-weighted and Rvachev-weighted $L^2$ spaces. Decide whether endpoint flatness produces stretched-exponential growth, and identify the saddle governing it.
\end{problem}

A solution would quantify how a completely explicit real spectrum coexists with strong transient amplification in a nonnormal basis.

\subsection{Irreversibility beyond a single moment}

\Cref{thm:general-nonreversible} supplies a scalar witness but not a full measure of time asymmetry.

\begin{problem}[Entropy production and Appell antisymmetry]
Let $\Gamma_k^{(r)}=(r^{nk}-r^{mk})G_{mn}^{(r)}$ be the antisymmetric Appell lag matrix. Relate its Schatten norms, finite principal minors, or generating function to the relative entropy between the forward and time-reversed two-step laws. Determine the leading behavior as $r\to0$ and $r\to1^-$.
\end{problem}

The exact factor \eqref{eq:G24-factor} predicts fourth-order vanishing as $r\to1$, but other mode pairs may dominate a global entropy-production scale.

\subsection{General atomic masks and multivariate up-like functions}

The affine proof of \cref{thm:koopman-eigen} uses only a stationary MGF cocycle. It therefore extends to vector recursions
\begin{equation}
 X_{k+1}=AX_k+\xi_{k+1}
 \label{eq:vector-AR}
\end{equation}
with spectral radius $\rho(A)<1$, producing multivariate Appell eigenpolynomials organized by tensor powers of $A$. This is directly relevant to multivariate compactly supported $C^\infty$ functions generated by nonstationary subdivision \cite{CharinaContiDyn}.

\begin{conjecture}[Subdivision Stein--Koopman correspondence]
For every convergent atomic subdivision mask with a probabilistic affine realization, the reciprocal moment polynomials diagonalize the associated refinement Markov operator; the multivariate transfer determinant factors over symmetric tensor powers of the dilation matrix; and the adjoint differentiated Stein operator is the mask's distributional refinement equation.
\end{conjecture}

The scalar dyadic theory developed here is the rank-one model of this proposed correspondence.

\subsection{\texorpdfstring{Complex and reciprocal $q$}{Complex and reciprocal q}}

For complex $r$ with $|r|<1$, the MGF cocycle and formal Appell eigenlaw remain valid, although positivity is lost. The determinant $(z;r)_\infty$ is still analytic. It is natural to compare its Stokes geometry as $r$ approaches roots of unity with the repository's reciprocal-$q$ vertical-line moment functionals.

\begin{problem}[Root-of-unity transfer limits]
Construct renormalized limits of the finite transfer determinants and polynomial resolvents as $r$ approaches a root of unity radially. Determine whether Gaussian-binomial cancellation produces finite cyclic blocks and whether their signs recover position-parity refinements of Thue--Morse.
\end{problem}

\section{Formalization roadmap}
\label{sec:formalization}

The repository already formalizes the Rvachev moment/Appell family and a substantial part of the dyadic and $q$-binomial infrastructure. The new results have a favorable dependency graph.

\begin{enumerate}
\item \textbf{Polynomial affine operator.} Define $P_r$ on polynomials by integrating a polynomial against the uniform moment functional. Prove degree preservation and $DP_r=rP_rD$ by coefficient arithmetic.
\item \textbf{MGF coefficient identity.} Avoid analytic convergence initially: encode $M(t)=M(rt)m(t)$ as a formal power-series identity and derive $P_r\Acal_n=r^n\Acal_n$ coefficientwise.
\item \textbf{Spectral coordinates.} Formalize $\E\Acal_n=\delta_{n0}$ and the triangular derivative formula, then prove \eqref{eq:spectral-coordinate} by polynomial induction.
\item \textbf{$q$-commuting powers.} Reuse the existing Gaussian-binomial theorem to prove \eqref{eq:qbinomial-operator}; the only new input is the commutation relation.
\item \textbf{Finite determinant.} In the Appell basis, the matrix is diagonal. The determinant theorem reduces to the existing finite $q$-Pochhammer product.
\item \textbf{Poisson and martingale laws.} These are finite sums and one-step conditional expectations once the eigenvalue theorem is available.
\item \textbf{Irreversibility certificate.} Compute $\kappa_2,\kappa_4,\kappa_6$, prove $G_{24}=\kappa_6+8\kappa_2\kappa_4$, and normalize the rational witness $-128/33075$.
\item \textbf{Stein kernel.} Formalize the two Fabius primitives, the exact ratio \eqref{eq:tau-q}, and the quantile change of variables. These need only one-dimensional integration and existing inverse definitions.
\item \textbf{Endpoint theorem.} This is the only analytically deep layer. It should depend on the formalized lower-Lambert phase and a separate lemma controlling the exponentially weighted Watson integral.
\end{enumerate}

A compact first milestone would formalize the chain through the finite determinant and exact nonreversibility theorem without any asymptotic analysis.

\section{Conclusion}
\label{sec:conclusion}

The Fabius--Rvachev corpus already contains an unusually rich collection of exact structures: Thue--Morse signs, dyadic arithmetic, q-binomial interpolation, infinite sinc products, Bell--Bernoulli cumulants, Legendre and sampling representations, and lower-Lambert endpoint expansions. The affine smoothing chain supplies a missing common dynamical layer.

Its reciprocal-moment Appell polynomials are exact Koopman eigenfunctions. Once that observation is made, several consequences become finite and explicit: spectral coordinates are expectations of derivatives, dual coordinates are Rvachev derivatives, Poisson equations divide by $1-r^n$, Gaussian coefficients encode noncommuting operator powers, and the transfer determinant is a q-Pochhammer product. The same eigenmodes generate martingales and exact lag correlations. Their nonorthogonality is not an accident but a signature of genuine time irreversibility.

The fixed-point equation also opens a Stein-method direction. The simple residual $I-P$ characterizes the law and is quantitatively equivalent to Wasserstein distance, while $(I-P)D$ has Rvachev's differential equation as its adjoint. Independently, the scalar Stein kernel has an exact Fabius formula and an inverse-Fabius bridge. Its endpoint is governed by the same lower-Lambert phase and Gamma--zeta periodicity as the quantile, but with a new Watson correction.

The resulting research program is concrete. Its finite algebra is ready for formalization; its exact rational experiments are reproducible; its endpoint conjectures inherit a mature all-orders calculus; and its natural-space spectral, nonnormal, Thue--Morse, Legendre, and multivariate extensions are sharply posed.

\section{Derivation of the universal irreversibility witness}
\label{app:G24}

Let
\[
 \mathscr G(s,t)=\frac{M_r(s+t)}{M_r(s)M_r(t)}.
\]
Up to total degree six,
\begin{align*}
 \log\mathscr G(s,t)
 ={}&\frac{\kappa_2}{2}\bigl((s+t)^2-s^2-t^2\bigr)\\
 &+\frac{\kappa_4}{24}\bigl((s+t)^4-s^4-t^4\bigr)\\
 &+\frac{\kappa_6}{720}\bigl((s+t)^6-s^6-t^6\bigr)+\BigO_7.
\end{align*}
The coefficient of $s^2t^4/(2!4!)$ in $\exp(\log\mathscr G)$ has two contributions: the direct sixth-cumulant term and the product of the quadratic and quartic cross terms. This gives
\[
 G_{24}=\kappa_6+8\kappa_2\kappa_4.
\]
From \eqref{eq:r-cumulants},
\begin{align*}
 \kappa_2&=\frac{1-r}{3(1+r)},\\
 \kappa_4&=-\frac{2(1-r)^4}{15(1-r^4)},\\
 \kappa_6&=\frac{16(1-r)^6}{63(1-r^6)}.
\end{align*}
Putting these over a common denominator and factoring yields \eqref{eq:G24-factor}. At $r=1/2$ it reduces to $-128/33075$.
