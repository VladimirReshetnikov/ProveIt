\section{Entire same-direction order continuation}
\label{sdco:sec:continuation}
The following continuation answers the spectral-representation question for
same-direction products, without asserting generic finite closure. Full
proofs and verification records are given in \cite{SDCO2026}. Put
$F_s(x)=-\operatorname{Li}_s(-x)$ and $g(p)=\pi/\sin(\pi p)$.

\begin{theorem}[Same-direction complex-order master]
\label{sdco:thm:master}
Let $z,w>0$, $\operatorname{Re}N>0$, and
$-2<\operatorname{Re}\beta<\operatorname{Re}N$. Choose $c_1,c_2\in(0,1)$ with
$0<\operatorname{Re}\beta+c_1+c_2<\operatorname{Re}N$.
For every $s,t\in\mathbb C$,
\begin{align}
 &\int_0^\infty\frac{x^{\beta-1}F_s(zx)F_t(wx)}{(1+x)^N}\,\mathrm dx
 \label{sdco:eq:master}\\
 &\quad=\frac1{(2\pi i)^2}\int_{(c_1)}\!\int_{(c_2)}
 g(p)g(r)p^{-s}r^{-t}z^pw^r
 B(\beta+p+r,N-\beta-p-r)\,\mathrm dr\,\mathrm dp.\notag
\end{align}
Both sides are entire in $(s,t)$. The contour integral is absolute, and all
fixed spectral and $\beta$ derivatives are legitimate under the integrals.
For $z=w=1$, an independent locally normally convergent series is
\begin{equation}\label{sdco:eq:Newton}
 \sum_{m,n\ge1}A_m(s)A_n(t)B(\beta+m+n,N-\beta),\qquad
 A_n(s)=\sum_{k=1}^n(-1)^{k+1}\binom{n-1}{k-1}k^{-s}.
\end{equation}
\end{theorem}
\begin{proof}
The Fermi transform gives
$F_s(x)=(2\pi i)^{-1}\int_{(c)}g(p)p^{-s}x^p\,\mathrm dp$ for $0<c<1$;
exponential cosecant decay continues it entirely in $s$. Insert this twice
and apply the absolutely convergent beta integral. For the series, use
$u=x/(1+x)$ and the Euler transform
$F_s(u/(1-u))=\sum_{n\ge1}A_n(s)u^n$.
On compact order sets, every fixed derivative of $A_n$ is
$O((1+\log n)^M/n)$, as follows from its normalized Mellin integral
$\Gamma(s)^{-1}\int_0^\infty v^{s-1}e^{-v}(1-e^{-v})^{n-1}\,\mathrm dv$
for sufficiently large $n$. Grouping $m+n$ proves absolute local normal
convergence, including all fixed derivatives.
\end{proof}

For positive integers $m$, the exact coefficients are
$A_n(m)=H_n^\star(\{1\}^{m-1})/n$, with inclusive upper index $n$.
For nonpositive orders the coefficients have finite support, generated by
$Q_0(u)=u$ and $Q_{m+1}(u)=u(1-u)Q_m'(u)$.

\begin{theorem}[Entire connected coordinates]
\label{sdco:thm:connected}
For $\operatorname{Re}q>0$, the initially convergent expression
\[
 C_{A,B}(q)=\zeta(A,B)-Z_q(A,B)+\zeta(A,q)D_B(q),\qquad
 D_u(q)=\zeta(u,q)-\zeta(u),
\]
where $Z_q(A,B)=\sum_{n>m\ge0}(n+q)^{-A}(m+q)^{-B}$,
extends entirely in $A,B$. Its global identities include
\begin{align}
 C_{A,B}(q)+C_{B,A}(q)&=D_A(q)D_B(q)+D_{A+B}(q),\label{sdco:eq:stuffle}\\
 C_{A,B}(q+1)-C_{A,B}(q)&=-q^{-A}D_B(q),\label{sdco:eq:shift}\\
 C_{A,0}(q)&=D_A(q)-D_{A-1}(q).\label{sdco:eq:boundary}
\end{align}
At a positive integer $L$, $C_{A,B}(L)=\sum_{1\le k<j\le L-1}j^{-A}k^{-B}$.
\end{theorem}
\begin{proof}
The double Mellin kernel is the second divided difference of $y^{2-q}$
at $1,e^v,e^{v+t}$. It is jointly smooth at the two axes and all fixed
derivatives decrease exponentially. The normalized double Mellin transform
therefore extends entirely by repeated integration by parts. In its
absolutely convergent domain, geometric expansion gives the stated
connected expression. Stuffle and shift extend by holomorphy; the boundary
formula follows from the zero-index Faulhaber polynomial $X-1$.
\end{proof}

The connected Barnes-cone formula in \cite{SDCO2026}, Theorem 3.4, recovers
the inherited positive-integer finite double-Hurwitz master exactly. This
compatibility does not justify replacing an integer finite-sum upper limit
by a complex number.

\subsection{All logarithmic weights and the depth cancellation}
Let $\operatorname{Comp}(m)$ denote the compositions of $m$ into positive
parts. Put $c_0=1$ and $c_v=2(1-2^{1-2v})\zeta(2v)$ for $v\ge1$.
For $\operatorname{Re}s>2$, define
\begin{equation}\label{sdco:eq:T}
\begin{split}
 T_m(s)={}&\sum_{2v+j+d=m}
 c_v\frac{(-1)^d(s-1)_j}{j!}\zeta(s+j-1,\{1\}^d)\\
 &-(-1)^m\sum_{\alpha\in\operatorname{Comp}(m)}\zeta(s-1,\alpha).
\end{split}
\end{equation}
The first sum is over nonnegative $v,j,d$; an empty inner index denotes an
ordinary zeta function.

\begin{theorem}[All logarithmic moments]
\label{sdco:thm:logarithmic}
For every integer $r\ge0$ and initially $\operatorname{Re}s>2$,
\begin{equation}\label{sdco:eq:logarithmic}
 \int_0^\infty\frac{F_s(x)\log^r(1+x)}{(1+x)^2}\,\mathrm dx
 =r!\sum_{m=1}^{r+1}T_m(s).
\end{equation}
Identical terms in \eqref{sdco:eq:T} cancel so that the remaining multiple
zeta functions have depth at most $r+1$. The complete expression has an
entire continuation in $s$, equal to the ordinary integral.
\end{theorem}
\begin{proof}
Expand the paired Gamma-residue generator of \cite{SDCO2026}, Theorem 6.1.
Its strict and weak finite harmonic products respectively yield the first
sum and the composition sum in \eqref{sdco:eq:T}. The even factor
$\Gamma(1+\lambda)\Gamma(1-\lambda)=\sum c_v\lambda^{2v}$ must be retained.
The coefficient of $\lambda^r$ after division by $\lambda(1-\lambda)$ is
\eqref{sdco:eq:logarithmic}. The only depth-$m+1$ term of the first sum,
$(-1)^m\zeta(s-1,\{1\}^m)$, cancels the all-ones composition in the second.
Ordinary-integral holomorphy fixes the entire continuation of the combination.
\end{proof}
In particular the dilogarithm has the exact all-index specialization
\[
 \int_0^\infty\frac{-\operatorname{Li}_2(-x)\log^r(1+x)}{(1+x)^2}\,\mathrm dx
 =r!\sum_{j=0}^r(j+1)\zeta(j+2).
\]
This follows independently from the trigamma generator
$\int_0^\infty F_2(x)(1+x)^{\lambda-2}\,\mathrm dx
=\psi'(1-\lambda)/(1-\lambda)$ for $\operatorname{Re}\lambda<1$.

\subsection{An ordinary Stieltjes--Gamma boundary kernel}
The boundary in \eqref{sdco:eq:boundary} is the normalized Mellin transform of
\[
 W_q(v)=\frac{e^{-2v}+(1-q)e^{-qv}-(2-q)e^{-(q+1)v}}{(1-e^{-v})^2},
 \qquad W_q(0)=\frac{(q-1)(q-2)}2.
\]
Differentiation at Mellin order one gives, for $\operatorname{Re}q>0$,
\begin{align*}
 \int_0^\infty W_q(v)\,\mathrm dv&=q-1-\psi(q)-\gamma,\\
 \int_0^\infty(\log v+\gamma)W_q(v)\,\mathrm dv
 &=\gamma_1-\gamma_1(q)-\log\Gamma(q).
\end{align*}
The integrals are ordinary and absolute. The all-index Bell-polynomial
version and every anchored primitive are given in \cite{SDCO2026},
Theorems 7.1--7.2. For example,
\[
 \int_1^q\int_0^\infty W_a(v)\,\mathrm dv\,\mathrm da
 =\frac{(q-1)^2}{2}-\log\Gamma(q)-\gamma(q-1).
\]

\paragraph{Status boundary.}
These results prove a complex-order representation and specified finite
reductions. They do not prove generic finite closure, arithmetic minimal
depth, or the Gaussian $S_6$ and revised $S_8$ identities. Numerical checks
are diagnostics rather than interval certificates. Six listed incoming
archives were not inspected; provenance and overlap limits are documented
in the standalone source audit.
