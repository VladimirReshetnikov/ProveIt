def need(ok, message):
    if not ok:
        raise ValueError(message)

def exact(a,b):
    if type(a) is not type(b): return False
    if type(a) is dict: return a.keys()==b.keys() and all(type(k) is str and exact(a[k],b[k]) for k in a)
    if type(a) in (tuple,list): return len(a)==len(b) and all(exact(x,y) for x,y in zip(a,b))
    return a==b

@contextmanager
def dependencies(root):
    root=Path(root).resolve();papers=root.parents[1];verification=papers/'verification'
    for name,(relative,digest) in PINS.items():
        need(hashlib.sha256((papers/relative).read_bytes()).hexdigest()==digest,'source pin '+name)
    stems={p.stem for folder in (root,verification) for p in folder.glob('*.py')}
    prior={name:sys.modules[name] for name in stems if name in sys.modules}
    oldpath=list(sys.path);oldmeta=list(sys.meta_path)
    class SourceOnlyLoader(importlib.abc.Loader):
        def __init__(self,name,path):self.name,self.path=name,path
        def create_module(self,spec):return None
        def exec_module(self,module):
            data=self.path.read_bytes()
            need(hashlib.sha256(data).hexdigest()==PINS[self.name][1],'executed source pin '+self.name)
            module.__file__=str(self.path)
            exec(compile(data,str(self.path),'exec'),module.__dict__)
    class AuthenticatedFinder(importlib.abc.MetaPathFinder):
        def find_spec(self,fullname,path=None,target=None):
            if fullname in stems:
                need(fullname in PINS,'unlisted local dependency '+fullname)
                location=papers/PINS[fullname][0]
                return importlib.util.spec_from_file_location(fullname,location,loader=SourceOnlyLoader(fullname,location))
            return None
    try:
        for name in stems:sys.modules.pop(name,None)
        sys.path[:0]=[str(root),str(verification)];sys.meta_path.insert(0,AuthenticatedFinder())
        yield importlib.import_module('pcp_affine_slope_class_history')
    finally:

def table(program):
    need(type(program) is tuple and program and all(type(n) is int and n>=0 for n in program),'fixed natural tuple')
    return tuple(row for n in program for row in ((2,0,1,0),(2,1,2*4**n,2*(4**n-1)//3)))

def formal_bound(source,names):
    degree={name:1 for name in names}
    for name,op,a,b in source:
        da=degree[a] if type(a) is str else 0;db=degree[b] if type(b) is str else 0
        degree[name]=da+db if op=='*' else max(da,db)
    return degree

def build(program=(0,1,1),*,unit_product=True,root=None):
    need(type(unit_product) is bool,'exact Boolean finalizer')
    maps=table(program)
    root=Path(__file__).resolve().parent if root is None else Path(root)
    with dependencies(root) as component:
        return _build_authenticated(program,unit_product,component)

def _build_authenticated(program,unit_product,component):
    maps=table(program)
    raw,adapter_hash=adapted(component)
    packet=raw(maps,len(program))
    if unit_product:
        packet=component.units.rewrite(packet)
        circuit,output=component.units.polynomial_source(packet)
    else:
        circuit,output=component.parent.polynomial_source(packet)
    packet=dict(packet,program=program,phase_count=len(program),unit_product=unit_product,
                adapter_sha256=adapter_hash,polynomial_source=circuit,output=output)
    counts=Counter('M' if op=='*' else 'A' for _,op,_,_ in circuit)
    degree=formal_bound(circuit,packet['parameters']+packet['auxiliaries'])
    packet['polynomial_ledger']=dict(operations=len(circuit),M=counts['M'],A=counts['A'],witnesses=len(packet['auxiliaries']),comparisons=len(packet['comparisons']),degree_upper=degree[output])

def execute(source,values):
    env=dict(values)
    for name,op,a,b in source:
        need(name not in env and op in ('+','-','*'),'SSA operation')
        aa=env[a] if type(a) is str else a;bb=env[b] if type(b) is str else b
        env[name]=aa*bb if op=='*' else aa+bb if op=='+' else aa-bb
    return env

def evaluate(packet,values,*,signed=False,root=None):

def manual_outer(packet,values):
    s=packet['tiles'];m=packet['phase_count'];u=len(packet['groups_U']);v=len(packet['groups_V']);g=u+v
    P0=3*values['x']+values['Z0'];Ufinal=P0*values['Vfinal']+values['x']

    sel=[values[f'Shat{i}']-1 for i in range(s)];J=sum(sel);P=(B-1)*J+1
    zu=[values[f'ZUhat{i}']-1 for i in range(u)];zv=[values[f'ZVhat{i}']-1 for i in range(v)]
    group=[sum(sel[i] for i in entry['tiles']) for entry in packet['groups_U']+packet['groups_V']]
    pack=lambda row:sum(x*P**i for i,x in enumerate(row))
    rep=lambda n:sum(P**i for i in range(n))

    RM=(D-1)*J*(1+P);common=P**g*S+P**(g+s)*T;oldN=g+s+4

    M=Mb+P**g*Mc+P**(g+s)*RM+P**(g+s+2)*(B-1)+P**oldN*(P0-1)
    Z=Zb+common;scale=P**(oldN+1)
    nextU=2*values['H_U']+sum(row[1]*a for row,a in zip(packet['maps'],sel))
    nextV=values['H_V']+sum(entry['difference']*a for entry,a in zip(packet['groups_V'],zv))+sum(row[3]*a for row,a in zip(packet['maps'],sel))
    Q=sum((i//2+1)*a for i,a in enumerate(sel));Next=sum((((i//2-1)%m)+1)*a for i,a in enumerate(sel))
    rr=[values['H_U']+values['H_V']+sum(zu)+sum(zv)+g+values['global_bound']-P,
        B*nextU+1-values['H_U']-P*Ufinal,
        B*nextV+1-values['H_V']-P*values['Vfinal'],
        B*Next+values['phase_initial']-Q-P*m]
    face=dict(P0=P0,Ufinal=Ufinal,D=D,B=B,J=J,P=P,Q=Q,Next=Next,H=H,M=M,Z=Z,scale=scale,
              Hb=Hb,Mb=Mb,Zb=Zb,S=S,Mc=Mc,T=T,RM=RM,Gpack=pack(group),nextU=nextU,nextV=nextV)

def outer_fixture(packet,x,z0,heads,*,root,_component=None):

def enumerate_closures(program,max_t=8):
    for t in range(1,max_t+1):
        for heads in itertools.product((0,1),repeat=t):
            S=1;C=0;bits=0
            for i,d in enumerate(heads):
                C+=d*S;bits+=d*2**i;S*=1+(2*4**program[i%len(program)]-1)*d
            if (1<<t)%S:continue
            P0=(1<<t)//S;z=(1<<t)-P0*C-3*bits
            if z<=0 or (P0-z)%3 or P0<=z:continue
            yield (P0-z)//3,z,heads
