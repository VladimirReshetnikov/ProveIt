def monoid_power_sum(b,n,p):
 # (b^n,1+b+...+b^(n-1)), using concatenation of geometric blocks.
 power,total=1,0; block_power,block_sum=b%p,1
 while n:
  if n&1:total=(total+power*block_sum)%p;power=power*block_power%p
  block_sum=block_sum*(1+block_power)%p;block_power=block_power*block_power%p;n//=2
 return power,total

def constants(rows,p=None):
 out=[]
 for i,r in enumerate(rows):
  op=r[0]
  if op=='int':
   require(len(r)==2 and type(r[1]) is str,'literal type');z=int(r[1]);require(str(z)==r[1],'noncanonical integer')
  elif op in ('pow2','geom4'):
   require(len(r)==2 and type(r[1]) is int and 0<=r[1]<=10**7,'fixed exponent type/range')
   z=(pow(2,r[1],p) if p else 1<<r[1]) if op=='pow2' else (monoid_power_sum(4,r[1],p)[1] if p else ((1<<(2*r[1]))-1)//3)
  else:
   require(op in ('add','sub','mul') and len(r)==3,'recipe opcode')
   require(all(type(j)is int and -i<=j<0 for j in r[1:]),'recipe is not closed in prior fixed numerals')
   if p:
    a,b=(out[-j-1] for j in r[1:]);z=a+b if op=='add' else a-b if op=='sub' else a*b
   else:z=0 # Structural-only mode never expands giant composite numerals.
  out.append(z%p if p else 0)
 return out
