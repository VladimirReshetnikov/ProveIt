Cofinal natural-number tail
  cut=w; heads=[]
  S = w^2
  N = w^2

One omega head over that tail
  cut=w; heads=[w]
  S = w^2
  N = w^2 + 1

Constant omega
  cut=w + 1; heads=[]
  S = w^2
  N = w^2 + w

Constant omega+1
  cut=w + 2; heads=[]
  S = w^2 + w
  N = w^2 + w*2

Infinite remainder: the leading one is absorbed
  cut=w; heads=[w*2 + 3]
  S = w^2 + w + 3
  N = w^2 + w + 3

Cofinal tail below omega^omega
  cut=w^(w); heads=[]
  S = w^(w)
  N = w^(w)

One omega^omega head above that tail
  cut=w^(w); heads=[w^(w)]
  S = w^(w)*2
  N = w^(w)*2

Constant omega^omega
  cut=w^(w) + 1; heads=[]
  S = w^(w + 1)
  N = w^(w + 1)
