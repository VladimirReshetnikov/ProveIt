\date{1 October 2026}

\begin{abstract}
Let $B_{n,k}$ be the number of permutations of $kn$ distinct cards, partitioned
into $n$ ranks of size $k$, in which neighboring cards never have the same
rank.  The OEIS entry A330266~\cite{oeis330266} records the fixed-$k$
conjecture (S.~Kirgizov, 29 September 2023)
\[
  \frac{B_{n,k}}{(kn)!}\longrightarrow \ee^{-(k-1)}.
\]
The case $k=4$ was subsequently deduced on that entry from an approximation
to a related sequence (D.~Radcliffe, 9 September 2025), but the general
statement remained posed there.  We
give a direct proof for every fixed $k\ge2$ and prove considerably more.
If $X_{n,k}$ is the number of equal-rank adjacencies in a uniformly random
shuffle, then
\[
  X_{n,k}\xrightarrow{d}\Pois(k-1).
\]
We derive an exact probability-generating integral whose kernel is an
associated Laguerre polynomial, an exact inclusion--exclusion formula, a
gamma-expectation representation, and a constructive expansion to arbitrary
order.  In particular, uniformly for fixed $k$,
\[
\begin{aligned}
\frac{B_{n,k}}{(kn)!}=\ee^{-(k-1)}\bigg[&
1-\frac{(k-1)^2}{2kn}
+\frac{(k-1)^2(3k^2-14k+7)}{24k^2n^2}\\
&-\frac{(k-1)^4(k^2-10k+17)}{48k^3n^3}
+O_k(n^{-4})\bigg].
\end{aligned}
\]
The first four factorial cumulants are obtained in exact rational form; the
mean is exactly $k-1$ and the variance is
$(k-1)-(k-1)^2/(kn-1)$.  For A330266 this gives
\[
\frac{B_{n,4}}{(4n)!}=\ee^{-3}\left(1-\frac{9}{8n}
-\frac{3}{128n^2}+\frac{189}{1024n^3}+O(n^{-4})\right).
\]
We also treat the corresponding multiset sequences, local probabilities,
Hamiltonian paths in complete multipartite graphs, ratio asymptotics, and a
Lambert-$W$ inverse transseries.  Exact reproducibility code accompanies the
article.
\end{abstract}

\section{The OEIS problem and the family of sequences}

Fix integers $n\ge0$ and $k\ge2$.  There are $n$ ranks, each containing $k$
distinct cards.  A shuffle is a permutation of all $N=kn$ cards.  Let
\begin{equation}\label{bsw:eq:Xdef}
 X_{n,k}=\#\{1\le j<N:\text{the cards at positions $j$ and $j+1$
have the same rank}\}.
\end{equation}
Write
\begin{equation}\label{bsw:eq:defB}
 B_{n,k}=N!\,\PP(X_{n,k}=0),
 \qquad p_{n,k}=\frac{B_{n,k}}{N!}.
\end{equation}
If cards within one rank are made indistinguishable, let $S_{n,k}$ denote the
number of admissible multiset words.  Every rank word lifts to exactly
$(k!)^n$ labeled permutations, so
\begin{equation}\label{bsw:eq:labeled-multiset}
 B_{n,k}=(k!)^n S_{n,k},
 \qquad
 p_{n,k}=\frac{S_{n,k}}{(kn)!/(k!)^n}.
\end{equation}
Thus the same probability governs the labeled and multiset versions.

The most visible OEIS rows are as follows.
\begin{center}
\begin{tabular}{ccll}
\toprule
$k$ & labeled count $B_{n,k}$ & interpretation & multiset count $S_{n,k}$\\
\midrule
2 & A007060~\cite{oeis007060} & couples with no adjacent spouses & A114938~\cite{oeis114938}\\
3 & A193624~\cite{oeis193624} & triples with no adjacent siblings & A193638~\cite{oeis193638}\\
4 & A330266~\cite{oeis330266} & four cards in each rank & A321633~\cite{oeis321633}\\
\bottomrule
\end{tabular}
\end{center}
A330266~\cite{oeis330266} records the exact integral
\[
 B_{n,4}=\int_0^\infty
 (t^4-12t^3+36t^2-24t)^n\ee^{-t}\,dt
\]
and the conjectured fixed-multiplicity law, posed there by S.~Kirgizov on
29 September 2023,
\begin{equation}\label{bsw:eq:oeis-conj}
 p_{n,k}\longrightarrow \ee^{-(k-1)}.
\end{equation}
The entry also records a September 2025 deduction of the $k=4$ equivalent
from a separate approximation: D.~Radcliffe (9 September 2025) combined
$\textnormal{A330266}(n)=24^n\,n!\,\textnormal{A190830}(n)$ with
V.~Kot\v{e}\v{s}ovec's approximation of A190830 and Stirling's formula.  Our derivation below is direct, applies to
every fixed $k$, and identifies the full limiting distribution behind
\eqref{bsw:eq:oeis-conj}.

\subsection{Why a Poisson law is plausible but not automatic}

For each of the $N-1$ adjacent position pairs, the probability of equal ranks
is $(k-1)/(N-1)$.  Hence
\begin{equation}\label{bsw:eq:mean-quick}
 \EE X_{n,k}=(N-1)\frac{k-1}{N-1}=k-1.
\end{equation}
If the rare adjacency indicators were independent, one would expect a
Poisson limit of mean $k-1$ and hence a zero probability $\ee^{-(k-1)}$.
They are not independent: overlapping position pairs and the fixed rank
multiplicities produce both local and global dependence.  The proof therefore
requires more than the heuristic ``rare events with bounded mean.''  The
exact generating function will make the cancellation transparent.

\subsection{Hamiltonian-path interpretation}

Replace the deck by the complete $n$-partite graph with $n$ parts of size
$k$.  A vertex ordering is an admissible shuffle precisely when consecutive
vertices are joined.  Consequently $B_{n,k}$ is the number of \emph{oriented}
Hamiltonian paths, counted as ordered vertex lists, in
$K_{k,k,\ldots,k}$.  Unoriented paths are counted by $B_{n,k}/2$ when
$kn>1$.  This connection places the problem among balanced Smirnov words and
complete multipartite Hamiltonian enumeration~\cite{eriksson-martin,mehiri}.

\section{A weighted run transform}

It is useful to enumerate all shuffles while recording the number of bad
adjacencies.  Define
\begin{equation}\label{bsw:eq:weightedA}
 A_{n,k}(u)=\sum_{\pi\in\mathfrak S_{kn}}u^{X_{n,k}(\pi)},
 \qquad
 G_{n,k}(u)=\frac{A_{n,k}(u)}{(kn)!}=\EE u^{X_{n,k}}.
\end{equation}
Thus $A_{n,k}(0)=B_{n,k}$ and $G_{n,k}(0)=p_{n,k}$.

\begin{theorem}[Run-transform generating function]\label{bsw:thm:run-transform}
For indeterminates $x_1,\ldots,x_n$,
\begin{equation}\label{bsw:eq:runGF}
 \sum_{w}u^{X(w)}x_1^{|w|_1}\cdots x_n^{|w|_n}
 =\frac{1}{1-\displaystyle\sum_{i=1}^n
             \frac{x_i}{1+(1-u)x_i}},
\end{equation}
where the sum is over all finite words on $\{1,\ldots,n\}$, including the
empty word.  Therefore
\begin{equation}\label{bsw:eq:coeffA}
 A_{n,k}(u)=(k!)^n\coef{x_1^k\cdots x_n^k}
 \frac{1}{1-\displaystyle\sum_{i=1}^n
             \frac{x_i}{1+(1-u)x_i}}.
\end{equation}
\end{theorem}

\begin{proof}
Compress every maximal run of equal letters to one letter.  The compressed
word is a Smirnov word, meaning that consecutive letters are distinct.  A run
of letter $i$ and length $r\ge1$ contributes
$u^{r-1}x_i^r$, whose generating series is
\[
 y_i=\sum_{r\ge1}u^{r-1}x_i^r=\frac{x_i}{1-ux_i}.
\]
The classical multivariate generating function of Smirnov
words~\cite{stanley,flajolet-sedgewick} is
\[
 \frac{1}{1-\sum_i y_i/(1+y_i)}.
\]
Indeed, it follows from the ordinary word series
$1/(1-\sum_i z_i)$ by replacing each nonempty run by one marked letter, or
by the standard run-decomposition bijection.  Substitution gives
\[
 \frac{y_i}{1+y_i}=\frac{x_i}{1+(1-u)x_i},
\]
which proves \eqref{bsw:eq:runGF}.  Extracting $k$ occurrences of each rank gives
multiset words; labeling the $k$ copies independently in each rank contributes
$(k!)^n$ and proves \eqref{bsw:eq:coeffA}.
\end{proof}

\section{The exact Laguerre integral}

Set $v=u-1$ and define
\begin{equation}\label{bsw:eq:Pku}
 P_{k,u}(t)=k!\coef{x^k}
 \exp\!\left(\frac{tx}{1-vx}\right).
\end{equation}

\begin{theorem}[Exact integral and polynomial kernel]\label{bsw:thm:integral}
For every $n\ge0$, $k\ge2$, and complex $u$,
\begin{equation}\label{bsw:eq:exact-integral}
 \boxed{\displaystyle
 A_{n,k}(u)=\int_0^\infty \ee^{-t}P_{k,u}(t)^n\,dt.}
\end{equation}
The kernel has the explicit forms
\begin{align}
 P_{k,u}(t)
 &=\sum_{j=1}^k \binom{k-1}{j-1}\frac{k!}{j!}
       t^j(u-1)^{k-j},\label{bsw:eq:Pexplicit}\\
 &=k!(u-1)^k L_k^{(-1)}\!\left(-\frac{t}{u-1}\right),
 \label{bsw:eq:Plaguerre}
\end{align}
where the value at $u=1$ is interpreted by continuity.
\end{theorem}

\begin{proof}
Use the coefficientwise Laplace identity
\[
 \frac1{1-s}=\int_0^\infty \ee^{-t}\ee^{ts}\,dt.
\]
In \eqref{bsw:eq:coeffA}, the exponential factors into $n$ identical univariate
series.  Extracting $x_i^k$ from each factor gives
\eqref{bsw:eq:exact-integral}.  Expanding
$\exp(tx/(1-vx))$ first by the exponential degree and then extracting $x^k$
gives \eqref{bsw:eq:Pexplicit}.  Finally, the standard
expansion~\cite{dlmf}
\[
 L_k^{(-1)}(z)=\sum_{j=1}^k
 \binom{k-1}{j-1}\frac{(-z)^j}{j!}
\]
gives \eqref{bsw:eq:Plaguerre}.
\end{proof}

For no equal adjacencies, $u=0$ and $v=-1$.  Write
\begin{equation}\label{bsw:eq:Pk}
 P_k(t):=P_{k,0}(t)=(-1)^k k!L_k^{(-1)}(t).
\end{equation}
The first kernels are
\begin{align*}
 P_2(t)&=t^2-2t,\\
 P_3(t)&=t^3-6t^2+6t,\\
 P_4(t)&=t^4-12t^3+36t^2-24t,\\
 P_5(t)&=t^5-20t^4+120t^3-240t^2+120t.
\end{align*}
The third line recovers the integral printed on A330266, while the theorem
explains it as one row of a uniform Laguerre family.

\section{Exact inclusion--exclusion and a compact reciprocal polynomial}

For $0\le m\le k-1$, put
\begin{equation}\label{bsw:eq:akm}
 a_{k,m}=\binom{k-1}{m}\frac{k!}{(k-m)!}
        =m!\binom{k}{m}\binom{k-1}{m},
\end{equation}
and define the degree-$(k-1)$ polynomial
\begin{equation}\label{bsw:eq:Rk}
 R_k(z)=\sum_{m=0}^{k-1}a_{k,m}z^m.
\end{equation}
The no-adjacency reciprocal polynomial is
\begin{equation}\label{bsw:eq:Ck}
 C_k(z)=z^kP_k(1/z)=R_k(-z).
\end{equation}

\begin{theorem}[Exact finite formula]\label{bsw:thm:exact-sum}
Let $N=kn$.  Then
\begin{align}
 B_{n,k}
 &=\sum_{m_1,\ldots,m_n=0}^{k-1}
 (-1)^M(N-M)!\prod_{i=1}^n a_{k,m_i},
 \qquad M=m_1+\cdots+m_n,\label{bsw:eq:multi-sum}\\
 &=\sum_{M=0}^{n(k-1)}(N-M)!\coef{z^M}C_k(z)^n.
 \label{bsw:eq:grouped-sum}
\end{align}
More generally, the factorial probability generating function
\begin{equation}\label{bsw:eq:Phi-def}
 \Phi_{n,k}(v):=\EE(1+v)^{X_{n,k}}=G_{n,k}(1+v)
\end{equation}
satisfies the exact identity
\begin{equation}\label{bsw:eq:Phi-exact}
 \boxed{\displaystyle
 \Phi_{n,k}(v)=
 \sum_{M=0}^{n(k-1)}
 \frac{\coef{z^M}R_k(z)^n}{\fall{N}{M}}v^M.}
\end{equation}
\end{theorem}

\begin{proof}
From \eqref{bsw:eq:Pexplicit},
\[
 z^kP_{k,u}(1/z)=\sum_{m=0}^{k-1}a_{k,m}v^mz^m=R_k(vz).
\]
Raise to the $n$th power and integrate termwise in
\eqref{bsw:eq:exact-integral}.  Since
$\int_0^\infty t^{N-M}\ee^{-t}\,dt=(N-M)!$, division by $N!$ yields
\eqref{bsw:eq:Phi-exact}.  Setting $v=-1$ gives
\eqref{bsw:eq:multi-sum} and \eqref{bsw:eq:grouped-sum}.
\end{proof}

Formula \eqref{bsw:eq:grouped-sum} is a rook-polynomial form of
inclusion--exclusion.  It uses only a degree-$(k-1)$ polynomial raised to the
$n$th power and is efficient for exact integer computation.  In particular,
for $k=4$ it reproduces
\[
 B_{13,4}=
3668033946384704437729512814619767610579526911188666362431432294400,
\]
the standard-deck value recorded on A330266.

\section{The Poisson adjacency theorem}

Recall that the $r$th falling factorial is
$\fall{x}{r}=x(x-1)\cdots(x-r+1)$.

\begin{theorem}[Factorial moments]\label{bsw:thm:factorial-moments}
For fixed $k\ge2$ and fixed $r\ge0$,
\begin{equation}\label{bsw:eq:fm-limit}
 \EE\fall{X_{n,k}}{r}=(k-1)^r+O_{k,r}(n^{-1}).
\end{equation}
Consequently,
\begin{equation}\label{bsw:eq:poisson-limit}
 \boxed{\displaystyle
 X_{n,k}\xrightarrow[n\to\infty]{d}\Pois(k-1).}
\end{equation}
\end{theorem}

\begin{proof}
The factorial moment is $r!$ times the coefficient of $v^r$ in
\eqref{bsw:eq:Phi-exact}.  Since every term $a_{k,m}z^m$ in $R_k(z)$ has the same
$z$-degree as the corresponding $v$-degree, the denominator for this
coefficient is exactly $\fall{N}{r}$.  The leading contribution to
$\coef{z^r}R_k(z)^n$ selects $a_{k,1}z$ from $r$ distinct factors and $1$
from the rest:
\[
 \coef{z^r}R_k(z)^n
 =\binom nr a_{k,1}^r+O_{k,r}(n^{r-1}).
\]
Here $a_{k,1}=k(k-1)$.  Therefore
\begin{align*}
 \EE\fall{X_{n,k}}r
 &=\frac{r!\binom nr[k(k-1)]^r+O_{k,r}(n^{r-1})}
         {\fall{kn}{r}}\\
 &=(k-1)^r+O_{k,r}(n^{-1}).
\end{align*}
These are the factorial moments of $\Pois(k-1)$.  Ordinary moments are finite
linear combinations of factorial moments, so every ordinary moment converges.
The Poisson distribution is moment-determinate; the method of moments gives
\eqref{bsw:eq:poisson-limit}.
\end{proof}

\begin{corollary}[The fixed-multiplicity OEIS law]\label{bsw:cor:main-limit}
For every fixed integer $k\ge2$,
\begin{equation}\label{bsw:eq:main-limit}
 \boxed{\displaystyle
 \frac{B_{n,k}}{(kn)!}=p_{n,k}\longrightarrow\ee^{-(k-1)}.}
\end{equation}
Equivalently,
\begin{equation}\label{bsw:eq:multiset-leading}
 S_{n,k}\sim\frac{(kn)!}{(k!)^n}\ee^{-(k-1)}.
\end{equation}
\end{corollary}

\begin{proof}
Because the variables are integer-valued,
\[
 \PP(X_{n,k}=0)=\PP(-\tfrac12<X_{n,k}<\tfrac12).
\]
The endpoints $\pm\tfrac12$ carry no mass under the limiting Poisson law, so
weak convergence implies convergence of this probability.  Hence
$\PP(X_{n,k}=0)\to\PP(\Pois(k-1)=0)=\ee^{-(k-1)}$.
Equation \eqref{bsw:eq:multiset-leading} follows from
\eqref{bsw:eq:labeled-multiset}.
\end{proof}

\begin{remark}
The proof is short because the correct object is not merely the zero-event
count.  The entire adjacency statistic has a stable Poisson limit.  The
constant $k-1$ is already forced by the exact mean \eqref{bsw:eq:mean-quick}.
\end{remark}

\section{A gamma expectation and an all-orders expansion engine}

The exact finite sum admits a useful probabilistic transform.  Let
$T_N$ have the gamma density
\begin{equation}\label{bsw:eq:gamma-density}
 f_N(t)=\frac{t^N\ee^{-t}}{N!},\qquad t>0,
 \qquad T_N\sim\GammaDist(N+1,1).
\end{equation}
Since
\begin{equation}\label{bsw:eq:negative-moment}
 \EE T_N^{-M}=\frac{(N-M)!}{N!}=\frac1{\fall NM}
 \qquad(0\le M\le N),
\end{equation}
we obtain the following exact formula.

\begin{proposition}[Gamma representation]\label{bsw:prop:gamma}
For $N=kn$,
\begin{equation}\label{bsw:eq:gamma-rep}
 \boxed{\displaystyle
 \Phi_{n,k}(v)=\EE\left[R_k\!\left(\frac v{T_N}\right)^n\right].}
\end{equation}
\end{proposition}

The gamma variable is concentrated at scale $\sqrt N$ around $N+1$.  Since
$R_k(w)=1+k(k-1)w+O_k(w^2)$, the exponent in
\eqref{bsw:eq:gamma-rep} tends to $(k-1)v$.  This already supplies another proof
of the Poisson limit.  More importantly, it supports a complete Laplace
expansion.

For a nonnegative integer $M$, define polynomials $H_r(M)$ by
\begin{equation}\label{bsw:eq:Hr-def}
 \prod_{j=0}^{M-1}\frac1{1-jx}=\sum_{r\ge0}H_r(M)x^r.
\end{equation}
Thus $H_r(M)=h_r(0,1,\ldots,M-1)$, a complete homogeneous symmetric
polynomial.  The first four are
\begin{align}
 H_0(M)&=1,\label{bsw:eq:H0}\\
 H_1(M)&=\frac{M(M-1)}2,\label{bsw:eq:H1}\\
 H_2(M)&=\frac{M(M-1)(3M^2+M-2)}{24},\label{bsw:eq:H2}\\
 H_3(M)&=\frac{M^2(M-1)^2(M+1)(M+2)}{48}.
 \label{bsw:eq:H3}
\end{align}
Let $D=v\,d/dv$.

\begin{theorem}[Constructive all-orders expansion]\label{bsw:thm:all-orders}
Fix $k\ge2$.  As $n\to\infty$, uniformly for $v$ in every compact subset of
$\CC$, $\Phi_{n,k}(v)$ has a Poincare expansion in integer powers of $1/n$.
A coefficientwise construction is
\begin{equation}\label{bsw:eq:operator-expansion}
 \boxed{\displaystyle
 \Phi_{n,k}(v)\sim
 \sum_{r\ge0}\frac1{N^r}
 H_r(D)\left[R_k\!\left(\frac vN\right)^n\right],
 \qquad N=kn.}
\end{equation}
After expanding the bracket in $1/N$, collecting total powers gives any
desired correction term.  In particular, every coefficient is an explicit
polynomial in $v$ and a rational function of $k$, multiplied by
$\ee^{(k-1)v}$.
\end{theorem}

\begin{proof}
Write
\[
 R_k(z)^n=\sum_{M=0}^{n(k-1)}r_{n,M}z^M.
\]
Then \eqref{bsw:eq:Phi-exact} is
\[
 \Phi_{n,k}(v)=\sum_M r_{n,M}\frac{v^M}{N^M}
 \frac{N^M}{\fall NM}.
\]
For fixed $M$,
\[
 \frac{N^M}{\fall NM}
 =\prod_{j=0}^{M-1}\left(1-\frac jN\right)^{-1}
 \sim\sum_{r\ge0}\frac{H_r(M)}{N^r}.
\]
Since $D$ acts on $v^M$ by multiplication by $M$, this is precisely the
formal operator rule \eqref{bsw:eq:operator-expansion}.

It remains to justify the asymptotic use when $M$ is not fixed.  The exact
gamma representation \eqref{bsw:eq:gamma-rep} does so.  Split the integral into
$|T_N-N|\le N^{2/3}$ and its complement.  On the upper tail, gamma Chernoff
bounds are enough.  On the lower tail one combines the polynomial bound
$|R_k(v/t)|\le C_{k,V}(1+t^{1-k})$ for $|v|\le V$ with Stirling's formula;
the exact negative moments show that terms whose total polynomial degree is
proportional to $N$ are in fact superexponentially small.  Thus the weighted
complement is smaller than every fixed power of $N^{-1}$.  On the central
interval, $T_N^{-1}$, $\log R_k(v/T_N)$, and the gamma density all possess
Taylor expansions with remainders uniform for $v$ in a prescribed compact
set.
After the change $T_N=N+\sqrt N\,y$, each coefficient is a polynomial in $y$
times a Gaussian majorant, so termwise integration is valid to arbitrary
fixed order.  Odd half-powers cancel in the integrated expansion, leaving
integer powers of $1/N$.  The coefficientwise operator computation and the
Laplace computation agree because both expand the same exact negative moments
\eqref{bsw:eq:negative-moment}.  This proves the Poincare expansion and the stated
uniformity.
\end{proof}

\section{Exact factorial cumulants}

Define factorial cumulants $\kappa_r^{(F)}$ by
\begin{equation}\label{bsw:eq:fact-cumulants}
 \log\Phi_{n,k}(v)=\sum_{r\ge1}\kappa_r^{(F)}\frac{v^r}{r!}.
\end{equation}
They are particularly natural for a Poisson limit: a Poisson variable of mean
$\lambda$ has $\kappa_1^{(F)}=\lambda$ and all higher factorial cumulants zero.

\begin{theorem}[First four factorial cumulants]\label{bsw:thm:cumulants}
Let $N=kn$.  Whenever the displayed denominators are nonzero,
\begin{align}
 \kappa_1^{(F)}&=k-1,\label{bsw:eq:kappa1}\\
 \kappa_2^{(F)}&=-\frac{(k-1)^2}{N-1},\label{bsw:eq:kappa2}\\
 \kappa_3^{(F)}&=\frac{2(k-2)(k-1)^2}{(N-2)(N-1)},\label{bsw:eq:kappa3}\\
 \kappa_4^{(F)}&=-\frac{6(k-1)^2
 \{N(k^2-6k+7)+4k-6\}}
 {(N-3)(N-2)(N-1)^2}.
 \label{bsw:eq:kappa4}
\end{align}
In particular,
\begin{equation}\label{bsw:eq:mean-var}
 \boxed{\displaystyle
 \EE X_{n,k}=k-1,
 \qquad
 \Var X_{n,k}=(k-1)-\frac{(k-1)^2}{kn-1}.}
\end{equation}
\end{theorem}

\begin{proof}
Keep terms through $v^4$ in \eqref{bsw:eq:Phi-exact}.  Since
\[
 R_k(z)=1+a_{k,1}z+a_{k,2}z^2+a_{k,3}z^3+a_{k,4}z^4+O(z^5),
\]
ordinary multinomial expansion gives the first four factorial moments as
rational functions of $n$ and $N$.  Taking the formal logarithm and using
\eqref{bsw:eq:akm} simplifies to \eqref{bsw:eq:kappa1}--\eqref{bsw:eq:kappa4}.  This is
finite polynomial algebra; Appendix~\ref{bsw:app:cumulant-algebra} records a
reproducible symbolic check.  For any integer-valued random variable,
$\EE X=\kappa_1^{(F)}$ and
$\Var X=\kappa_1^{(F)}+\kappa_2^{(F)}$, which gives
\eqref{bsw:eq:mean-var}.
\end{proof}

All higher factorial cumulants tend to zero.  The all-orders theorem gives the
sharper scale $\kappa_r^{(F)}=O_{k,r}(n^{1-r})$ for fixed $r\ge2$, a
mod-Poisson pattern that controls local corrections.
[2 October 2026, batch 77P5: this scale does not follow from the existence
of the expansion alone; it is proved as Theorem~\ref{bsw:rp:thm:cumulant}
of Part~II.]

\section{Explicit expansion through third order}

Put $\lambda=k-1$.  Expanding the exact cumulants and applying
Theorem~\ref{bsw:thm:all-orders} gives the following compact logarithmic form.

\begin{theorem}[Logarithmic factorial-PGF expansion]\label{bsw:thm:log-pgf}
For fixed $k\ge2$, uniformly for $v$ in compact subsets of $\CC$,
\begin{align}
 \log\Phi_{n,k}(v)
={}&\lambda v
-\frac{\lambda^2v^2}{2kn}\notag\\
&+\frac{\lambda^2v^2\{2(k-2)v-3\}}{6k^2n^2}\notag\\
&-\frac{\lambda^2v^2
 \{(k^2-6k+7)v^2-4(k-2)v+2\}}
 {4k^3n^3}
+O_{k,v}(n^{-4}).
\label{bsw:eq:log-pgf-exp}
\end{align}
\end{theorem}

\begin{proof}
Insert \eqref{bsw:eq:kappa2}--\eqref{bsw:eq:kappa4} into
\eqref{bsw:eq:fact-cumulants} and expand their denominators in $1/n$.  These terms
supply every contribution through $n^{-3}$ because the fifth and higher
factorial cumulants begin at $O_k(n^{-4})$, as follows from
Theorem~\ref{bsw:thm:all-orders}.  Collecting powers of $v$ gives
\eqref{bsw:eq:log-pgf-exp}.
[2 October 2026, batch 77P5: the step on the fifth and higher factorial
cumulants is Theorem~\ref{bsw:rp:thm:cumulant} of Part~II, and
\eqref{bsw:rp:eq:q123} there re-derives the three displayed coefficients
from the full operator.]
\end{proof}

Setting $v=-1$ produces the no-adjacency probability.

\begin{corollary}[No-adjacency transseries]\label{bsw:cor:p-expansion}
For fixed $k\ge2$,
\begin{align}
 \log p_{n,k}
={}&-(k-1)
-\frac{(k-1)^2}{2kn}
-\frac{(k-1)^2(2k-1)}{6k^2n^2}
-\frac{(k-1)^4}{4k^3n^3}
+O_k(n^{-4}),
\label{bsw:eq:logp}\\[0.5em]
 p_{n,k}
={}&\ee^{-(k-1)}\left[
 1-\frac{(k-1)^2}{2kn}
 +\frac{(k-1)^2(3k^2-14k+7)}{24k^2n^2}\right.\notag\\
&\hspace{8em}\left.
 -\frac{(k-1)^4(k^2-10k+17)}{48k^3n^3}
 +O_k(n^{-4})\right].
\label{bsw:eq:pfull}
\end{align}
\end{corollary}

The logarithmic form is often numerically superior because exponentiating a
truncated log preserves positivity and partially resums products of lower
orders.
[5 October 2026, batch~101: the $n^{-4}$ terms of \eqref{bsw:eq:logp} and
\eqref{bsw:eq:pfull} are \eqref{bsw:ut:eq:q4minus} and
\eqref{bsw:ut:eq:p4} of Part~III divided by $k^4$

\subsection{A first local correction}

The expansion describes more than the mass at zero.

\begin{corollary}[Fixed local probabilities]\label{bsw:cor:local}
For each fixed $j\ge0$,
\begin{equation}\label{bsw:eq:local}
 \PP(X_{n,k}=j)
 =\ee^{-\lambda}\frac{\lambda^j}{j!}
 \left[1-\frac{(\lambda-j)^2-j}{2kn}+O_{k,j}(n^{-2})\right].
\end{equation}
\end{corollary}

\begin{proof}
Exponentiating the first correction in \eqref{bsw:eq:log-pgf-exp} gives
\[
 \Phi_{n,k}(v)=\ee^{\lambda v}
 \left(1-\frac{\lambda^2v^2}{2kn}+O(n^{-2})\right).
\]
Write $v=u-1$ and extract the coefficient of $u^j$.  Dividing the correction
by the Poisson mass gives
\[
 -\frac{\lambda^2}{2kn}
 \left(1-\frac{2j}{\lambda}+\frac{j(j-1)}{\lambda^2}\right),
\]
which is \eqref{bsw:eq:local}.
\end{proof}

\section{Asymptotics of the counts, ratios, and inverse functions}

\subsection{Logarithmic count transseries}

Let $N=kn$ and $\lambda=k-1$.  Combining Stirling's expansion with
\eqref{bsw:eq:logp} yields
\begin{align}
 \log B_{n,k}
={}&N(\log N-1)+\frac12\log(2\pi N)-\lambda
 +\frac{A_1}{N}+\frac{A_2}{N^2}+\frac{A_3}{N^3}
 +O_k(N^{-4}),
\label{bsw:eq:logB}
\end{align}
where
\begin{equation}\label{bsw:eq:A123}
 A_1=\frac1{12}-\frac{\lambda^2}{2},\qquad
 A_2=-\frac{\lambda^2(2k-1)}6,\qquad
 A_3=-\frac1{360}-\frac{\lambda^4}{4}.
\end{equation}
For multiset words,
\begin{equation}\label{bsw:eq:logS}
 \log S_{n,k}=\log B_{n,k}-n\log(k!).
\end{equation}
These formulas separate the universal factorial growth from the finite
adjacency penalty.

\subsection{Successive-term ratios}

For fixed $k$,
\begin{equation}\label{bsw:eq:ratio}
 \frac{B_{n+1,k}}{B_{n,k}}
 =\prod_{j=1}^k(kn+j)
 \left[1+\frac{(k-1)^2}{2kn^2}
 +\frac{(k-1)^2(k-2)}{6k^2n^3}
 +O_k(n^{-4})\right].
\end{equation}
Indeed, the factorial ratio is exact and subtraction of
$\log p_{n,k}$ from $\log p_{n+1,k}$ gives the bracket.  Such a ratio is
useful for checking recurrences attached to individual OEIS rows.

\section{Relation to rook polynomials and recent multipartite work}

The coefficients $a_{k,m}=m!\binom{k}{m}\binom{k-1}{m}$ are the familiar
rook numbers of a rectangular board of dimensions $k$ by $k-1$.  This
explains why rook-polynomial methods naturally occur in exact deck-shuffle
enumeration.  Enouen's report on the standard deck~\cite{enouen} develops such a direct
rook-polynomial approach.  The reciprocal polynomial $C_k(z)=R_k(-z)$ packages
all ranks into the compact power $C_k(z)^n$.

Balanced Smirnov words have also been connected explicitly with Hamiltonian
paths and cycles in complete multipartite graphs.  Recent work by
Mehiri~\cite{mehiri} gives
bijections and inclusion--exclusion formulas, including nonuniform part sizes.
The present asymptotic regime keeps the part size $k$ fixed and lets the number
of parts $n$ grow.  In this regime, Theorem~\ref{bsw:thm:factorial-moments} says:
\begin{quote}
Among all ordered vertex lists of $K_{k,k,\ldots,k}$, the fraction that form
Hamiltonian paths tends to $\ee^{-(k-1)}$, and the number of forbidden
within-part consecutive steps tends to $\Pois(k-1)$.
\end{quote}
This probabilistic statement complements exact Hamiltonian enumerations and
suggests analogous cyclic and nonuniform limit laws.

\section{Conclusion}

The constant $\ee^{-(k-1)}$ is not an isolated feature of three OEIS rows.  It
is the zero mass of a limiting Poisson law for the complete adjacency
statistic.  The run transform converts the combinatorics into a single
Laguerre polynomial; reciprocal coefficients give exact inclusion--exclusion;
and a gamma negative-moment representation produces systematic correction
terms.  The resulting framework simultaneously proves the fixed-$k$
conjecture recorded on A330266, sharpens the known $k=2,3,4$ asymptotics, and
opens a tractable family of cyclic, nonuniform, and growing-multiplicity
problems.

\section{Exact model and factorial-moment domination}\label{bsw:rp:sec:model}

There are $n\ge1$ ranks, each with $k\ge2$ distinct cards.  Choose a
permutation of the $N=kn$ cards uniformly.  Let $X=X_{n,k}$ be the number
of positions $j<N$ at which cards $j,j+1$ have the same rank.  Equivalently
choose a multiset word uniformly with $k$ copies of each of $n$ letters.
Write
\[
 \lambda=k-1,\qquad d=n(k-1),\qquad
 \Phi_{n,k}(v)=\EE(1+v)^X,\qquad p_{n,k}=\PP(X=0).
\]
The labeled no-adjacency count is $B_{n,k}=N!\,p_{n,k}$; the multiset count
is $S_{n,k}=B_{n,k}/(k!)^n$.  We have $0\le X\le d<N$.  For $k=2,3,4$,
respectively, the labeled counts are OEIS A007060, A193624 and
A330266~\cite{oeis007060,oeis193624,oeis330266}; the corresponding
multiset-word counts are A114938, A193638 and
A321633~\cite{oeis114938,oeis193638,oeis321633}.  The probability is
common to both models, but the counts are different.

With $\fall a0=1$, $a_{k,m}$ and $R_k$ as in
\eqref{bsw:eq:akm}--\eqref{bsw:eq:Rk} (the manuscript writes
$a_{k,m}=\binom{k-1}m\fall km$, the same number), put
\begin{equation}\label{bsw:rp:eq:c}
 r_{n,M}=\coef{z^M}{R_k(z)^n},\qquad
 c_{n,M}=\EE\binom XM=\frac{r_{n,M}}{\fall NM}\quad(0\le M\le d).
\end{equation}
Set $r_{n,M}=c_{n,M}=0$ for $M>d$; do not evaluate the quotient outside its
stated support.  The exact identity
\[
 \Phi_{n,k}(v)=\sum_{M=0}^d\frac{\coef{z^M}{R_k(z)^n}}{\fall NM}v^M
\]
is \eqref{bsw:eq:Phi-exact} of Theorem~\ref{bsw:thm:exact-sum}, which
Part~I obtains by weighted run transforms.

\begin{proof}[Second proof of \eqref{bsw:eq:Phi-exact} (manuscript 11)]
For completeness, there is a direct contraction proof.  Canonically number
the occurrences of each letter from left to right.  The $k-1$ possible bonds
for a letter join consecutive occurrences; a bond is present when those
occurrences are adjacent.  Select $m_i$ bonds for letter $i$ and contract
them.  The contracted words have $k-m_i$ copies of letter $i$, and expansion
according to the selected components is inverse to contraction.  If
$M=\sum_i m_i$, the number of such words is $(N-M)!/\prod_i(k-m_i)!$.
Double counting words with $M$ selected present bonds and dividing by
$N!/(k!)^n$ gives \eqref{bsw:rp:eq:c}.
\end{proof}

The Laguerre and rook-polynomial machinery is older: Gessel's
linear-permutation construction~\cite{gessel} (its Section~6 treats linear
permutations), Taylor's Theorem~2.1 and run generating
function~\cite{taylor} (his Theorem~2.1 credits Gessel's exact Carlitz-word
formula; his Lemma~2.2 gives the classical generating function), and
Enouen's standard-deck application~\cite{enouen} (its Sections~3--4 give
the rook-polynomial solution and the exact standard-deck count) give
relevant precedents.  Eriksson and Martin~\cite{eriksson-martin} enumerate
the same Carlitz multipermutation family.  This Part repairs a specified
proof, rather than claiming priority for those formulas.

\begin{theorem}[Exact Poisson factorial domination]\label{bsw:rp:thm:bound}
For all $n\ge1$ and integers $M\ge0$,
\begin{equation}\label{bsw:rp:eq:bound}
 0\le c_{n,M}\le\frac{\lambda^M}{M!},
 \qquad\text{equivalently}\qquad \EE\fall XM\le\lambda^M.
\end{equation}
Consequently $|\Phi_{n,k}(v)|\le\ee^{\lambda|v|}$ for every complex $v$.
\end{theorem}

\begin{proof}
Since $\fall km\le k^m$, nonnegative coefficients give
\[
 R_k(z)\preceq(1+kz)^{k-1},\qquad
 r_{n,M}\le k^M\binom dM.
\]
Here $\preceq$ means coefficientwise inequality.  For $0\le M\le d$,
\[
 c_{n,M}\le\frac{k^M}{M!}\prod_{j=0}^{M-1}\frac{d-j}{N-j}
 \le\frac{k^M}{M!}\left(\frac dN\right)^M
 =\frac{\lambda^M}{M!},
\]
since $(d-j)/(N-j)\le d/N$.  Empty products include $M=0$, and the zero
convention includes $M>d$.  Sum the absolute coefficients to obtain the
last assertion.
\end{proof}

\begin{remark}
This is factorial-moment domination; it is not a claim of stochastic

\section{The explicit uniform operator remainder}\label{bsw:rp:sec:operator}

For a finite list of nonnegative variables let $h_s$ denote the complete
homogeneous symmetric polynomial.  Set $h_0=1$; for the empty list set
$h_s=0$ when $s>0$.  For integers $M,r\ge0$ define
\[
 H_r(M)=h_r(0,1,\ldots,M-1),\qquad
 Q_M(x)=\prod_{j=0}^{M-1}(1-jx)^{-1},
\]
in agreement with \eqref{bsw:eq:Hr-def}.  The list for $M=0$ is empty.
Both $M=0$ and $M=1$ give $Q_M=1$ and $H_r(M)=0$ for $r\ge1$.

\begin{lemma}[Complete-homogeneous remainder]\label{bsw:rp:lem:h}
For nonnegative integers $a,b$, $h_{a+b}\le h_a h_b$ on nonnegative
variables.  Consequently, for an integer $J\ge0$ and $0\le x<1/(M-1)$ when
$M\ge2$,
\begin{equation}\label{bsw:rp:eq:hrem}
 0\le Q_M(x)-\sum_{r=0}^JH_r(M)x^r
 \le x^{J+1}H_{J+1}(M)Q_M(x).
\end{equation}
The inequality holds without restriction on $x\ge0$ for $M=0,1$.  For
every integer $s\ge1$,
\begin{equation}\label{bsw:rp:eq:hbound}
 H_s(M)\le\left(\frac{M(M-1)}2\right)^s\le M^{2s}.
\end{equation}
\end{lemma}

\begin{proof}
A term of $h_{a+b}$ is indexed by a nondecreasing string of $a+b$ indices.
Splitting after $a$ entries injects its terms, with unchanged weights, into
the pairs of strings counted by $h_a h_b$.  Thus
$H_{J+1+t}(M)\le H_{J+1}(M)H_t(M)$.  The convergent expansion
$Q_M(x)=\sum_{r\ge0}H_r(M)x^r$ now gives \eqref{bsw:rp:eq:hrem} by summing
over $t\ge0$.  Finally, the multinomial expansion of the sum of the
variables raised to $s$ counts each weak monomial with coefficient at least
one; this proves \eqref{bsw:rp:eq:hbound}.
\end{proof}

Put $F_N(v)=R_k(v/N)^n$.  The symbol $H_r(D)$ denotes a polynomial in the
Euler differential operator $D=v\,d/dv$.  Indeed, if
\[
 \sigma_j(M)=\sum_{i=0}^{M-1}i^j,
\]
then
\begin{equation}\label{bsw:rp:eq:formalh}
 \sum_{r\ge0}H_r(M)x^r
 =\exp\left(\sum_{j\ge1}\frac{x^j}{j}\sigma_j(M)\right).
\end{equation}
Faulhaber's formula gives $\deg\sigma_j=j+1$; coefficient extraction shows
that $H_r(M)$ is a polynomial in $M$ of degree at most $2r$, with the
stated values also at $M=0$.

\begin{theorem}[Quantitative compact-complex remainder]\label{bsw:rp:thm:operator}
For every integer $J\ge0$ and real $V\ge0$, for all $n\ge1$,
\begin{equation}\label{bsw:rp:eq:operr}
 \sup_{|v|\le V}\left|\Phi_{n,k}(v)-\sum_{r=0}^J N^{-r}H_r(D)F_N(v)\right|
 \le \frac{\mathcal C_{k,V,J}}{N^{J+1}},
\end{equation}
where, with $a=\lambda V$,
\begin{equation}\label{bsw:rp:eq:constant}
 \mathcal C_{k,V,J}=\sum_{M=0}^{\infty}H_{J+1}(M)\frac{a^M}{M!}
 \le \ee^a\mathcal T_{2J+2}(a)<\infty.
\end{equation}
Here $\mathcal T_j(a)=\ee^{-a}\sum_{M\ge0}M^j a^M/M!$ is the $j$th
Touchard polynomial.  For $V=0$ the constant is zero.
\end{theorem}

\begin{proof}
The exact finite identity \eqref{bsw:eq:Phi-exact} reads
\[
 \Phi_{n,k}(v)=\sum_{M=0}^d r_{n,M}(v/N)^M Q_M(1/N).
\]
Every $M$ in this sum satisfies $M\le d<N$, so Lemma~\ref{bsw:rp:lem:h}
applies.  Since $D(v^M)=Mv^M$, subtract the finite operator sum, take
absolute values, and use nonnegativity to obtain
\begin{align*}
 \text{error}
 &\le N^{-J-1}\sum_{M=0}^d H_{J+1}(M)r_{n,M}(V/N)^M Q_M(1/N)\\
 &=N^{-J-1}\sum_{M=0}^d H_{J+1}(M)c_{n,M}V^M\\
 &\le N^{-J-1}\mathcal C_{k,V,J}.
\end{align*}
The last step uses Theorem~\ref{bsw:rp:thm:bound}.  The bound
\eqref{bsw:rp:eq:hbound} proves finiteness and the Touchard upper bound.
In particular there is no need to control $Q_M(1/N)$ uniformly in $M$: its
entire value converts the majorant back into an exact factorial-moment
coefficient.
\end{proof}

\section{Integer powers and the normalized logarithm}\label{bsw:rp:sec:integer}

Choose the analytic germ
\[
 \log R_k(z)=\sum_{j\ge1}b_jz^j,\qquad \log R_k(0)=0,
 \qquad b_1=k\lambda.
\]
For $w$ near zero set
\begin{equation}\label{bsw:rp:eq:Fwv}
 F(w,v)=\exp\left(\frac{\log R_k(wv)}{kw}\right),
 \qquad F(0,v)=\ee^{\lambda v}.
\end{equation}
On each fixed $v$-disk the singularity at $w=0$ is removable and the
function is jointly analytic for sufficiently small $|w|$.  At $w=1/N$ it
equals $F_N(v)$ exactly, since $N/k=n$ is integral.  Its expansion is
\begin{align}
 F(w,v)&=\ee^{\lambda v}\sum_{s\ge0}f_s(v)w^s,\qquad f_0=1,
 \label{bsw:rp:eq:f}\\
 f_s(v)&=\sum_{\substack{m_1,\ldots,m_s\ge0\\\sum j m_j=s}}
 \prod_{j=1}^s\frac{\bigl((b_{j+1}/k)v^{j+1}\bigr)^{m_j}}{m_j!}.
 \label{bsw:rp:eq:fs}
\end{align}
Thus $f_s$ is a polynomial of degree at most $2s$.  Analyticity on a
slightly larger $v$-disk and Cauchy's estimates justify differentiating each
fixed-order Taylor remainder finitely many times, uniformly on $|v|\le V$.

\begin{corollary}[All-orders expansion]\label{bsw:rp:cor:all}
For each fixed integer $J\ge0$,
\begin{equation}\label{bsw:rp:eq:all}
 \Phi_{n,k}(v)=\ee^{\lambda v}\sum_{s=0}^J \mathcal P_s(v)N^{-s}
 +O_{k,V,J}(N^{-J-1}),\qquad |v|\le V,
\end{equation}
where $\mathcal P_0=1$ and
\begin{equation}\label{bsw:rp:eq:Ps}
 \mathcal P_s(v)=\ee^{-\lambda v}\sum_{r=0}^s
 H_r(D)\bigl(\ee^{\lambda v}f_{s-r}(v)\bigr).
\end{equation}
Each $\mathcal P_s$ is a polynomial in $v$, with rational-in-$k$
coefficients.
\end{corollary}

\begin{proof}
In Theorem~\ref{bsw:rp:thm:operator}, expand the $r$th differentiated
bracket through $w^{J-r}$.  Its multiplied remainder is $O(w^{J+1})$,
uniformly on the disk.  Summing finitely many such remainders gives
\eqref{bsw:rp:eq:all}.  All powers are integer powers from analytic Taylor
series; no half-power cancellation is assumed.
\end{proof}

On a fixed disk, $\ee^{-\lambda v}\Phi_{n,k}(v)=1+O(N^{-1})$.  Hence for
sufficiently large $n$ it lies within distance $1/2$ of $1$, on a
neighborhood of that disk.  Define
\begin{equation}\label{bsw:rp:eq:logbranch}
 \log\Phi_{n,k}(v):=\lambda v+\log\bigl(\ee^{-\lambda v}\Phi_{n,k}(v)\bigr),
\end{equation}
where the second logarithm is the analytic branch near $1$.  This branch has
value zero at $v=0$.  It has a uniform integer-power expansion.  An
unspecified principal logarithm on a large complex disk would not give the
same statement.

\section{The all-orders factorial-cumulant degree argument}\label{bsw:rp:sec:cumulant}

The next argument is separate from existence of the uniform expansion.  It
proves the degree cancellations needed to control all factorial cumulants.

Work formally in $w$; all operator expressions have finitely many
contributions to any fixed coefficient.  By \eqref{bsw:rp:eq:formalh},
\begin{equation}\label{bsw:rp:eq:Tw}
 \mathcal H_w:=\sum_{r\ge0}H_r(D)w^r=\exp\mathcal A_w(D),
 \qquad \mathcal A_w(D)=\sum_{j\ge1}\frac{w^j}{j}\sigma_j(D).
\end{equation}
This may be checked on each monomial $v^M$ using the scalar identity for
$Q_M(w)$.  No convergence of the infinite operator series at nonzero $w$ is
asserted.

Introduce a formal flow parameter $t$ and write
\[
 \ee^{t\mathcal A_w(D)}F(w,v)=\ee^{\Sigma(t,w,v)},\qquad
 \Sigma(t,w,v)=\lambda v+\sum_{s\ge1}\Sigma_s(t,v)w^s.
\]
After removal of $\ee^{\lambda v}$, this is a formal $w$-series with
polynomial coefficients in $v,t$ and constant term $1$; its formal logarithm
is uniquely defined.  Initially,
\[
 \Sigma_s(0,v)=\frac{b_{s+1}}k v^{s+1}.
\]

\begin{theorem}[Logarithmic degree and factorial cumulants]\label{bsw:rp:thm:cumulant}
The expansion of the normalized logarithm \eqref{bsw:rp:eq:logbranch} has
the form
\begin{equation}\label{bsw:rp:eq:q}
 \log\Phi_{n,k}(v)=\lambda v+\sum_{s=1}^J q_s(v)N^{-s}
 +O_{k,V,J}(N^{-J-1}),\qquad \deg_v q_s\le s+1.
\end{equation}
Consequently, for each fixed $r\ge2$,
\[
 \kappa_r^{(F)}:=\left.\frac{d^r}{dv^r}\log\Phi_{n,k}(v)\right|_{v=0}
 =O_{k,r}(N^{1-r})=O_{k,r}(n^{1-r}).
\]
Also $\kappa_1^{(F)}=\EE X=\lambda$ exactly.
\end{theorem}

\begin{proof}
The formal flow satisfies
\[
 \partial_t\Sigma=\ee^{-\Sigma}\mathcal A_w(D)\ee^{\Sigma},
\]
where the conjugated operator on the right acts on $1$.  Put
$\mathcal B_a=\ee^{-\Sigma}D^a\ee^{\Sigma}$.  Then $\mathcal B_0=1$ and
\[
 \mathcal B_{a+1}=D\mathcal B_a+(D\Sigma)\mathcal B_a.
\]
Thus each monomial of $\mathcal B_a$ is a product of at most $a$ factors
$D^\ell\Sigma$, $\ell\ge1$, apart from the constant for $a=0$.

We induct on $s$ to prove $\deg_v\Sigma_s(t,v)\le s+1$.  To extract $w^s$
from the $j$th summand of the generator, $1\le j\le s$, the orders $s_i$ of
these factors sum to $s-j$.  Each is strictly less than $s$.  Because
$\sigma_j$ has degree at most $j+1$, there are at most $b\le j+1$ factors.
The Euler derivative does not increase polynomial degree.  The induction
hypotheses, including $\Sigma_0=\lambda v$, give total degree at most
\[
 \sum_{i=1}^b(s_i+1)=s-j+b\le s+1.
\]
The zero-factor case causes no problem.  Therefore $\partial_t\Sigma_s$ has
this degree bound, and the initial value does too.  Integration in $t$
proves the induction.  At $t=1$, the coefficients are those of
$\mathcal H_wF$, which Corollary~\ref{bsw:rp:cor:all} has already
identified as the analytic asymptotic expansion of $\Phi_{n,k}$.  Finite
logarithmic-series algebra identifies $q_s=\Sigma_s(1,v)$.

For a fixed $r\ge2$, take $J=r-2$ in \eqref{bsw:rp:eq:q} on a disk about
zero.  All retained polynomials have degree less than $r$.  Cauchy's
coefficient formula applied to the uniform remainder gives the claimed
bound.  The exact mean follows from \eqref{bsw:rp:eq:c} with $M=1$.
\end{proof}

\begin{remark}\label{bsw:rp:rem:factorial}
These are \emph{factorial} cumulants, using derivatives in $v$ of
$\log\EE(1+v)^X$.  Ordinary cumulants use $\log\EE\ee^{tX}$ and converge to
the Poisson value $\lambda$; they do not have the displayed decay.
\end{remark}

\subsection{Carefully scoped Lambert inverse}\label{bsw:rp:sec:inverse}

This subsection inverts the labeled count $B_{n,k}$, not the multiset count

 L=\log Y+\lambda,\qquad N_0=\frac{L}{W_0(L/\ee)}.
\]

 +O_k\left(\frac1{N_0\log N_0}\right),\qquad n=N/k.

 (N-N_0)\log N_0+g(N_0)=O(1/N_0),

\section{Direct verification of the weighted gamma tail}\label{bsw:rp:app:gamma}

This appendix is unnecessary for Theorem~\ref{bsw:rp:thm:operator}, but
verifies the tail assertion in the original approach of
Theorem~\ref{bsw:thm:all-orders}.  Let $T_N\sim\GammaDist(N+1,1)$, with
density $t^N\ee^{-t}/N!$.  The exact representation
\eqref{bsw:eq:gamma-rep} is
\[
 \Phi_{n,k}(v)=\EE R_k(v/T_N)^n.
\]
For $E_N=\{t:|t-N|>N^{2/3}\}$, positivity and finite expansion give
\begin{equation}\label{bsw:rp:eq:mixture}
 \EE\bigl[|R_k(v/T_N)|^n\mathbf1_{E_N}(T_N)\bigr]
 \le\sum_{M=0}^d c_{n,M}V^M
 \PP\{\GammaDist(N-M+1,1)\in E_N\},\quad |v|\le V.
\end{equation}
Multiplying the original density by $t^{-M}$ changes its shape to $N-M+1$,
with normalizing factor $(N-M)!/N!$, which proves
\eqref{bsw:rp:eq:mixture}.  All shapes are positive.

Split at $M_N=\lfloor N^{1/3}\rfloor$.  For $M\le M_N$, the new mean
$q=N-M+1$ is within $M_N+1$ of $N$.  For all large $N$, membership in $E_N$
implies deviation from $q$ by more than $N^{2/3}/2$.  The gamma
moment-generating function $(1-s)^{-q}$ and Chernoff optimization give
\[
 \PP(T-q\ge a)\le\ee^{-a^2/[2(q+a)]},\qquad
 \PP(T-q\le-a)\le\ee^{-a^2/(2q)}\quad(0<a<q).
\]
For the upper bound optimize at $s=a/(q+a)$; for the lower, at
$s=-a/(q-a)$.  The resulting rate functions dominate the stated quadratic
expressions by integrating their derivatives.  Thus the $M\le M_N$
contribution in \eqref{bsw:rp:eq:mixture} is at most
$2\ee^{\lambda V-cN^{1/3}}$ for an absolute $c>0$ and sufficiently large
$N$.

The other terms total at most $\sum_{M>M_N}(\lambda V)^M/M!$.  For fixed
$a=\lambda V>0$, once $M_N+2\ge2a$ this is bounded by
\[
 \frac{2a^{M_N+1}}{(M_N+1)!}
 \le 2\left(\frac{\ee a}{M_N+1}\right)^{M_N+1}
 =\exp\{-\Omega(N^{1/3}\log N)\}.
\]
For $V=0$ it vanishes.  The entire weighted complement is therefore smaller
than every power of $N^{-1}$, uniformly on the disk.  The same factorial
bound for degrees $M\ge\varepsilon N$, fixed $\varepsilon>0$, gives
$\exp\{-\varepsilon N\log N+O(N)\}$ (or zero if beyond support).  No
uncontrolled prefactor $C^n$ is introduced.

\section{The exact identity by marked directed path forests}
\label{bsw:ut:sec:forest}

Part~I proves \eqref{bsw:eq:Phi-exact} by a weighted run transform and the
Laguerre integral, and Part~II adds a contraction proof on multiset words.
The manuscript counts labelled cards rank by rank instead.

\begin{proof}[Third proof of \eqref{bsw:eq:Phi-exact} (Report~98)]
Mark $M$ equal-rank adjacencies of a permutation.  Within each
rank they form a directed union of vertex-disjoint paths on its $k$
labelled cards: a card has at most one marked successor and at most one
marked predecessor, and the marked edges follow the left-to-right order.
If the rank has $j$ marked edges, the forest has $k-j$ path components,
isolated cards included.  The directed path forests with $j$ edges on $k$
labelled cards number
\[
 \frac{k!}{(k-j)!}\binom{k-1}{j}=a_{k,j}:
\]
arrange the $k$ labels in a row, keep $j$ of the $k-1$ intervening bonds,
and divide by the $(k-j)!$ orders of the resulting components.  Choose such
a forest for every rank, with $M$ edges in all, and contract each path to a
block.  The $N-M$ blocks can be arranged in $(N-M)!$ orders, and each
arrangement is a permutation in which the marked edges are equal-rank
adjacencies.  Further, unmarked, equal-rank adjacencies are allowed.  This
is a bijection onto pairs of a permutation and an $M$-element subset of its
equal-rank adjacencies.  Hence
$\sum_\pi\binom{X(\pi)}M=r_{n,M}(N-M)!$, which is
\eqref{bsw:eq:Phi-exact} after division by $N!$.  In particular the
support degree is $n(k-1)=N-n<N$.
\end{proof}

[Write: $a_{k,j}=\frac{k!}{(k-j)!}\binom{k-1}j$ is the Lah number
$L(k,k-j)$, the number of partitions of $k$ labelled cards into $k-j$
nonempty linearly ordered blocks; Section~11 of Part~I identifies the same
numbers as rook numbers of a $k\times(k-1)$ board.]

The count is local to each rank, so it applies verbatim to ranks of
different sizes.  The manuscript states this in its third further question;
the write states and proves it.

\begin{proposition}[Unequal rank sizes: the exact identity; Report~98,
proof written out at the write]\label{bsw:ut:prop:unequal-exact}
Let $\mathbf k=(k_1,\ldots,k_n)$ with integers $k_i\ge1$, put
$N=k_1+\cdots+k_n$, and let $X_{\mathbf k}$ count the equal-rank
adjacencies in a uniform permutation of $N$ distinct cards, $k_i$ of them
of rank $i$.  Define $a_{k,j}$ and $R_k$ by
\eqref{bsw:eq:akm}--\eqref{bsw:eq:Rk} also for $k=1$, so that $R_1=1$, and
put $r_{\mathbf k,M}=\coef{z^M}{\prod_{i=1}^nR_{k_i}(z)}$.  Then, for
$0\le M\le N-n$,
\begin{equation}\label{bsw:ut:eq:unequal-exact}
 \EE\binom{X_{\mathbf k}}M=\frac{r_{\mathbf k,M}}{\fall NM},
\end{equation}
and both sides vanish for $M>N-n$.  In particular
$\EE X_{\mathbf k}=\sum_ik_i(k_i-1)/N$.
\end{proposition}

\begin{proof}
Rank $i$ carries at most $k_i-1$ marked edges.  Repeat the third proof
above with $a_{k_i,j_i}$ directed path forests of $j_i$ edges for rank $i$
(for $k_i=1$ only $j_i=0$ occurs, with one forest).  Choosing forests with
$j_1+\cdots+j_n=M$ edges in all, contracting and arranging the $N-M$
blocks gives $\sum_\pi\binom{X_{\mathbf k}(\pi)}M
=(N-M)!\sum_{j_1+\cdots+j_n=M}\prod_ia_{k_i,j_i}
=(N-M)!\,r_{\mathbf k,M}$, the bijection being the same.  Divide by $N!$;
the quotient is defined because $M\le N-n<N$.  For $M>N-n=\sum_i(k_i-1)$
no choice of forests exists, and no permutation has more than $N-n$
equal-rank adjacencies.  For $M=1$, $r_{\mathbf k,1}=\sum_ia_{k_i,1}
=\sum_ik_i(k_i-1)$.
\end{proof}

\section{The factorial-moment majorant}\label{bsw:ut:sec:majorant}

Theorem~\ref{bsw:rp:thm:bound} bounds $c_{n,M}$ through the majorant
$(1+kz)^{k-1}$ and the inequality $(d-j)/(N-j)\le d/N$.  The manuscript's
Lemma~1 uses the majorant $(1+\lambda z)^k$, which needs no second step.

\begin{proof}[Second proof of Theorem~\ref{bsw:rp:thm:bound} (Report~98)]
For $0\le j\le k-1$ the $j$ factors of $\fall{k-1}j$ lie in $[0,\lambda]$,
so $a_{k,j}=\binom kj\fall{k-1}j\le\binom kj\lambda^j$.  Hence
\[
 R_k(z)\preceq(1+\lambda z)^k,\qquad R_k(z)^n\preceq(1+\lambda z)^N,
\]
since products of polynomials with nonnegative coefficients preserve
$\preceq$.  For $M\le d=n(k-1)$, dividing $r_{n,M}\le\binom NM\lambda^M$
by $\fall NM=M!\binom NM$ gives $c_{n,M}\le\lambda^M/M!$; beyond the
support $c_{n,M}=0$.
\end{proof}

\begin{corollary}[Tail bound; Report~98]\label{bsw:ut:cor:tail}
For $V\ge0$ and every integer $M_0\ge0$,
\begin{equation}\label{bsw:ut:eq:tail}
 \sup_{|v|\le V}\Bigl|\sum_{M\ge M_0}c_{n,M}v^M\Bigr|
 \le\sum_{M\ge M_0}\frac{(\lambda V)^M}{M!}
 \le\ee^{\lambda V}\frac{(\lambda V)^{M_0}}{M_0!}.
\end{equation}
\end{corollary}

\begin{proof}
Sum the bound of Theorem~\ref{bsw:rp:thm:bound} and use
$(M_0+j)!\ge M_0!\,j!$.
\end{proof}

The manuscript notes that this majorant accounts for the growing number of
rank cliques directly, and does not invoke the fixed-path-family hypotheses
of the path-forest report \texttt{a189281-\allowbreak path-\allowbreak forest-\allowbreak expansions}.  The
inequality holds for every $k$; only the asymptotic constants below depend
on the fixed $k$.  The same majorant works rank by rank.

\begin{proposition}[Unequal rank sizes: factorial-moment bound;
Report~98]\label{bsw:ut:prop:unequal}
In the setting of Proposition~\ref{bsw:ut:prop:unequal-exact}, let
$k_{\max}=\max_ik_i$.  For every integer $M\ge0$,
\begin{equation}\label{bsw:ut:eq:unequal}
 0\le\EE\binom{X_{\mathbf k}}M\le\frac{(k_{\max}-1)^M}{M!},
 \qquad\EE\fall{X_{\mathbf k}}M\le(k_{\max}-1)^M,
\end{equation}
and $|\EE(1+v)^{X_{\mathbf k}}|\le\ee^{(k_{\max}-1)|v|}$ for complex $v$.
\end{proposition}

\begin{proof}
For $0\le j\le k_i-1$ the $j$ factors of $\fall{k_i-1}j$ lie in
$[0,k_{\max}-1]$, so $a_{k_i,j}\le\binom{k_i}j(k_{\max}-1)^j$ and
$R_{k_i}(z)\preceq(1+(k_{\max}-1)z)^{k_i}$.  The product over $i$ gives
$\prod_iR_{k_i}(z)\preceq(1+(k_{\max}-1)z)^N$, so
$r_{\mathbf k,M}\le\binom NM(k_{\max}-1)^M$.  Divide by
$\fall NM=M!\binom NM$ for $M\le N-n$, using
\eqref{bsw:ut:eq:unequal-exact}; for $M>N-n$ the left side is zero.  The last
assertion follows by summing absolute values.  (If $k_{\max}=1$ then
$X_{\mathbf k}=0$, and the bound reads $0^M$ with $0^0=1$.)
\end{proof}

The manuscript does not claim that this bound alone gives a limit law or an
expansion for varying compositions.  The exact identity does give the limit
law when the multiplicities stay bounded.

\begin{proposition}[Bounded multiplicities: Poisson limit; write]
\label{bsw:ut:prop:unequal-poisson}
[Write, 5 October 2026.]  Fix $K\ge1$.  For each $n$ let
$\mathbf k=(k_1,\ldots,k_n)$ with $1\le k_i\le K$, let $N=\sum_ik_i$, and
put $\mu_n=\EE X_{\mathbf k}=\sum_ik_i(k_i-1)/N$.  For every fixed
$M\ge0$,
\[
 \EE\fall{X_{\mathbf k}}M=\mu_n^M+O_{K,M}(n^{-1}).
\]
Consequently, if $\mu_n\to\mu$ as $n\to\infty$, then
$X_{\mathbf k}\to\Pois(\mu)$ in distribution, and
$\PP(X_{\mathbf k}=0)\to\ee^{-\mu}$.
\end{proposition}

\begin{proof}
Put $x_i=a_{k_i,1}=k_i(k_i-1)\le K^2$, so $\sum_ix_i=\mu_nN$.  In
$r_{\mathbf k,M}=\sum_{j_1+\cdots+j_n=M}\prod_ia_{k_i,j_i}$, the terms with
every $j_i\le1$ sum to the elementary symmetric polynomial
$e_M(x_1,\ldots,x_n)$.  In every other term some $j_i\ge2$, so at most
$M-1$ indices have $j_i\ge1$.  There are at most $Mn^{M-1}$ such index sets
and at most $2^{M-1}$ ways to distribute $M$ among a given set, and each
product is at most $(2^KK^M)^M$, because
$a_{k,j}\le\binom kj(k-1)^j\le2^KK^j$ for $k\le K$.  Hence these terms sum
to $O_{K,M}(n^{M-1})$.  Next, $(\sum_ix_i)^M$ is the sum of
$x_{i_1}\cdots x_{i_M}$ over all $M$-tuples of indices; the tuples with
distinct entries contribute $M!\,e_M(x)$, and the at most
$\binom M2n^{M-1}$ tuples with a repeated entry contribute at most
$\binom M2n^{M-1}K^{2M}$.  Therefore
\[
 r_{\mathbf k,M}=\frac{(\mu_nN)^M}{M!}+O_{K,M}(n^{M-1}).
\]
If $N<2M$, then $n\le N<2M$ is bounded, and both
$\EE\fall{X_{\mathbf k}}M$ and $\mu_n^M$ lie in $[0,(K-1)^M]$ by
Proposition~\ref{bsw:ut:prop:unequal}, so the claim holds trivially.  Let
$N\ge2M$, so that $\fall NM\ge(N/2)^M$ and $N^M/\fall NM=1+O_M(N^{-1})$;
recall $n\le N$ and $\mu_n\le K-1$.  If $M\le N-n$,
Proposition~\ref{bsw:ut:prop:unequal-exact} gives
\[
 \EE\fall{X_{\mathbf k}}M=\frac{M!\,r_{\mathbf k,M}}{\fall NM}
 =\mu_n^M\frac{N^M}{\fall NM}
 +O_{K,M}\Bigl(\frac{n^{M-1}}{N^M}\Bigr)
 =\mu_n^M+O_{K,M}(n^{-1}).
\]
If $M>N-n$, then $\EE\fall{X_{\mathbf k}}M=0$ and $r_{\mathbf k,M}=0$, and
the display above gives $(\mu_nN)^M=O_{K,M}(n^{M-1})$, that is
$\mu_n^M=O_{K,M}(n^{M-1}/N^M)=O_{K,M}(n^{-1})$.
If $\mu_n\to\mu$, every factorial moment converges to $\mu^M$, the $M$th
factorial moment of $\Pois(\mu)$ (for $\mu=0$, of the point mass at $0$).
These laws are determined by their moments, so the method of moments gives
the convergence in distribution, exactly as in the proof of
Theorem~\ref{bsw:thm:factorial-moments}; the mass at $0$ converges because
$\pm\frac12$ carry no limiting mass.
\end{proof}

\begin{remark}[What this does for Part~I's question~4]
\label{bsw:ut:rem:unequal-scope}

\section{The operator remainder and a sharper constant}
\label{bsw:ut:sec:operator}

The manuscript's Lemma~2 is Lemma~\ref{bsw:rp:lem:h} at $x=1/N$, where
$Q_M(1/N)=N^M/\fall NM$, proved the same way: splitting a weakly
increasing index string after $a$ entries injects the terms of $h_{a+b}$
into those of $h_ah_b$.  Its Theorem~3 is
Theorem~\ref{bsw:rp:thm:operator}, again with the same proof: multiplying
the scalar remainder by $r_{n,M}|v|^M/N^M$ turns the factor $Q_M(1/N)$ back
into $c_{n,M}$, and Theorem~\ref{bsw:rp:thm:bound} then controls the whole
support at once.  In the manuscript's words, the estimate holds for the
full degree range, and no growing cutoff, exchange of an uncontrolled
infinite sum, gamma concentration estimate, or cancellation of half-integer
powers is involved.  The two differ only in the constant.

\begin{remark}[A sharper constant; Report~98]\label{bsw:ut:rem:constant}
With $a=\lambda V$, the constant of \eqref{bsw:rp:eq:constant} satisfies
\begin{equation}\label{bsw:ut:eq:C}
 \mathcal C_{k,V,J}=\sum_{M\ge0}H_{J+1}(M)\frac{a^M}{M!}
 \le2^{-(J+1)}\ee^a\mathcal T_{2J+2}(a).
\end{equation}
Indeed $h_{a+b}\le h_ah_b$, iterated, gives
$H_{J+1}(M)\le H_1(M)^{J+1}=(M(M-1)/2)^{J+1}\le M^{2J+2}/2^{J+1}$, and
$\sum_MM^{2J+2}a^M/M!=\ee^a\mathcal T_{2J+2}(a)$.  Part~II's
\eqref{bsw:rp:eq:hbound} records the middle bound and then discards the
factor $2^{-s}$, which costs the factor $2^{J+1}$ in
\eqref{bsw:rp:eq:constant}.  The refinement is elementary and changes no
asymptotic statement.
\end{remark}

\begin{remark}[Unequal rank sizes; write]\label{bsw:ut:rem:operator-unequal}
The proof of Theorem~\ref{bsw:rp:thm:operator} uses only the
identity $c_{n,M}=r_{n,M}/\fall NM$ on a support below $N$ and the bound
$c_{n,M}\le\lambda^M/M!$.  By
Propositions~\ref{bsw:ut:prop:unequal-exact}
and~\ref{bsw:ut:prop:unequal} both hold for unequal rank sizes with
$\lambda$ replaced by $k_{\max}-1$.  Hence, with
$F_{\mathbf k}(v)=\prod_iR_{k_i}(v/N)$ and $a=(k_{\max}-1)V$, the same
argument gives, for every $J\ge0$, $V\ge0$ and composition $\mathbf k$,
\[
 \sup_{|v|\le V}\Bigl|\EE(1+v)^{X_{\mathbf k}}
 -\sum_{r=0}^JN^{-r}H_r(D)F_{\mathbf k}(v)\Bigr|
 \le\frac1{N^{J+1}}\sum_{M\ge0}H_{J+1}(M)\frac{a^M}{M!},
\]
and the right side is at most $2^{-(J+1)}\ee^a\mathcal T_{2J+2}(a)/N^{J+1}$
by Remark~\ref{bsw:ut:rem:constant}.  Turning this operator rule into a collected expansion is
part of question~\ref{bsw:ut:q:unequal}.
\end{remark}
