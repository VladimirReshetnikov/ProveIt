import Direct80OffsetExact

/-! Algebraic extraction of the transport divisor from integral carrier pieces. -/
set_option autoImplicit false
set_option Elab.async false
noncomputable section
namespace TenPositiveTransportIntegral
variable {R : Type*} [CommRing R] [Nontrivial R] [NoZeroDivisors R]

theorem transport_divisor (h m H L B k e z n Z kappa rho : R)
    (hm : m≠0) (hH : H≠0)
    (hZ : Z=h*m*H*z) (hK : k=H*kappa) (he : e=m*rho)
    (hcar : Z=h*m*(H*(n-L*B)+k)-H*(m*B+h*e)) : h∣B := by
  have hprod : (m*H)*(B-h*(n-L*B+kappa-rho-z))=0 := by
    rw [hZ,hK,he] at hcar
    linear_combination hcar
  have hzero : B-h*(n-L*B+kappa-rho-z)=0 := (mul_eq_zero.mp hprod).resolve_left (mul_ne_zero hm hH)
  exact ⟨n-L*B+kappa-rho-z,sub_eq_zero.mp hzero⟩

#print axioms TenPositiveTransportIntegral.transport_divisor
end TenPositiveTransportIntegral
