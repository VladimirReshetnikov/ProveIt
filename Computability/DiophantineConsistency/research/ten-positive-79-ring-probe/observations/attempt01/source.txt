theorem old_core (p : Fin 4 → ℤ) (v : Fin 10 → R) :
    ((graph p).output).evalRing (oldVal p v) =
      core p v (v 2) (((p 0:R)*v 0+((1:ℤ):R))*(((p 0:R)*v 0+((1:ℤ):R))-v 1)-v 2) := by
  change core p v (((p 0:R)*v 0+((1:ℤ):R))-v 1-(((p 0:R)*v 0+((1:ℤ):R))-v 1-v 2))
      ((p 0:R)*v 0*(((p 0:R)*v 0+((1:ℤ):R))-v 1)+(((p 0:R)*v 0+((1:ℤ):R))-v 1-v 2)) = _
  congr 1 <;> ring

theorem new_core (p : Fin 4 → ℤ) (v : Fin 10 → R) :

theorem coordinate_identity (p : Fin 4 → ℤ) (v : Fin 10 → R) :
