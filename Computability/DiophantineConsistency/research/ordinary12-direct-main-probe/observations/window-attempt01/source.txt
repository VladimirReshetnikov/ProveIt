theorem graph_window (p : Fin 4 → ℕ) (v : Fin 12 → V)
    (hb : 0<(p 0:V)) (hl : (p 2:V)<(p 0:V)) (hh : (p 3:V)<2*(p 0:V))
    (hv : ∀j,0<v j) (hz : InternalZero (graph (DirectMainFirst.integerPorts p)).output v) :
    let b : V := v 0*(p 0:V)
    let q : V := 1+b
    let R : V := (v 2+(q-v 1)*b)*(q*q-1)+v 0*((p 2:V)+(p 3:V)*q)
    24≤R ∧ R+q^3+q+1≤q^4 := by
  let b : V := v 0*(p 0:V)
  let q : V := 1+b
  let R : V := (v 2+(q-v 1)*b)*(q*q-1)+v 0*((p 2:V)+(p 3:V)*q)
  have hgap : v 1+v 2<q := tsub_pos_iff_lt.mp (DirectMainTransport.transport p v hb hv hz).2
  have hq : q=(p 0:V)*v 0+1 := by dsimp [q,b]; ring
  have hw := scalar_rank_window q (p 0:V) (p 2:V) (p 3:V) (v 0) (v 1) (v 2)
    hb (hv 0) (hv 1) (hv 2) hl hh hq hgap
  have hqb : q-1=b := by dsimp [q]; rw [add_comm]; simp
  have hR : ((q-1)*(q-v 1)+v 2)*(q^2-1)+((p 2:V)+q*(p 3:V))*v 0 = R := by
    rw [hqb]
    dsimp [R]
    ring
  rw [hR] at hw
  exact hw

theorem graph_early (p : Fin 4 → ℕ) (v : Fin 12 → V)
    (hb : 0<(p 0:V)) (hl : (p 2:V)<(p 0:V)) (hh : (p 3:V)<2*(p 0:V))
    (hv : ∀j,0<v j) (hz : InternalZero (graph (DirectMainFirst.integerPorts p)).output v) :
    let b : V := v 0*(p 0:V)
    let q : V := 1+b
    let X : V := q*v 5
    let Y : V := (q*(q*q))*v 6
    let k : V := v 8+v 9
    let c : V := v 8+Y*k
    let a : V := X*Y+Y
    let Q : V := (X*Y)*Y
    let R : V := (v 2+(q-v 1)*b)*(q*q-1)+v 0*((p 2:V)+(p 3:V)*q)
    ∃ first main n m : V, 0<first ∧ 0<main ∧ 13≤n ∧ 0<m ∧ n≤k ∧ m≤c ∧
      xcoord n (2*(Q-1)+1)=first ∧ 2*ycoord n (2*(Q-1)+1)=k ∧
      xcoord m a=main ∧ ycoord m a=c ∧ (2*n)%(q^4)=R+1 ∧ R+1≤2*n ∧ R+1≤k := by
  exact DirectMainEarly.early_of_window p v hb hv hz (graph_window p v hb hl hh hv hz)
