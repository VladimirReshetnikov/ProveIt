import Mathlib.Algebra.Order.Ring.Basic
import Mathlib.Tactic.Linarith
import Mathlib.Tactic.Positivity
import Mathlib.Tactic.Ring

/-! Ordered-ring consequences of the index-three/index-two eta identity. -/
set_option autoImplicit false
noncomputable section
namespace Trace75ThreeGap
variable {R : Type*} [LinearOrderedCommRing R]

theorem gap_bounds (x y eta : R)
    (hx : 2≤x) (hy : 8≤y) (heta : 0<eta)
    (heupper : eta<8*x*y^2+4)
    (heq : eta+4*y^2*(x*(2*y-x-2)-1)=4*y*(2*x+1)+3) :
    x<2*y ∧ 2*y<x+3 := by
  constructor
  · by_contra hn
    have hge : 2*y≤x := le_of_not_gt hn
    have hmul : 0≤x*(x-2*y) := mul_nonneg (by linarith) (by linarith)
    have hbig : 0≤4*y^2*(x*(x-2*y)) := mul_nonneg (by positivity) hmul
    have hxy : 0≤y*x := mul_nonneg (by linarith) (by linarith)
    nlinarith [hbig,hxy]
  · by_contra hn
    have hge : x+3≤2*y := le_of_not_gt hn
    have hmul : 0≤(y-8)*(x-1) := mul_nonneg (by linarith) (by linarith)
    have hbase : 2≤y*(x-1)-2*(x+1) := by nlinarith [hmul]
    have hgap : 0≤x*(2*y-x-3) := mul_nonneg (by linarith) (by linarith)
    have hterm : 0≤4*y^2*(x*(2*y-x-3)) := mul_nonneg (by positivity) hgap
    have hbaseprod : 0≤4*y*(y*(x-1)-2*(x+1)-2) :=
      mul_nonneg (by positivity) (by linarith)
    nlinarith [hterm,hbaseprod]

#print axioms Trace75ThreeGap.gap_bounds
end Trace75ThreeGap
