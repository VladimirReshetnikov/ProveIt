import Trace75MainIndex
import Trace75ThreeExcluded

/-! The trace75 odd main index is at least five for beta at least two. -/
set_option autoImplicit false
set_option Elab.async false
noncomputable section
namespace Trace75MainFive
open LO LO.FirstOrder LO.FirstOrder.Arithmetic
open scoped LO.FirstOrder LO.FirstOrder.Arithmetic
open DiophantineConsistency ModelDifferenceRing TenPositiveTrace75 ArithmeticPellRecurrence
open Trace75Factors Trace75Pell Trace75MainIndex Trace75ThreeExcluded
variable {V : Type*} [ORingStructure V] [V↓[ℒₒᵣ] ⊧* 𝗜𝚺₁]

theorem graph_main_index_ge_five (p : Fin 4 → ℕ) (v : Fin 10 → V)
    (hb2 : 2≤(p 0:V)) (hv : ∀j,0<v j)
    (hz : value75 (ports p) (fun j => ι (v j)) = 0) :
    ∃ r j I : V, 5≤r ∧ r<j ∧ r%2=1 ∧ 0<j ∧ 0<I ∧
      c p v=ycoord r (A p v-1) ∧ 2*r<c p v ∧
      xcoord j (A p v-1)=1+2*(delta p v*U p v*U p v*E p v) ∧
      ycoord j (A p v-1)=I ∧ c p v*c p v∣I := by
  have hb : 0<(p 0:V) := lt_of_lt_of_le (by simp : (0:V)<2) hb2
  obtain ⟨r,j,I,hr3,hrj,hodd,hj,hI,hc,hrsize,hjx,hjy,hdiv⟩ := graph_main_index p v hb hv hz
  obtain ⟨tau,f,n,I0,_,_,_,_,_,_,_,hfirst,_⟩ := first_and_doubled p v hb hv hz
  have hne3 : r≠3 := by
    intro hr
    apply main_three_impossible p v n hb2 hv hz hfirst
    simpa only [hr] using hc
  have hrgt3 : 3<r := lt_of_le_of_ne hr3 (Ne.symm hne3)
  have hr4 : 4≤r := by simpa only [show (3:V)+1=4 by ring] using lt_iff_succ_le.mp hrgt3
  have hrgt4 : 4<r := by
    rcases lt_or_eq_of_le hr4 with h|h
    · exact h
    · rw [←h] at hodd
      simp at hodd
  have hr5 : 5≤r := by simpa only [show (4:V)+1=5 by ring] using lt_iff_succ_le.mp hrgt4
  exact ⟨r,j,I,hr5,hrj,hodd,hj,hI,hc,hrsize,hjx,hjy,hdiv⟩

#print axioms Trace75MainFive.graph_main_index_ge_five
end Trace75MainFive
