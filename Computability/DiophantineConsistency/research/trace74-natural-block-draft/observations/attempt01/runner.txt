#!/usr/bin/env python3
"""Held direct-pin check; release only after independent prelaunch review."""
import argparse
import fcntl
import hashlib
import json
import os
import re
import subprocess
import sys
import tempfile
import time
from pathlib import Path

if not sys.flags.isolated or not sys.flags.no_site or sys.flags.optimize:
    raise SystemExit('use python3 -I -S -B')
H = Path(__file__).resolve().parent
BASE = Path('/tmp/trace74-direct-four-block-1zltaw2f/lib')
PACK = Path('/tmp/ipra-foundation-guarded-v2-cjHUaJun/package-lib')
LEAN = Path('/home/codex/.elan/toolchains/leanprover--lean4---v4.32.0/bin/lean')
PINS = {
    H/'Trace74NaturalBlock.lean': 'b3f2e1f92ff8686d5f2743252fd19bdda095a87f7d0722e068f91ba758b4aed2',
    BASE/'Trace74DirectFourBlock.olean': '6e977f68e25e4bbc290d4c6b43f67f429696a500a071cee7b196c955985eb003',
    BASE/'DiophantineConsistency/ModelDifferenceRing.olean': 'd0fb8f9664675190dbfc25915a687cef203b8362681d9256606f1022a6dbde74',
    LEAN: 'e8baaa71855a616dc351028f3ad2200051b0671f423a1696a100e809302d5550',
    LEAN.parent.parent/'lib/lean/Init.olean': 'd2e85d72099f9d869c9582aad5fb13079968d9d121ac6061e8326b00800b306e',
}

def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def need(ok, message):
    if not ok:
        raise RuntimeError(message)

def verify():
    for p, digest in PINS.items():
        need(sha(p) == digest, 'changed pin: ' + str(p))

def resources():
    ps = subprocess.run(['ps', '-eo', 'pid=,comm=,rss=,args='],
                        capture_output=True, text=True, check=True).stdout.splitlines()
    live = [line.strip() for line in ps if len(line.split(None, 3)) == 4
            and line.split(None, 3)[1] == 'lean']
    available = int(next(line for line in Path('/proc/meminfo').read_text().splitlines()
                         if line.startswith('MemAvailable:')).split()[1])
    return {'time_ns': time.time_ns(), 'mem_available_kib': available, 'lean': live}

def check_report(out):
    pattern = r"\s*'Trace74NaturalBlock\.natural_block' (?:depends on axioms: \[([^\]]*)\]|(does not depend on any axioms))\s*"
    m = re.fullmatch(pattern, out, re.S)
    if not m:
        return False
    if m.group(1) is None:
        return True
    parts = [x.strip() for x in m.group(1).split(',')]
    return bool(parts) and all(parts) and len(parts) == len(set(parts)) and set(parts) <= {'propext', 'Classical.choice', 'Quot.sound'}

parser = argparse.ArgumentParser()
parser.add_argument('--preflight-only', action='store_true')
parser.add_argument('--attempt')
args = parser.parse_args()
need(args.preflight_only != bool(args.attempt), 'choose preflight-only or attempt')
if args.attempt:
    need(re.fullmatch(r'attempt[0-9]{2,}', args.attempt) is not None, 'bad attempt')
verify()
pre = resources()
need(pre['mem_available_kib'] >= 8*1024**2 and not pre['lean'], 'resource preflight')
runner_sha = sha(__file__)
if args.preflight_only:
    print(json.dumps({'status': 'PASS', 'pins': {str(p): h for p, h in PINS.items()},
                      'runner_sha256': runner_sha, 'resources': pre}, sort_keys=True, indent=2))
    sys.exit(0)

lock = os.open('/tmp/direct-bfs-80-coherent-rebuild.lock', os.O_RDWR|os.O_CREAT|os.O_NOFOLLOW, 0o600)
fcntl.flock(lock, fcntl.LOCK_EX|fcntl.LOCK_NB)
verify()
pre = resources()
need(pre['mem_available_kib'] >= 8*1024**2 and not pre['lean'], 'locked resource preflight')
obs = H/'observations'/args.attempt
obs.mkdir(parents=True, exist_ok=False)
root = Path(tempfile.mkdtemp(prefix='trace74-natural-block-'))
obj = root/'Trace74NaturalBlock.olean'
env = os.environ.copy()
for key in ('LD_PRELOAD', 'LD_LIBRARY_PATH', 'LD_AUDIT'):
    env.pop(key, None)
env['LEAN_PATH'] = str(BASE)+':'+str(PACK)
cmd = [str(LEAN), '-M3000', '-j1', '-DElab.async=false', '--root='+str(H),
       str(H/'Trace74NaturalBlock.lean'), '-o', str(obj)]
samples, reason = [], None
start = time.monotonic()
with (obs/'stdout').open('x') as out, (obs/'stderr').open('x') as err:
    proc = subprocess.Popen(cmd, env=env, stdout=out, stderr=err)
    try:
        while proc.poll() is None:
            time.sleep(1)
            s = resources()
            samples.append(s)
            if s['mem_available_kib'] < 8*1024**2 or len(s['lean']) > 1:
                reason = 'resource guard'
                proc.kill()
                break
            if time.monotonic()-start > 120:
                reason = 'timeout'
                proc.kill()
                break
    finally:
        if proc.poll() is None:
            proc.kill()
        proc.wait()
out = (obs/'stdout').read_text()
err = (obs/'stderr').read_text()
verify()
need(sha(__file__) == runner_sha, 'runner changed')
passed = (proc.returncode == 0 and reason is None and obj.is_file() and
          check_report(out) and not err and not re.search(r'(?i)\b(?:error|warning)(?:\([^:\n]*\))?:|\bsorryAx\b', out))
result = {'status': 'PASS' if passed else 'FAIL', 'command': cmd,
          'lean_path': env['LEAN_PATH'], 'source_and_direct_pins': {str(p): h for p,h in PINS.items()},
          'runner_sha256': runner_sha, 'exit': proc.returncode, 'reason': reason,
          'preflight': pre, 'postflight': resources(), 'samples': samples,
          'elapsed': time.monotonic()-start, 'stdout_sha256': sha(obs/'stdout'),
          'stderr_sha256': sha(obs/'stderr'), 'object_path': str(obj) if obj.exists() else None,
          'object_sha256': sha(obj) if obj.exists() else None,
          'scope': 'Conditional natural block equation only; direct project pins, inherited package/transitive/dynamic ancestry and search-path limits. No graph-zero implication or guard.'}
(obs/'terminal.json').write_text(json.dumps(result, sort_keys=True, indent=2)+'\n')
print(json.dumps(result, sort_keys=True, indent=2))
sys.exit(0 if passed else 1)
