import Direct80RankWindow
import Direct80RankSize

/-! The natural rank is smaller than half the main ordinate after transport. -/
set_option autoImplicit false
set_option Elab.async false
noncomputable section
namespace Direct80RankSmall
open LO LO.FirstOrder LO.FirstOrder.Arithmetic
open scoped LO.FirstOrder LO.FirstOrder.Arithmetic
open DiophantineConsistency AdaptiveEquationSyntax2011
open TenPositive80Graph TenPositive80Factors
variable {V : Type*} [ORingStructure V] [V↓[ℒₒᵣ] ⊧* 𝗜𝚺₁]

theorem graph_rank_small (p : Fin 4 → ℕ) (v : Fin 10 → V)
    (hb : 0<(p 0:V)) (hl : (p 2:V)<(p 0:V)) (hh : (p 3:V)<2*(p 0:V))
    (hv : ∀j,0<v j) (hz : InternalZero (graph (integerPorts p)).output v)
    (hgap : v 1+v 2<1+v 0*(p 0:V)) :
    let q0 : V := 1+v 0*(p 0:V)
    24≤Direct80RankWindow.rank p v ∧ Direct80RankWindow.rank p v<q0^4 ∧
      2*Direct80RankWindow.rank p v<c p v := by
  let q0 : V := 1+v 0*(p 0:V)
  let R : V := Direct80RankWindow.rank p v
  have hw := Direct80RankWindow.window p v hb hl hh hv hgap
  have hr : R<q0^4 := calc
    R<R+q0^3+q0+1 := lt_of_le_of_lt (by simp) (lt_add_one (R+q0^3+q0))
    _≤q0^4 := hw.2
  have hq : q0=q p v := by dsimp [q0,q,b]; ring
  have hc : 2*q0^4<c p v := by rw [hq]; exact (Direct80RankSize.graph_rank_size p v hb hv hz).2
  have h2 : 2*R<c p v := lt_of_lt_of_lt (mul_lt_mul_of_pos_left hr (by norm_num)) hc
  exact ⟨hw.1,hr,h2⟩

#print axioms Direct80RankSmall.graph_rank_small
end Direct80RankSmall
