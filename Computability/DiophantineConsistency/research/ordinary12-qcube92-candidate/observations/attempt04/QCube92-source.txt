variable {R : Type*} [CommRing R]

def qcube (p : Fin 4 → ℤ) (v : Fin 12 → R) : R :=
  let q := 1 + v 0 * (p 0 : R)
  q * (q * q)

def lift (p : Fin 4 → ℤ) (v : Fin 12 → R) : Fin 12 → R :=
  Function.update v 6 (qcube p v * v 6)

/-- Generic factored identity; the seven operands can themselves be large terms. -/
theorem scaled_first (b q3 ell k r bt s : R) :
    ((q3*b*((ell+k)+r-k*bt)+(q3*s)*b-q3*bt) *
      (q3*b*((ell+k)+r-k*bt)+(q3*s)*b-q3*bt) -
      ((q3*b)*(q3*b))*(1+ell*(ell+k))) =
    (q3*q3)*((b*((ell+k)+r-k*bt+s)-bt)*
      (b*((ell+k)+r-k*bt+s)-bt) - (b*b)*(1+ell*(ell+k))) := by
  ring

theorem first_eval (p : Fin 4 → ℤ) (v : Fin 12 → R) :
    (FreeJSumSquares.firstResidual p).evalRing (lift p v) =
      (qcube p v * qcube p v) * ((graph p).first.evalRing v) := by
  let b := v 0 * (p 0 : R)
  let q := 1+b
  let q3 := q*(q*q)
  let Y := q3 * v 6
  let k := v 8+v 9
  let ell := (v 5*Y)*(Y*k)
  let bt := ((q-v 1)-v 2)*(v 5+(p 1 : R)*q) - v 1*q
  simp only [FreeJSumSquares.firstResidual, FreeJSumSquares.rightAdd,
    Ordinary12FreeJ.core, graph, Expr.evalRing_add, Expr.evalRing_sub,
    Expr.evalRing_mul, Expr.evalRing_var, Expr.evalRing_constant,
    Expr.evalRing_ofNat, Int.cast_one, lift_0, lift_1, lift_2, lift_3, lift_4, lift_5, lift_6, lift_7, lift_8, lift_9, lift_10, lift_11, qcube]
  change ((q3*b*((ell+k)+v 7-k*bt)+(q3*v 6)*b-q3*bt) *
      (q3*b*((ell+k)+v 7-k*bt)+(q3*v 6)*b-q3*bt) -
      ((q3*b)*(q3*b))*(1+ell*(ell+k))) =
    (q3*q3)*((b*((ell+k)+v 7-k*bt+v 6)-bt)*
      (b*((ell+k)+v 7-k*bt+v 6)-bt) - (b*b)*(1+ell*(ell+k)))
  exact scaled_first b q3 ell k (v 7) bt (v 6)
