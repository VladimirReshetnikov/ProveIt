def need(v,m):
    if not v:raise RuntimeError(m)
def sha(p):return sha256(Path(p).read_bytes()).hexdigest()
def rec(p):
    p=Path(p)
    if not os.path.lexists(p):return None
    need(p.is_file(),'non-file '+str(p))
    return {'sha256':sha(p),'resolved':str(p.resolve(strict=True)), 'link':os.readlink(p) if p.is_symlink() else None}
def resource():
    lines=subprocess.run(['ps','-eo','pid=,comm=,rss=,args='],capture_output=True,text=True,check=True).stdout.splitlines();ps=[]
    for line in lines:
        f=line.strip().split(None,3)
        if len(f)==4 and f[1]=='lean':ps.append({'pid':int(f[0]),'rss_kib':int(f[2]),'args':f[3]})
    m=next(x for x in Path('/proc/meminfo').read_text().splitlines() if x.startswith('MemAvailable:'))
    return {'time_ns':time.time_ns(),'mem_available_kib':int(m.split()[1]),'lean':ps}
def basecheck(base):
    actual={p.relative_to(BASELIB).as_posix():rec(p) for p in BASELIB.rglob('*') if p.is_file()}
    need(actual==base['union'] and len(actual)==1288,'base union drift')
    return actual

        receipt=H/'observations'/attemptname/'terminal.json';d=json.loads(receipt.read_text())

        obj=Path(d['output_root'])/(name+'.olean');source=H/(name+'.lean')

        need(d['before'][str(source)]==sha(source),'producer source drift '+name)

            try: rel=Path(path).relative_to(PROVIDER).as_posix()
            except ValueError: rel=Path(path).name
            if rel in base['union']:
                expected=base['union'][rel]['sha256']
            else:
                producers={n+'.olean':a for n,a in extra}
                need(rel in producers,'provider object missing in candidate union '+rel)
                previous=json.loads((H/'observations'/producers[rel]/'terminal.json').read_text())
                expected=previous['object_sha256']
            need(value['sha256']==expected,'provider overlap mismatch '+rel)
        result[name]={'receipt':str(receipt),'receipt_sha256':sha(receipt),'object':str(obj),'object_sha256':sha(obj),'source_sha256':sha(source)}
    # Verify the fresh divisibility producer's two actual project imports against the union.
    for name in ('DiophantineConsistency/Ordinary12FreeJ.olean','DiophantineConsistency/ArithmeticAdaptiveFirstDivisibility2011.olean'):
        need(name in base['union'] and sha(PROVIDER/name)==base['union'][name]['sha256'],'divisibility import mismatch '+name)
    return result

need(len(union)==1292,'wrong union count')

    names=[]

        if line.startswith('import '):

                path=name.replace('.','/')+'.olean';need(path in union,'missing declared import '+path);names.append({'module':name,'sha256':union[path]['sha256']})
    imports[source.name]=names
src=H/'QCube92Halt.lean';copy=out/src.name;shutil.copyfile(src,copy);shutil.copyfile(src,OBS/'source.txt');sourcehash=sha(src); runnerhash=sha(H/'run-halt.py')
