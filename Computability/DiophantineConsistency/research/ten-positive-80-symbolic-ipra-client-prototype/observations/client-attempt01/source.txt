theorem positive_bridge : PositiveModelBridge (graph CompactSymbolicModelIff.p).output := by
  intro V _ _
  let p := ports betaNat (paramKernel rho) lowNat highNat
  have h0 : p 0=betaNat := rfl
  have h1 : p 1=paramKernel rho := rfl
  have h2 : p 2=lowNat := rfl
  have h3 : p 3=betaNat+highNat := rfl
  have h := symbolic_iff_of_four_ports (V:=V) p h0 h1 h2 h3
  change (∃ v : Fin 10 → V, (∀j,0<v j) ∧
    InternalZero (graph CompactSymbolicModelIff.p).output v) ↔ _ at h
  exact h

  standard_equivalence_of_positive _ positive_bridge
