variable {V : Type*} [ORingStructure V] [V↓[ℒₒᵣ] ⊧* 𝗜𝚺₁]

theorem graph_positive_rho (p : Fin 4 → ℕ) (v : Fin 10 → V)
    (hb : 0<(p 0:V)) (hl : (p 2:V)<(p 0:V)) (hh : (p 3:V)<2*(p 0:V))
    (hv : ∀j,0<v j) (hz : InternalZero (graph (integerPorts p)).output v)
    (hbeta : ∃u : V, (p 0:V)=2*u+1) :
    ∃ rho : V, 0<rho ∧ (eExpr (integerPorts p)).evalRing (fun j => ι (v j))=ι ((mu p v*mu p v)*rho) := by
  obtain ⟨_n,s,_hn,_halign,hk,he⟩ := graph_offset_exact p v hb hl hh hv hz hbeta
  have hklo := (graph_rank_size p v hb hv hz).1
  have hwindow := (graph_rank_exact p v hb hl hh hv hz).2.1
  have hRplus : rankNat p v+1≤(q p v)^4 := by
    calc
      rankNat p v+1≤rankNat p v+(q p v)^3+q p v+1 := by
        simpa only [add_comm] using _root_.add_le_add_right
          (show rankNat p v≤rankNat p v+(q p v)^3+q p v by simp) (1:V)
      _≤(q p v)^4 := hwindow
  have hq : 0<q p v := add_pos (mul_pos hb (hv 0)) (by simp)
  have hY : 0<Y p v := mul_pos (mul_pos (mul_pos hq hq) hq) (hv 6)
  have hX : 0<X p v := mul_pos (hv 5) hq
  have hmu : 0<mu p v := add_pos (mul_pos hq hq) (by simp)
  have hS : q p v*q p v≤mu p v*Y p v := calc
    q p v*q p v≤mu p v := by dsimp [mu]; simp
    _≤mu p v*Y p v := le_mul_of_one_le_right (by simp) (pos_iff_one_le.mp hY)
  have hP : (q p v)^4≤firstParam p v := calc
    (q p v)^4=(q p v*q p v)*(q p v*q p v) := by ring
    _≤(mu p v*Y p v)*(mu p v*Y p v) := mul_le_mul hS hS (by simp) (by simp)
    _≤X p v*((mu p v*Y p v)*(mu p v*Y p v)) :=
      le_mul_of_one_le_left (by simp) (pos_iff_one_le.mp hX)
    _=firstParam p v := rfl
  have hPpos : 0<firstParam p v := mul_pos hX (mul_pos (mul_pos hmu hY) (mul_pos hmu hY))
  have hPk : firstParam p v<k v := calc
    firstParam p v≤8*firstParam p v := le_mul_of_one_le_left (by simp) (by norm_num)
    _<8*firstParam p v+4 := lt_add_of_pos_right _ (by norm_num)
    _≤k v := hklo
  have hRk : rankNat p v+1<k v := lt_of_le_of_lt (le_trans hRplus hP) hPk
  have hs : 0<s := by
    apply pos_iff_ne_zero.mpr
    intro hs0
    rw [hs0,mul_zero,add_zero] at hk
    exact (ne_of_lt hRk) hk.symm
  refine ⟨4*X p v*(Y p v*Y p v)*s,?_,?_⟩
  · exact mul_pos (mul_pos (mul_pos (by simp) hX) (mul_pos hY hY)) hs
  · rw [he,←map_mul]
    congr 1
    dsimp [firstParam]
    ring

#print axioms TenPositive80PositiveOffset.graph_positive_rho
end TenPositive80PositiveOffset
