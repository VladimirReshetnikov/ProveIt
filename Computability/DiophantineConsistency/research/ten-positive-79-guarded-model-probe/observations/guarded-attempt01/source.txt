variable {V : Type*} [ORingStructure V] [V↓[ℒₒᵣ] ⊧* 𝗜𝚺₁]

def flip (p : Fin 4 → ℕ) (v : Fin 10 → V) : Fin 10 → V :=
  ![v 0,v 1,q p v-v 1-v 2,v 3,v 4,v 5,v 6,v 7,v 8,v 9]

theorem flip_positive (p : Fin 4 → ℕ) (v : Fin 10 → V)
    (hv : ∀j,0<v j) (hmask : v 1+v 2<q p v) : ∀j,0<flip p v j := by
  have hf : v 1≤q p v := le_trans (by simp) (le_of_lt hmask)
  have ha : v 2<q p v-v 1 := (lt_tsub_iff_right hf).mpr (by simpa [add_comm] using hmask)
  have hp : 0<q p v-v 1-v 2 := tsub_pos_of_lt ha
  intro j
  fin_cases j <;> simp [flip,hv,hp]

theorem oldVal_cast (p : Fin 4 → ℕ) (v : Fin 10 → V)
    (hmask : v 1+v 2<q p v) :
    oldVal (integerPorts p) (fun j => ι (v j)) = fun j => ι (flip p v j) := by
  have hf : v 1≤q p v := le_trans (by simp) (le_of_lt hmask)
  have ha : v 2≤q p v-v 1 := le_of_lt ((lt_tsub_iff_right hf).mpr (by simpa [add_comm] using hmask))
  have hsub {x y : V} (hy : y≤x) : (ι (x-y):DiffRing V)=ι x-ι y := by
    apply eq_sub_iff_add_eq.mpr
    rw [←map_add,tsub_add_cancel_of_le hy]
  have hcast : (ι (q p v-v 1-v 2):DiffRing V)=ι (q p v)-ι (v 1)-ι (v 2) := by
    rw [hsub ha,hsub hf]
  funext j
  fin_cases j <;> simp [oldVal,flip,hcast,q,b,integerPorts,map_add,map_mul,map_one,map_natCast]

theorem guarded_to_old (p : Fin 4 → ℕ) (v : Fin 10 → V)
    (hv : ∀j,0<v j) (hmask : v 1+v 2<q p v)
    (hz : InternalZero (output79 (integerPorts p)) v) :
    (∀j,0<flip p v j) ∧ InternalZero (graph (integerPorts p)).output (flip p v) := by
  refine ⟨flip_positive p v hv hmask,?_⟩
  apply (internalZero_iff _ _).mpr
  have hh := coordinate_identity (integerPorts p) (fun j => ι (v j))
  rw [oldVal_cast p v hmask] at hh
  rw [←hh]
  exact (internalZero_iff _ _).mp hz

#print axioms TenPositive79Guarded.guarded_to_old
end TenPositive79Guarded
