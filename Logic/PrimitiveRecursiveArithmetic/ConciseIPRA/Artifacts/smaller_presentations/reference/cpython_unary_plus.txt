static PyObject *
long_long(PyObject *v)
{
    if (PyLong_CheckExact(v)) {
        return Py_NewRef(v);
    }
    else {
        return _PyLong_Copy((PyLongObject *)v);
    }
}

Exact ints return Py_NewRef(v) from long_long; the unary-positive slot uses long_long. Subtypes, including bool, are copied to an exact int instead.
