namespace IPRAFormalLanguages.RegularOutline

abbrev Word := List Nat

structure DfaCode where
  alphabet : Nat
  states : Nat
  initial : Nat
  table : List Nat                 -- row-major states * alphabet entries
  finals : List Bool               -- exactly states entries
  deriving Repr

structure Edge where
  source : Nat
  letter : Option Nat              -- none is epsilon
  target : Nat
  deriving Repr

structure ENfaCode where
  alphabet : Nat
  states : Nat
  starts : List Bool
  finals : List Bool
  edges : List Edge
  deriving Repr

def bit (xs : List Bool) (i : Nat) : Bool := (xs[i]?).getD false

def validDFA (M : DfaCode) : Bool :=
  decide (0 < M.alphabet) && decide (0 < M.states) &&
  decide (M.initial < M.states) &&
  (M.table.length == M.states * M.alphabet) &&
  (M.finals.length == M.states) &&
  M.table.all (fun q => decide (q < M.states))

def validENFA (M : ENfaCode) : Bool :=
  decide (0 < M.alphabet) && decide (0 < M.states) &&
  (M.starts.length == M.states) && (M.finals.length == M.states) &&
  M.edges.all (fun e => decide (e.source < M.states) &&
    decide (e.target < M.states) &&
    match e.letter with
    | none => true
    | some a => decide (a < M.alphabet))

def validWord (k : Nat) (w : Word) : Bool :=
  w.all (fun a => decide (a < k))

def step (M : DfaCode) (q a : Nat) : Nat :=
  (M.table[q * M.alphabet + a]?).getD 0

def run (M : DfaCode) (q : Nat) (w : Word) : Nat :=
  w.foldl (step M) q

def acceptDFA (M : DfaCode) (w : Word) : Bool :=
  bit M.finals (run M M.initial w)

-- Candidate semantic proof; to be checked against the pinned Lean version.
theorem run_append (M : DfaCode) (q : Nat) (u v : Word) :
    run M q (u ++ v) = run M (run M q u) v := by
  simp [run, List.foldl_append]

-- Every use below fixes f to an explicitly PR finite-table operation.
def iterate {α : Type} (f : α → α) : Nat → α → α
  | 0, x => x
  | n + 1, x => iterate f n (f x)

def epsilonRound (M : ENfaCode) (S : List Bool) : List Bool :=
  (List.range M.states).map fun r =>
    bit S r || M.edges.any (fun e =>
      decide (e.letter = none) && bit S e.source && (e.target == r))

def epsilonClosure (M : ENfaCode) (S : List Bool) : List Bool :=
  iterate (epsilonRound M) M.states S

def letterStep (M : ENfaCode) (S : List Bool) (a : Nat) : List Bool :=
  epsilonClosure M ((List.range M.states).map fun r =>
    M.edges.any (fun e => decide (e.letter = some a) &&
      bit S e.source && (e.target == r)))

def eRun (M : ENfaCode) (w : Word) : List Bool :=
  w.foldl (letterStep M) (epsilonClosure M M.starts)

def acceptENFA (M : ENfaCode) (w : Word) : Bool :=
  (List.range M.states).any (fun q => bit (eRun M w) q && bit M.finals q)

def maskBit (s q : Nat) : Bool := ((s / 2^q) % 2) == 1

def decodeMask (n s : Nat) : List Bool :=
  (List.range n).map (maskBit s)

def encodeMask (n : Nat) (S : List Bool) : Nat :=
  (List.range n).foldl (fun total q =>
    total + if bit S q then 2^q else 0) 0

def determinize (M : ENfaCode) : DfaCode where
  alphabet := M.alphabet
  states := 2^M.states
  initial := encodeMask M.states (epsilonClosure M M.starts)
  table := (List.range (2^M.states)).flatMap fun s =>
    (List.range M.alphabet).map fun a =>
      encodeMask M.states (letterStep M (decodeMask M.states s) a)
  finals := (List.range (2^M.states)).map fun s =>
    (List.range M.states).any (fun q => maskBit s q && bit M.finals q)

-- A semantic regex view. Arithmetic syntax instead uses postfix node tables
-- with child indices < the current index and a PR well-formedness checker.
-- The checker must also enforce one final root, exactly one parent per
-- nonroot node, and connectivity. Child-index order alone permits DAG sharing;
-- reusing the same Thompson child in (times P P) would be unsound. An alternative is
-- explicitly unfolding DAG sharing before construction and counting the
-- unfolded tree size in the 2s state bound.
inductive Regex where
  | zero | epsilon | symbol (a : Nat)
  | plus (p q : Regex)
  | times (p q : Regex)
  | star (p : Regex)
  deriving Repr

-- The interval-table implementation for Regex is specified as follows:
--   zero[i,j] = false; epsilon[i,j] = (i=j)
--   symbol a[i,j] = (j=i+1 and w[i]=a)
--   plus P Q[i,j] = P[i,j] or Q[i,j]
--   times P Q[i,j] = OR_{i<=t<=j} (P[i,t] and Q[t,j])
--   star P[i,j] = reachability from i to j in the DAG with edges
--                 s->t iff i<=s<t<=j and P[s,t], allowing the empty path.
-- Every table has at most nodeCount * (wordLength+1)^2 entries. Children are
-- evaluated first. These equations are the complete algorithmic definition;
-- implementing their finite loops is the next syntactic engineering step.

structure RegularObligations where
  mask_roundtrip : ∀ n S,
    S.length = n → decodeMask n (encodeMask n S) = S
  determinize_valid : ∀ M,
    validENFA M = true → validDFA (determinize M) = true
  subset_correct : ∀ M w,
    validENFA M = true → validWord M.alphabet w = true →
    acceptDFA (determinize M) w = acceptENFA M w
  epsilon_idempotent : ∀ M S,
    validENFA M = true → S.length = M.states →
    epsilonClosure M (epsilonClosure M S) = epsilonClosure M S
